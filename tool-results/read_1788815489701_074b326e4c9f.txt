    72→function parseUserAgent(ua: string | null): { device: string; browser: string } {
    73→  if (!ua) return { device: 'Unknown device', browser: 'Unknown browser' };
    74→
    75→  let device = 'Desktop';
    76→  if (/iPhone|iPad/.test(ua)) device = 'iOS';
    77→  else if (/Android/.test(ua)) device = 'Android';
    78→  else if (/Mobile|Windows Phone/.test(ua)) device = 'Mobile';
    79→
    80→  let browser = 'Unknown';
    81→  if (/Edg\//.test(ua)) browser = 'Edge';
    82→  else if (/Chrome\//.test(ua) && !/Edg\//.test(ua)) browser = 'Chrome';
    83→  else if (/Firefox\//.test(ua)) browser = 'Firefox';
    84→  else if (/Safari\//.test(ua) && !/Chrome\//.test(ua)) browser = 'Safari';
    85→
    86→  return { device, browser };
    87→}

    89→function formatRelativeTime(dateStr: string): string {
    90→  const date = new Date(dateStr);
    91→  const now = new Date();
    92→  const diffMs = now.getTime() - date.getTime();
    93→  const diffMin = Math.floor(diffMs / 60_000);
    94→  const diffHr = Math.floor(diffMs / 3_600_000);
    95→  const diffDay = Math.floor(diffMs / 86_400_000);
    96→
    97→  if (diffMin < 1) return 'Just now';
    98→  if (diffMin < 60) return `${diffMin}m ago`;
    99→  if (diffHr < 24) return `${diffHr}h ago`;
   100→  if (diffDay < 7) return `${diffDay}d ago`;
   101→  return date.toLocaleDateString();
   102→}
   103→
   104→function formatDate(dateStr: string): string {
   105→  return new Date(dateStr).toLocaleDateString(undefined, {
   106→    year: 'numeric', month: 'short', day: 'numeric', hour: '2-digit', minute: '2-digit',
   107→  });
   108→}

   307→function PasswordTab() {
   308→  const [currentPassword, setCurrentPassword] = useState('');
   309→  const [newPassword, setNewPassword] = useState('');
   310→  const [confirmPassword, setConfirmPassword] = useState('');
   311→  const [showCurrent, setShowCurrent] = useState(false);
   312→  const [showNew, setShowNew] = useState(false);
   313→  const [showConfirm, setShowConfirm] = useState(false);
   314→  const [loading, setLoading] = useState(false);
   315→
   316→  // Password strength indicators
   317→  const hasMinLength = newPassword.length >= 8;
   318→  const hasUppercase = /[A-Z]/.test(newPassword);
   319→  const hasLowercase = /[a-z]/.test(newPassword);
   320→  const hasNumber = /\d/.test(newPassword);
   321→  const hasSpecial = /[^A-Za-z0-9]/.test(newPassword);
   322→  const passwordsMatch = newPassword === confirmPassword && newPassword.length > 0;
   323→  const isFormValid = hasMinLength && hasUppercase && hasLowercase && hasNumber && hasSpecial && passwordsMatch && currentPassword.length > 0;
   324→
   325→  const strengthCount = [hasMinLength, hasUppercase, hasLowercase, hasNumber, hasSpecial].filter(Boolean).length;
   326→  const strengthLabel = strengthCount <= 2 ? 'Weak' : strengthCount <= 4 ? 'Fair' : 'Strong';
   327→  const strengthColor = strengthCount <= 2 ? 'text-red-500' : strengthCount <= 4 ? 'text-amber-500' : 'text-emerald-500';
   328→
   329→  const handleSubmit = async () => {
   330→    if (!isFormValid) return;
   331→    setLoading(true);
   332→    try {
   333→      const data = await apiFetch<{ success: boolean; message: string; sessionsRevoked: number }>(
   334→        '/api/users/me/password',
   335→        {
   336→          method: 'PATCH',
   337→          body: JSON.stringify({ currentPassword, newPassword }),
   338→        },
   339→      );
   340→
   341→      if (data.success) {
   342→        toast.success(data.message, {
   343→          description: data.sessionsRevoked > 0
   344→            ? `${data.sessionsRevoked} other session(s) have been revoked for security.`
   345→            : undefined,
   346→        });
   347→        setCurrentPassword('');
   348→        setNewPassword('');
   349→        setConfirmPassword('');
   350→      }
   351→    } catch (err) {
   352→      toast.error(err instanceof Error ? err.message : 'Failed to change password');
   353→    } finally {
   354→      setLoading(false);
   355→    }
   356→  };
   357→
   358→  return (
   359→    <div className="space-y-6">
   360→      <Card>
   361→        <CardHeader>
   362→          <CardTitle className="text-base">Change Password</CardTitle>
   363→          <CardDescription>
   364→            After changing your password, all other active sessions will be revoked for security.
   365→          </CardDescription>
   366→        </CardHeader>
   367→        <CardContent className="space-y-4">
   368→          {/* Current Password */}
   369→          <div className="space-y-2">
   370→            <Label htmlFor="current-password">Current Password</Label>
   371→            <div className="relative">
   372→              <Input
   373→                id="current-password"
   374→                type={showCurrent ? 'text' : 'password'}
   375→                value={currentPassword}
   376→                onChange={(e) => setCurrentPassword(e.target.value)}
   377→                placeholder="Enter your current password"
   378→                className="pr-10"
   379→              />
   380→              <button
   381→                type="button"
   382→                onClick={() => setShowCurrent(!showCurrent)}
   383→                className="absolute right-3 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground"
   384→              >
   385→                {showCurrent ? <EyeOff className="h-4 w-4" /> : <Eye className="h-4 w-4" />}
   386→              </button>
   387→            </div>
   388→          </div>
   389→
   390→          <Separator />
   391→
   392→          {/* New Password */}
   393→          <div className="space-y-2">
   394→            <Label htmlFor="new-password">New Password</Label>
   395→            <div className="relative">
   396→              <Input
   397→                id="new-password"
   398→                type={showNew ? 'text' : 'password'}
   399→                value={newPassword}
   400→                onChange={(e) => setNewPassword(e.target.value)}
   401→                placeholder="Enter a strong password"
   402→                className="pr-10"
   403→              />
   404→              <button
   405→                type="button"
   406→                onClick={() => setShowNew(!showNew)}
   407→                className="absolute right-3 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground"
   408→              >
   409→                {showNew ? <EyeOff className="h-4 w-4" /> : <Eye className="h-4 w-4" />}
   410→              </button>
   411→            </div>
   412→
   413→            {/* Strength Indicator */}
   414→            {newPassword.length > 0 && (
   415→              <div className="space-y-2">
   416→                <div className="flex gap-1">
   417→                  {[1, 2, 3, 4, 5].map((i) => (
   418→                    <div
   419→                      key={i}
   420→                      className={`h-1.5 flex-1 rounded-full transition-colors ${
   421→                        i <= strengthCount
   422→                          ? strengthCount <= 2 ? 'bg-red-500' : strengthCount <= 4 ? 'bg-amber-500' : 'bg-emerald-500'
   423→                          : 'bg-muted'
   424→                      }`}
   425→                    />
   426→                  ))}
   427→                </div>
   428→                <p className={`text-xs ${strengthColor}`}>{strengthLabel}</p>
   429→
   430→                {/* Requirements */}
   431→                <div className="grid grid-cols-1 gap-1 text-xs">
   432→                  {[
   433→                    { met: hasMinLength, label: 'At least 8 characters' },
   434→                    { met: hasUppercase, label: 'One uppercase letter' },
   435→                    { met: hasLowercase, label: 'One lowercase letter' },
   436→                    { met: hasNumber, label: 'One number' },

   440→                      {met ? <CheckCircle2 className="h-3 w-3" /> : <AlertCircle className="h-3 w-3" />}
   441→                      {label}
   442→                    </div>
   443→                  ))}
   444→                </div>
   445→              </div>
   446→            )}
   447→          </div>
   448→
   449→          {/* Confirm New Password */}
   450→          <div className="space-y-2">
   451→            <Label htmlFor="confirm-password">Confirm New Password</Label>

   474→          <div className="flex justify-end">

   485→      </Card>
   486→    </div>

   492→function SessionsTab() {
   493→  const [sessions, setSessions] = useState<SessionInfo[]>([]);
   494→  const [loading, setLoading] = useState(true);
   495→  const [revokeId, setRevokeId] = useState<string | null>(null);
   496→  const [revoking, setRevoking] = useState(false);
   497→
   498→  const fetchSessions = useCallback(async () => {
   499→    setLoading(true);
   500→    try {
   501→      const data = await apiFetch<{ success: boolean; sessions: SessionInfo[] }>('/api/users/me/sessions');
   502→      if (data.success) {
   503→        setSessions(data.sessions);
   504→      }
   505→    } catch {
   506→      toast.error('Failed to load sessions');
   507→    } finally {
   508→      setLoading(false);
   509→    }
   510→  }, []);
   511→
   512→  useEffect(() => { fetchSessions(); }, [fetchSessions]);
   513→
   514→  const handleRevoke = async (sessionId: string) => {
   515→    setRevoking(true);
   516→    try {
   517→      const data = await apiFetch<{ success: boolean; message: string }>(
   518→        `/api/users/me/sessions/${sessionId}`,
   519→        { method: 'DELETE' },
   520→      );
   521→      if (data.success) {
   522→        toast.success('Session revoked');
   523→        setSessions((prev) => prev.filter((s) => s.id !== sessionId));
   524→      }
   525→    } catch (err) {
   526→      toast.error(err instanceof Error ? err.message : 'Failed to revoke session');
   527→    } finally {
   528→      setRevoking(false);
   529→      setRevokeId(null);
   530→    }
   531→  };
   532→
   533→  return (
   534→    <div className="space-y-6">
   535→      <Card>
   536→        <CardHeader>
   537→          <CardTitle className="text-base">Active Sessions</CardTitle>
   538→          <CardDescription>
   539→            Devices and browsers where you&apos;re currently logged in. You can revoke any session to sign out of that device.
   540→          </CardDescription>
   541→        </CardHeader>
   542→        <CardContent>
   543→          {loading ? (
   544→            <div className="space-y-3">
   545→              {[1, 2, 3].map((i) => (
   546→                <div key={i} className="h-16 bg-muted animate-pulse rounded-lg" />
   547→              ))}
   548→            </div>
   549→          ) : sessions.length === 0 ? (
   550→            <p className="text-sm text-muted-foreground py-4 text-center">No active sessions found.</p>
   551→          ) : (
   552→            <div className="space-y-3">
   553→              {sessions.map((session) => {
   554→                const { device, browser } = parseUserAgent(session.userAgent);
   555→                return (
   556→                  <div
   557→                    key={session.id}
   558→                    className={`flex items-center gap-4 p-4 rounded-lg border ${
   559→                      session.isCurrent ? 'border-emerald-300 bg-emerald-50/50' : 'border-border'
   560→                    }`}
   561→                  >
   562→                    {/* Device Icon */}
   563→                    <div className={`flex-shrink-0 h-10 w-10 rounded-full flex items-center justify-center ${
   564→                      session.isCurrent ? 'bg-emerald-100 text-emerald-600' : 'bg-muted text-muted-foreground'
   565→                    }`}>
   566→                      {device === 'iOS' || device === 'Android' || device === 'Mobile'
   567→                        ? <Smartphone className="h-5 w-5" />
   568→                        : <Monitor className="h-5 w-5" />}
   569→                    </div>
   570→
   571→                    {/* Session Info */}
   572→                    <div className="flex-1 min-w-0">
   573→                      <div className="flex items-center gap-2">
   574→                        <p className="text-sm font-medium truncate">
   575→                          {browser} on {device}
   576→                        </p>
   577→                        {session.isCurrent && (
   578→                          <Badge variant="outline" className="text-xs text-emerald-600 border-emerald-300 bg-emerald-50">
   579→                            Current
   580→                          </Badge>
   581→                        )}
   582→                      </div>
   583→                      <div className="flex items-center gap-3 mt-1 text-xs text-muted-foreground">
   584→                        {session.ipAddress && (
   585→                          <span className="flex items-center gap-1">
   586→                            <Globe className="h-3 w-3" />
   587→                            {session.ipAddress}
   588→                          </span>
   589→                        )}
   590→                        <span className="flex items-center gap-1">
   591→                          <Clock className="h-3 w-3" />
   592→                          {formatRelativeTime(session.lastActivityAt)}
   593→                        </span>
   594→                        <span className="hidden sm:inline">
   595→                          Created {formatDate(session.createdAt)}
   596→                        </span>
   597→                      </div>
   598→                    </div>
   599→
   600→                    {/* Revoke Button */}
   601→                    {!session.isCurrent && (
   602→                      <Button
   603→                        variant="ghost"
   604→                        size="sm"
   605→                        onClick={() => setRevokeId(session.id)}
   606→                        className="text-red-500 hover:text-red-700 hover:bg-red-50 flex-shrink-0"
   607→                      >
   608→                        <Trash2 className="h-4 w-4" />
   609→                      </Button>
   610→                    )}
   611→                  </div>
   612→                );
   613→              })}
   614→            </div>
   615→          )}
   616→        </CardContent>
   617→      </Card>
   618→
   619→      {/* Revoke Confirmation Dialog */}
   620→      <AlertDialog open={revokeId !== null} onOpenChange={(open) => !open && setRevokeId(null)}>
   621→        <AlertDialogContent>
   622→          <AlertDialogHeader>
   623→            <AlertDialogTitle>Revoke Session?</AlertDialogTitle>
   624→            <AlertDialogDescription>
   625→              This will sign out the device associated with this session. They will need to log in again to access their account.
   626→            </AlertDialogDescription>
   627→          </AlertDialogHeader>
   628→          <AlertDialogFooter>
   629→            <AlertDialogCancel disabled={revoking}>Cancel</AlertDialogCancel>
   630→            <AlertDialogAction
   631→              onClick={() => revokeId && handleRevoke(revokeId)}
   632→              disabled={revoking}
   633→              className="bg-red-600 hover:bg-red-700 text-white"
   634→            >
   635→              {revoking ? <Loader2 className="h-4 w-4 mr-2 animate-spin" /> : null}
   636→              Revoke Session
   637→            </AlertDialogAction>
   638→          </AlertDialogFooter>
   639→        </AlertDialogContent>
   640→      </AlertDialog>
   641→    </div>
   642→  );
   643→}
