    99→function parseUserAgent(ua: string | null): { device: string; browser: string } {
   100→  if (!ua) return { device: 'Unknown device', browser: 'Unknown browser' };
   101→
   102→  let device = 'Desktop';
   103→  if (/iPhone|iPad/.test(ua)) device = 'iOS';
   104→  else if (/Android/.test(ua)) device = 'Android';
   105→  else if (/Mobile|Windows Phone/.test(ua)) device = 'Mobile';
   106→
   107→  let browser = 'Unknown';
   108→  if (/Edg\//.test(ua)) browser = 'Edge';
   109→  else if (/Chrome\//.test(ua) && !/Edg\//.test(ua)) browser = 'Chrome';
   110→  else if (/Firefox\//.test(ua)) browser = 'Firefox';
   111→  else if (/Safari\//.test(ua) && !/Chrome\//.test(ua)) browser = 'Safari';
   112→
   113→  return { device, browser };
   114→}

   116→function formatRelativeTime(dateStr: string): string {
   117→  const date = new Date(dateStr);
   118→  const now = new Date();
   119→  const diffMs = now.getTime() - date.getTime();
   120→  const diffMin = Math.floor(diffMs / 60_000);
   121→  const diffHr = Math.floor(diffMs / 3_600_000);
   122→  const diffDay = Math.floor(diffMs / 86_400_000);
   123→
   124→  if (diffMin < 1) return 'Just now';
   125→  if (diffMin < 60) return `${diffMin}m ago`;
   126→  if (diffHr < 24) return `${diffHr}h ago`;
   127→  if (diffDay < 7) return `${diffDay}d ago`;
   128→  return date.toLocaleDateString();
   129→}

   134→  });
   135→}

   334→function PasswordTab() {
   335→  const [currentPassword, setCurrentPassword] = useState('');
   336→  const [newPassword, setNewPassword] = useState('');
   337→  const [confirmPassword, setConfirmPassword] = useState('');
   338→  const [showCurrent, setShowCurrent] = useState(false);
   339→  const [showNew, setShowNew] = useState(false);
   340→  const [showConfirm, setShowConfirm] = useState(false);
   341→  const [loading, setLoading] = useState(false);
   342→
   343→  // Password strength indicators
   344→  const hasMinLength = newPassword.length >= 8;
   345→  const hasUppercase = /[A-Z]/.test(newPassword);
   346→  const hasLowercase = /[a-z]/.test(newPassword);
   347→  const hasNumber = /\d/.test(newPassword);
   348→  const hasSpecial = /[^A-Za-z0-9]/.test(newPassword);
   349→  const passwordsMatch = newPassword === confirmPassword && newPassword.length > 0;
   350→  const isFormValid = hasMinLength && hasUppercase && hasLowercase && hasNumber && hasSpecial && passwordsMatch && currentPassword.length > 0;
   351→
   352→  const strengthCount = [hasMinLength, hasUppercase, hasLowercase, hasNumber, hasSpecial].filter(Boolean).length;
   353→  const strengthLabel = strengthCount <= 2 ? 'Weak' : strengthCount <= 4 ? 'Fair' : 'Strong';
   354→  const strengthColor = strengthCount <= 2 ? 'text-red-500' : strengthCount <= 4 ? 'text-amber-500' : 'text-emerald-500';
   355→
   356→  const handleSubmit = async () => {
   357→    if (!isFormValid) return;
   358→    setLoading(true);
   359→    try {
   360→      const data = await apiFetch<{ success: boolean; message: string; sessionsRevoked: number }>(
   361→        '/api/users/me/password',
   362→        {
   363→          method: 'PATCH',
   364→          body: JSON.stringify({ currentPassword, newPassword }),
   365→        },
   366→      );
   367→
   368→      if (data.success) {
   369→        toast.success(data.message, {
   370→          description: data.sessionsRevoked > 0
   371→            ? `${data.sessionsRevoked} other session(s) have been revoked for security.`
   372→            : undefined,
   373→        });
   374→        setCurrentPassword('');
   375→        setNewPassword('');
   376→        setConfirmPassword('');
   377→      }
   378→    } catch (err) {
   379→      toast.error(err instanceof Error ? err.message : 'Failed to change password');
   380→    } finally {
   381→      setLoading(false);
   382→    }
   383→  };
   384→
   385→  return (
   386→    <div className="space-y-6">
   387→      <Card>
   388→        <CardHeader>
   389→          <CardTitle className="text-base">Change Password</CardTitle>
   390→          <CardDescription>
   391→            After changing your password, all other active sessions will be revoked for security.
   392→          </CardDescription>
   393→        </CardHeader>
   394→        <CardContent className="space-y-4">
   395→          {/* Current Password */}
   396→          <div className="space-y-2">
   397→            <Label htmlFor="current-password">Current Password</Label>
   398→            <div className="relative">
   399→              <Input
   400→                id="current-password"
   401→                type={showCurrent ? 'text' : 'password'}
   402→                value={currentPassword}
   403→                onChange={(e) => setCurrentPassword(e.target.value)}
   404→                placeholder="Enter your current password"
   405→                className="pr-10"
   406→              />
   407→              <button
   408→                type="button"
   409→                onClick={() => setShowCurrent(!showCurrent)}
   410→                className="absolute right-3 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground"

   412→                {showCurrent ? <EyeOff className="h-4 w-4" /> : <Eye className="h-4 w-4" />}

   416→
   417→          <Separator />

   421→            <Label htmlFor="new-password">New Password</Label>

   423→              <Input

   519→function SessionsTab() {
   520→  const [sessions, setSessions] = useState<SessionInfo[]>([]);
   521→  const [loading, setLoading] = useState(true);
   522→  const [revokeId, setRevokeId] = useState<string | null>(null);
   523→  const [revoking, setRevoking] = useState(false);
   524→
   525→  const fetchSessions = useCallback(async () => {
   526→    setLoading(true);
   527→    try {
   528→      const data = await apiFetch<{ success: boolean; sessions: SessionInfo[] }>('/api/users/me/sessions');
   529→      if (data.success) {
   530→        setSessions(data.sessions);
   531→      }
   532→    } catch {
   533→      toast.error('Failed to load sessions');
   534→    } finally {
   535→      setLoading(false);
   536→    }
   537→  }, []);
   538→
   539→  useEffect(() => { fetchSessions(); }, [fetchSessions]);
   540→
   541→  const handleRevoke = async (sessionId: string) => {
   542→    setRevoking(true);
   543→    try {
   544→      const data = await apiFetch<{ success: boolean; message: string }>(
   545→        `/api/users/me/sessions/${sessionId}`,
   546→        { method: 'DELETE' },
   547→      );
   548→      if (data.success) {
   549→        toast.success('Session revoked');
   550→        setSessions((prev) => prev.filter((s) => s.id !== sessionId));
   551→      }
   552→    } catch (err) {
   553→      toast.error(err instanceof Error ? err.message : 'Failed to revoke session');
   554→    } finally {
   555→      setRevoking(false);
   556→      setRevokeId(null);
   557→    }
   558→  };
   559→
   560→  return (
   561→    <div className="space-y-6">
   562→      <Card>
   563→        <CardHeader>
   564→          <CardTitle className="text-base">Active Sessions</CardTitle>
   565→          <CardDescription>
   566→            Devices and browsers where you&apos;re currently logged in. You can revoke any session to sign out of that device.
   567→          </CardDescription>
   568→        </CardHeader>
   569→        <CardContent>
   570→          {loading ? (
   571→            <div className="space-y-3">
   572→              {[1, 2, 3].map((i) => (
   573→                <div key={i} className="h-16 bg-muted animate-pulse rounded-lg" />
   574→              ))}
   575→            </div>
   576→          ) : sessions.length === 0 ? (
