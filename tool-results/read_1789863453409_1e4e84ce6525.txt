    19→describe('Invitation Token Security', () => {
    20→  it('generates a secure random token', () => {
    21→    const token = generateSecureToken();
    22→    expect(token).toBeDefined();
    23→    expect(typeof token).toBe('string');
    24→    expect(token.length).toBeGreaterThan(0);
    25→    // UUID format
    26→    expect(token).toMatch(/^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/);
    27→  });
    28→
    29→  it('hashes tokens using SHA-256 (not stored as plaintext)', async () => {
    30→    const rawToken = generateSecureToken();
    31→    const hashedToken = await hashToken(rawToken);
    32→
    33→    // Hash should be different from raw token
    34→    expect(hashedToken).not.toBe(rawToken);
    35→
    36→    // Hash should be 64 hex chars (SHA-256)
    37→    expect(hashedToken).toMatch(/^[0-9a-f]{64}$/);
    38→  });
    39→
    40→  it('verifyTokenHash correctly validates a token against its hash', async () => {
    41→    const rawToken = generateSecureToken();
    42→    const hashedToken = await hashToken(rawToken);
    43→
    44→    const isValid = await verifyTokenHash(rawToken, hashedToken);
    45→    expect(isValid).toBe(true);
    46→
    47→    const isInvalid = await verifyTokenHash('wrong-token', hashedToken);
    48→    expect(isInvalid).toBe(false);
    49→  });
    50→
    51→  it('different tokens produce different hashes', async () => {
    52→    const token1 = generateSecureToken();
    53→    const token2 = generateSecureToken();
    54→    const hash1 = await hashToken(token1);
    55→    const hash2 = await hashToken(token2);
    56→
    57→    expect(hash1).not.toBe(hash2);
    58→  });
    59→
    60→  it('same token always produces the same hash', async () => {
    61→    const token = generateSecureToken();
    62→    const hash1 = await hashToken(token);
    63→    const hash2 = await hashToken(token);
    64→
    65→    expect(hash1).toBe(hash2);
    66→  });
