    59→    59→

    61→    61→  PUBLISHED: 'bg-green-100 text-green-700',
    62→    62→  DRAFT: 'bg-gray-100 text-gray-700',
    63→    63→  PENDING: 'bg-yellow-100 text-yellow-700',
    64→    64→  CANCELLED: 'bg-red-100 text-red-700',

    68→    68→
    69→    69→export function OrganizerEditEvent() {
    70→    70→  const { selectedEventId, navigate, user } = useAppStore();
    71→    71→  const [loading, setLoading] = useState(true);
    72→    72→  const [saving, setSaving] = useState(false);
    73→    73→  const [categories, setCategories] = useState<Category[]>([]);
    74→    74→  const [eventStatus, setEventStatus] = useState<string>('DRAFT');
    75→    75→  const [statusDialog, setStatusDialog] = useState<{ open: boolean; target: string }>({
    76→    76→    open: false,
    77→    77→    target: '',
    78→    78→  });
    79→    79→
    80→    80→  const [form, setForm] = useState({
    81→    81→    title: '',
    82→    82→    description: '',
    83→    83→    shortDescription: '',
    84→    84→    categoryId: '',
    85→    85→    coverImage: '',
    86→    86→    startDate: '',
    87→    87→    endDate: '',
    88→    88→    startTime: '',
    89→    89→    endTime: '',
    90→    90→    timezone: 'UTC',
    91→    91→    venueName: '',
    92→    92→    venueAddress: '',
    93→    93→    venueCity: '',
    94→    94→    venueState: '',
    95→    95→    venueCountry: '',
    96→    96→    isVirtual: false,
    97→    97→    virtualUrl: '',
    98→    98→    isPaid: false,
    99→    99→    currency: 'USD',
   100→   100→    capacity: '',
   101→   101→    tags: '',
   102→   102→  });
   103→   103→
   104→   104→  const [ticketTypes, setTicketTypes] = useState<TicketTypeForm[]>([]);
   105→   105→
   106→   106→  // Load event data and categories
   107→   107→  useEffect(() => {
   108→   108→    if (!selectedEventId) return;
   109→   109→    setLoading(true);
   110→   110→    Promise.all([
   111→   111→      apiFetch<{
   112→   112→        event: {
   113→   113→          title: string;
   114→   114→          description: string;
   115→   115→          shortDescription: string | null;
   116→   116→          coverImage: string | null;
   117→   117→          startDate: string;
   118→   118→          endDate: string | null;
   119→   119→          startTime: string | null;
   120→   120→          endTime: string | null;
   121→   121→          timezone: string;
   122→   122→          venueName: string | null;
   123→   123→          venueAddress: string | null;
   124→   124→          venueCity: string | null;
   125→   125→          venueState: string | null;
   126→   126→          venueCountry: string | null;
   127→   127→          isVirtual: boolean;
   128→   128→          virtualUrl: string | null;
   129→   129→          isPaid: boolean;
   130→   130→          currency: string;
   131→   131→          capacity: number | null;
   132→   132→          status: string;
   133→   133→          categoryId: string | null;
   134→   134→          category: { id: string; name: string; slug: string } | null;
   135→   135→          tags: { tag: { name: string } }[];
   136→   136→          ticketTypes: {
   137→   137→            id: string;
   138→   138→            name: string;
   139→   139→            price: number;
   140→   140→            quantity: number;
   141→   141→            soldCount: number;
   142→   142→          }[];
   143→   143→        };
   144→   144→      }>(`/api/events/${selectedEventId}`),
   145→   145→      apiFetch<{ categories: Category[] }>('/api/categories'),
   146→   146→    ])
   147→   147→      .then(([eventData, catData]) => {
   148→   148→        const e = eventData.event;
   149→   149→        setForm({
   150→   150→          title: e.title,
   151→   151→          description: e.description,
   152→   152→          shortDescription: e.shortDescription || '',
   153→   153→          categoryId: e.category?.id || '',
   154→   154→          coverImage: e.coverImage || '',
   155→   155→          startDate: e.startDate ? new Date(e.startDate).toISOString().split('T')[0] : '',
   156→   156→          endDate: e.endDate ? new Date(e.endDate).toISOString().split('T')[0] : '',
   157→   157→          startTime: e.startTime || '',
   158→   158→          endTime: e.endTime || '',
   159→   159→          timezone: e.timezone || 'UTC',
   160→   160→          venueName: e.venueName || '',
   161→   161→          venueAddress: e.venueAddress || '',
   162→   162→          venueCity: e.venueCity || '',
   163→   163→          venueState: e.venueState || '',
   164→   164→          venueCountry: e.venueCountry || '',
   165→   165→          isVirtual: e.isVirtual,
   166→   166→          virtualUrl: e.virtualUrl || '',
   167→   167→          isPaid: e.isPaid,
   168→   168→          currency: e.currency || 'USD',
   169→   169→          capacity: e.capacity ? String(e.capacity) : '',
   170→   170→          tags: e.tags?.map((t: { tag: { name: string } }) => t.tag.name).join(', ') || '',
   171→   171→        });
   172→   172→        setEventStatus(e.status);
   173→   173→        setTicketTypes(
   174→   174→          e.ticketTypes.map((tt) => ({
   175→   175→            id: tt.id,
   176→   176→            name: tt.name,
   177→   177→            price: String(tt.price),
   178→   178→            quantity: String(tt.quantity),
   179→   179→          }))
   180→   180→        );
   181→   181→        setCategories(catData.categories || []);
   182→   182→      })
   183→   183→      .catch(() => toast.error('Failed to load event'))
   184→   184→      .finally(() => setLoading(false));
   185→   185→  }, [selectedEventId]);
   186→   186→
   187→   187→  const updateForm = (key: string, value: string | boolean) =>
   188→   188→    setForm((prev) => ({ ...prev, [key]: value }));
   189→   189→
   190→   190→  const addTicketType = () =>
   191→   191→    setTicketTypes((prev) => [
   192→   192→      ...prev,
   193→   193→      { name: '', price: '0', quantity: '50', isNew: true },
   194→   194→    ]);
   195→   195→  const removeTicketType = (i: number) => {
   196→   196→    const tt = ticketTypes[i];
   197→   197→    if (tt.isNew) {
   198→   198→      setTicketTypes((prev) => prev.filter((_, idx) => idx !== i));
   199→   199→    } else {
   200→   200→      // Mark existing for deletion
   201→   201→      setTicketTypes((prev) =>
   202→   202→        prev.map((t, idx) => (idx === i ? { ...t, toDelete: true } : t))
   203→   203→      );
   204→   204→    }
   205→   205→  };
   206→   206→  const restoreTicketType = (i: number) => {
   207→   207→    setTicketTypes((prev) =>
   208→   208→      prev.map((t, idx) => (idx === i ? { ...t, toDelete: false } : t))
   209→   209→    );
   210→   210→  };
   211→   211→  const updateTicketType = (i: number, key: keyof TicketTypeForm, value: string) => {
   212→   212→    setTicketTypes((prev) =>
   213→   213→      prev.map((t, idx) => (idx === i ? { ...t, [key]: value } : t))
   214→   214→    );
   215→   215→  };
   216→   216→
   217→   217→  const handleSubmit = async (e: React.FormEvent) => {
   218→   218→    e.preventDefault();
   219→   219→    if (!form.title || !form.startDate) {
   220→   220→      toast.error('Please fill in required fields (title, date)');
   221→   221→      return;
   222→   222→    }
   223→   223→    if (form.isVirtual && !form.virtualUrl) {
   224→   224→      toast.error('Please provide a virtual event URL');
   225→   225→      return;
   226→   226→    }
   227→   227→
   228→   228→    setSaving(true);
   229→   229→    try {
   230→   230→      // Build PATCH body (only send changed fields)
   231→   231→      const body: Record<string, unknown> = {
   232→   232→        title: form.title,
   233→   233→        description: form.description,
   234→   234→        shortDescription: form.shortDescription || undefined,
   235→   235→        coverImage: form.coverImage || undefined,
   236→   236→        startDate: form.startDate,
   237→   237→        endDate: form.endDate || undefined,
   238→   238→        startTime: form.startTime || undefined,
   239→   239→        endTime: form.endTime || undefined,
   240→   240→        timezone: form.timezone,
   241→   241→        venueName: form.isVirtual ? undefined : form.venueName,
   242→   242→        venueAddress: form.isVirtual ? undefined : form.venueAddress,
   243→   243→        venueCity: form.isVirtual ? undefined : form.venueCity,
   244→   244→        venueState: form.isVirtual ? undefined : form.venueState,
   245→   245→        venueCountry: form.isVirtual ? undefined : form.venueCountry,
   246→   246→        isVirtual: form.isVirtual,
   247→   247→        virtualUrl: form.isVirtual ? form.virtualUrl : undefined,
   248→   248→        isPaid: form.isPaid,
   249→   249→        currency: form.currency,
   250→   250→        capacity: form.capacity ? parseInt(form.capacity) : undefined,
   251→   251→        categoryId: form.categoryId || undefined,
   252→   252→        tags: form.tags
   253→   253→          ? form.tags.split(',').map((t: string) => t.trim()).filter(Boolean)
   254→   254→          : [],
   255→   255→      };
   256→   256→
   257→   257→      await apiFetch(`/api/events/${selectedEventId}`, {
   258→   258→        method: 'PATCH',
   259→   259→        body: JSON.stringify(body),
   260→   260→      });
   261→   261→
   262→   262→      // Handle ticket type changes
   263→   263→      // Delete marked ticket types
   264→   264→      for (const tt of ticketTypes.filter((t) => t.toDelete && t.id)) {
   265→   265→        await apiFetch(
   266→   266→          `/api/events/${selectedEventId}/ticket-types/${tt.id}`,
   267→   267→          { method: 'DELETE' }
   268→   268→        );
   269→   269→      }
   270→   270→
   271→   271→      // Add new ticket types
   272→   272→      for (const tt of ticketTypes.filter((t) => t.isNew)) {
   273→   273→        await apiFetch(`/api/events/${selectedEventId}/ticket-types`, {
   274→   274→          method: 'POST',
   275→   275→          body: JSON.stringify({
   276→   276→            name: tt.name,
   277→   277→            price: parseFloat(tt.price) || 0,
   278→   278→            quantity: parseInt(tt.quantity) || 50,
   279→   279→            currency: form.currency,
   280→   280→          }),
   281→   281→        });
   282→   282→      }
   283→   283→
   284→   284→      // Update existing ticket types (changed values)
   285→   285→      for (const tt of ticketTypes.filter((t) => t.id && !t.isNew && !t.toDelete)) {
   286→   286→        await apiFetch(
   287→   287→          `/api/events/${selectedEventId}/ticket-types/${tt.id}`,
   288→   288→          {
   289→   289→            method: 'PATCH',
   290→   290→            body: JSON.stringify({
   291→   291→              name: tt.name,
   292→   292→              price: parseFloat(tt.price),
   293→   293→              quantity: parseInt(tt.quantity),
   294→   294→            }),
   295→   295→          }
   296→   296→        );
   297→   297→      }
   298→   298→
   299→   299→      toast.success('Event updated successfully!');
   300→   300→      navigate('organizer-events');
   301→   301→    } catch (err: unknown) {
   302→   302→      const message = err instanceof Error ? err.message : 'Failed to update event';
   303→   303→      toast.error(message);
   304→   304→    } finally {
   305→   305→      setSaving(false);
   306→   306→    }
   307→   307→  };
   308→   308→
   309→   309→  // Status transition handlers
   310→   310→  const handleStatusChange = async (targetStatus: string) => {
   311→   311→    setStatusDialog({ open: false, target: '' });
   312→   312→    setSaving(true);
   313→   313→    try {
   314→   314→      await apiFetch(`/api/events/${selectedEventId}`, {
   315→   315→        method: 'PATCH',
   316→   316→        body: JSON.stringify({ status: targetStatus }),
   317→   317→      });
   318→   318→      setEventStatus(targetStatus);
   319→   319→      toast.success(`Event status changed to ${targetStatus}`);
   320→   320→    } catch (err: unknown) {
   321→   321→      const message = err instanceof Error ? err.message : 'Failed to change status';
   322→   322→      toast.error(message);
   323→   323→    } finally {
   324→   324→      setSaving(false);
   325→   325→    }
   326→   326→  };
   327→   327→
   328→   328→  if (loading) {
   329→   329→    return (
   330→   330→      <div className="space-y-6 max-w-3xl">
   331→   331→        <div className="h-8 w-48 bg-muted animate-pulse rounded" />
   332→   332→        <div className="h-64 bg-muted animate-pulse rounded-xl" />
   333→   333→        <div className="h-48 bg-muted animate-pulse rounded-xl" />
   334→   334→      </div>
   335→   335→    );
   336→   336→  }
   337→   337→
   338→   338→  // Determine available status transitions based on current status
   339→   339→  const getStatusActions = () => {
   340→   340→    if (!user) return [];
   341→   341→    const actions: { target: string; label: string; icon: React.ReactNode; variant: string }[] = [];
   342→   342→
   343→   343→    if (user.role === 'ORGANIZER' || user.role === 'SUPER_ADMIN') {
   344→   344→      if (eventStatus === 'DRAFT') {
   345→   345→        actions.push({
   346→   346→          target: 'PENDING',
   347→   347→          label: 'Submit for Review',
   348→   348→          icon: <Send className="h-4 w-4" />,
   349→   349→          variant: 'default',
   350→   350→        });
   351→   351→      }
   352→   352→      if (eventStatus === 'PENDING') {
   353→   353→        actions.push({
   354→   354→          target: 'DRAFT',
   355→   355→          label: 'Withdraw from Review',
   356→   356→          icon: <XCircle className="h-4 w-4" />,
   357→   357→          variant: 'outline',
   358→   358→        });
   359→   359→      }
   360→   360→      if (eventStatus === 'PUBLISHED') {
   361→   361→        actions.push({
   362→   362→          target: 'CANCELLED',
   363→   363→          label: 'Cancel Event',
   364→   364→          icon: <XCircle className="h-4 w-4" />,
   365→   365→          variant: 'destructive',
   366→   366→        });
   367→   367→      }
   368→   368→    }
   369→   369→
   370→   370→    // SUPER_ADMIN can additionally publish and reject directly
   371→   371→    if (user.role === 'SUPER_ADMIN') {
   372→   372→      if (eventStatus === 'PENDING') {
   373→   373→        actions.push({
   374→   374→          target: 'PUBLISHED',
   375→   375→          label: 'Approve & Publish',
   376→   376→          icon: <Send className="h-4 w-4" />,
   377→   377→          variant: 'default',
   378→   378→        });
   379→   379→        actions.push({
   380→   380→          target: 'REJECTED',
   381→   381→          label: 'Reject Event',
   382→   382→          icon: <XCircle className="h-4 w-4" />,
   383→   383→          variant: 'destructive',
   384→   384→        });
   385→   385→      }
   386→   386→      if (eventStatus === 'DRAFT') {
   387→   387→        actions.push({
   388→   388→          target: 'PUBLISHED',
   389→   389→          label: 'Publish Directly',
   390→   390→          icon: <Send className="h-4 w-4" />,
   391→   391→          variant: 'default',
   392→   392→        });
   393→   393→      }
   394→   394→    }
   395→   395→
   396→   396→    return actions;
   397→   397→  };
   398→   398→
   399→   399→  return (
   400→   400→    <form onSubmit={handleSubmit} className="space-y-6 max-w-3xl">
   401→   401→      <div className="flex items-center justify-between">
   402→   402→        <h1 className="text-2xl font-bold">Edit Event</h1>
   403→   403→        <Badge className={STATUS_COLORS[eventStatus] || ''}>{eventStatus}</Badge>
   404→   404→      </div>
   405→   405→
   406→   406→      {/* Status Workflow Actions */}
   407→   407→      {getStatusActions().length > 0 && (
   408→   408→        <Card>
   409→   409→          <CardHeader>
   410→   410→            <CardTitle className="text-base">Event Status</CardTitle>
   411→   411→            <CardDescription>Change the event status to control visibility</CardDescription>
   412→   412→          </CardHeader>
   413→   413→          <CardContent>
   414→   414→            <div className="flex flex-wrap gap-2">
   415→   415→              {getStatusActions().map((action) => (
   416→   416→                <Button
   417→   417→                  key={action.target}
   418→   418→                  type="button"
   419→   419→                  variant={action.variant === 'destructive' ? 'destructive' : action.variant === 'default' ? 'default' : 'outline'}
   420→   420→                  className={action.variant === 'default' ? 'bg-emerald-600 hover:bg-emerald-700' : ''}
   421→   421→                  onClick={() =>
   422→   422→                    setStatusDialog({ open: true, target: action.target })
   423→   423→                  }
   424→   424→                  disabled={saving}
   425→   425→                >
   426→   426→                  {action.icon}
   427→   427→                  <span className="ml-2">{action.label}</span>
   428→   428→                </Button>
   429→   429→              ))}
   430→   430→            </div>
   431→   431→          </CardContent>
   432→   432→        </Card>
   433→   433→      )}
   434→   434→
   435→   435→      {/* Basic Info */}
   436→   436→      <Card>
   437→   437→        <CardHeader>
   438→   438→          <CardTitle className="flex items-center gap-2">
   439→   439→            <Info className="h-5 w-5 text-emerald-500" /> Basic Information
   440→   440→          </CardTitle>
   441→   441→        </CardHeader>
   442→   442→        <CardContent className="space-y-4">
   443→   443→          <div className="space-y-2">
   444→   444→            <Label htmlFor="title">Title *</Label>
   445→   445→            <Input
   446→   446→              id="title"
   447→   447→              placeholder="Event title"
   448→   448→              value={form.title}
   449→   449→              onChange={(e) => updateForm('title', e.target.value)}
   450→   450→              required
   451→   451→            />
   452→   452→          </div>
   453→   453→          <div className="space-y-2">
   454→   454→            <Label htmlFor="shortDesc">Short Description</Label>
   455→   455→            <Input
   456→   456→              id="shortDesc"
   457→   457→              placeholder="Brief summary (1-2 sentences)"
   458→   458→              value={form.shortDescription}
   459→   459→              onChange={(e) => updateForm('shortDescription', e.target.value)}
   460→   460→            />
   461→   461→          </div>
   462→   462→          <div className="space-y-2">
   463→   463→            <Label htmlFor="desc">Full Description</Label>
   464→   464→            <Textarea
   465→   465→              id="desc"
   466→   466→              placeholder="Detailed event description..."
   467→   467→              rows={5}
   468→   468→              value={form.description}
   469→   469→              onChange={(e) => updateForm('description', e.target.value)}
   470→   470→            />
   471→   471→          </div>
   472→   472→          <div className="grid sm:grid-cols-2 gap-4">
   473→   473→            <div className="space-y-2">
   474→   474→              <Label>Category</Label>
   475→   475→              <Select
   476→   476→                value={form.categoryId}
   477→   477→                onValueChange={(v) => updateForm('categoryId', v)}
   478→   478→              >
   479→   479→                <SelectTrigger>
   480→   480→                  <SelectValue placeholder="Select category" />
   481→   481→                </SelectTrigger>
   482→   482→                <SelectContent>
   483→   483→                  {categories.map((c) => (
   484→   484→                    <SelectItem key={c.id} value={c.id}>
   485→   485→                      {c.name}
   486→   486→                    </SelectItem>
   487→   487→                  ))}
   488→   488→                </SelectContent>
   489→   489→              </Select>
   490→   490→            </div>
   491→   491→            <div className="space-y-2">
   492→   492→              <Label htmlFor="coverImg">Cover Image URL</Label>
   493→   493→              <Input
   494→   494→                id="coverImg"
   495→   495→                placeholder="https://..."
   496→   496→                value={form.coverImage}
   497→   497→                onChange={(e) => updateForm('coverImage', e.target.value)}
   498→   498→              />
   499→   499→            </div>
   500→   500→          </div>
