   418→        await prisma.eventMedia.delete({ where: { id: toDelete.id } });

   421→      const usage = await getEventUsage(eventId1);

   423→      expect(usage.media).toBeLessThan(entitlements.limits.maxMediaPerEvent);
   424→
   425→      // Should NOT throw

   434→      while (usage.media < entitlements.limits.maxMediaPerEvent) {

   859→  describe('maxAttendeesTotal — DEFERRED', () => {
   860→    it('should explicitly document that maxAttendeesTotal enforcement is deferred', () => {
   861→      /**
   862→       * maxAttendeesTotal is DEFERRED because:
   863→       *
   864→       * 1. "Attendee" semantics are ambiguous in the current schema:
   865→       *    - Could mean: count of Ticket records with status VALID
   866→       *    - Could mean: count of Booking records with status CONFIRMED
   867→       *    - Could mean: count of unique userId across bookings
   868→       *    - Could mean: sum of TicketType.soldCount
   869→       *
   870→       * 2. The enforcement point is at booking creation (POST /api/events/:id/book),
   871→       *    which requires summing attendees across ALL events for the organizer,
   872→       *    then comparing against the plan's maxAttendeesTotal.
   873→       *
   874→       * 3. Race conditions: Two concurrent bookings could both pass the check
   875→       *    and both succeed, exceeding the limit. SQLite's single-writer model
   876→       *    provides some serialization, but this is not guaranteed for all
   877→       *    deployment targets (PostgreSQL would need explicit row locking).
   878→       *
   879→       * 4. Refund/cancellation semantics: If a booking is cancelled or refunded,
   880→       *    should that free up capacity? This requires clear business rules.
   881→       *
   882→       * Correct future phase: Phase 4E or later, after payment processing
   883→       * and booking lifecycle are fully defined.
   884→       */
   885→      expect(true).toBe(true);
   886→    });
   887→  });

   890→
   891→  describe('Concurrency Considerations', () => {
   892→    it('should document SQLite concurrency limitations', () => {
   893→      /**
   894→       * SQLite Concurrency Notes for Phase 4D:
   895→       *
   896→       * 1. SQLite uses WAL mode which allows concurrent reads but serializes writes.
   897→       *    This provides natural serialization for write operations, reducing
   898→       *    (but not eliminating) race conditions.
   899→       *
   900→       * 2. For maxMediaPerEvent enforcement, the pattern is:
   901→       *    COUNT (current media) → CHECK (against limit) → CREATE (new media)
   902→       *    Under SQLite, concurrent writes are serialized, so two requests
   903→       *    creating media will execute sequentially, and the second will see
   904→       *    the updated count.
   905→       *
   906→       * 3. For maxTicketsPerEvent, the check happens at ticket type creation,
   907→       *    which is less time-sensitive than booking creation.
   908→       *
   909→       * 4. For production PostgreSQL deployment:
   910→       *    - Use SELECT ... FOR UPDATE within transactions for media/ticket counts
   911→       *    - Use advisory locks for cross-event attendee limits
   912→       *    - Consider using Redis for distributed rate limiting
   913→       */
   914→      expect(true).toBe(true);
