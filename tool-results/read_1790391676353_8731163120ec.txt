   137→// ─── Error Handling ───
   138→
   139→function handleApiError(err: unknown) {
   140→  if (err instanceof ApiFetchError && err.isEntitlementError) {
   141→    if (err.code === 'PLAN_LIMIT_REACHED') {
   142→      const details = err.details as { limitName?: string } | undefined;
   143→      const limitName = details?.limitName || 'item';
   144→      toast.error(`You've reached the ${limitName} limit for your plan. Upgrade to add more.`);
   145→    } else if (['ORGANIZER_NOT_OPERABLE', 'ORGANIZER_SUSPENDED', 'ORGANIZER_DEACTIVATED'].includes(err.code)) {
   146→      toast.error('Your organizer account is not active. Contact support.');
   147→    } else {
   148→      toast.error(err.message);
   149→    }
   150→  } else if (err instanceof Error) {
   151→    toast.error(err.message);
   152→  } else {
   153→    toast.error('An unexpected error occurred');
   154→  }
   155→}
