    97→
    98→export function EventDetail() {
    99→  const { selectedEventId, goBack, navigate, user } = useAppStore();
   100→  const [event, setEvent] = useState<Event | null>(null);
   101→  const [loading, setLoading] = useState(true);
   102→  const [bookDialog, setBookDialog] = useState(false);
   103→  const [selectedTicket, setSelectedTicket] = useState<TicketType | null>(null);
   104→  const [quantity, setQuantity] = useState(1);
   105→  const [booking, setBooking] = useState(false);
   106→  const [relatedEvents, setRelatedEvents] = useState<any[]>([]);
   107→  const [reviewDialog, setReviewDialog] = useState(false);
   108→  const [reviewRating, setReviewRating] = useState(5);
   109→  const [reviewComment, setReviewComment] = useState('');
   110→  const [submittingReview, setSubmittingReview] = useState(false);
   111→  const [userHasReviewed, setUserHasReviewed] = useState(false);
   112→  const [sessions, setSessions] = useState<EventSession[]>([]);
   113→  const [participants, setParticipants] = useState<EventParticipant[]>([]);
   114→  const [media, setMedia] = useState<EventMediaItem[]>([]);
   115→
   116→  useEffect(() => {
   117→    if (!selectedEventId) return;
   118→    setLoading(true);
   119→    apiFetch<{ event: Event }>(`/api/events/${selectedEventId}`)
   120→      .then(data => {
   121→        setEvent(data.event);
   122→        // Check if current user has already reviewed
   123→        if (user && data.event.reviews) {
   124→          setUserHasReviewed(data.event.reviews.some((r: Review) => r.user?.id === user.id));
   125→        }
   126→      })
   127→      .catch(() => toast.error('Failed to load event'))
   128→      .finally(() => setLoading(false));
   129→    // Load sessions, participants, and media for public view
   130→    apiFetch<{ sessions: EventSession[] }>(`/api/events/${selectedEventId}/sessions`)
   131→      .then(data => setSessions(data.sessions))
   132→      .catch(() => {});
   133→    apiFetch<{ participants: EventParticipant[] }>(`/api/events/${selectedEventId}/participants`)
   134→      .then(data => setParticipants(data.participants))
   135→      .catch(() => {});
   136→    apiFetch<{ media: EventMediaItem[] }>(`/api/events/${selectedEventId}/media`)
   137→      .then(data => setMedia(data.media))
   138→      .catch(() => {});
   139→  }, [selectedEventId, user?.id]);
   140→
   141→  useEffect(() => {
   142→    if (!event?.category?.slug) return;
   143→    apiFetch<{ events: any[] }>(`/api/events?category=${event.category.slug}&limit=3&featured=false`)
   144→      .then(data => setRelatedEvents(data.events.filter((e: any) => e.id !== event.id).slice(0, 3)))
   145→      .catch(() => {});
   146→  }, [event?.category?.slug]);
   147→
   148→  const handleBook = (ticket: TicketType) => {
   149→    if (!user) {
   150→      navigate('login');
   151→      toast.error('Please login to book tickets');
   152→      return;
   153→    }
   154→    setSelectedTicket(ticket);
   155→    setQuantity(1);
   156→    setBookDialog(true);
   157→  };
   158→
   159→  const confirmBooking = async () => {
   160→    if (!selectedTicket || !selectedEventId) return;
   161→    setBooking(true);
   162→    try {
   163→      await apiFetch('/api/events/' + selectedEventId + '/book', {
   164→        method: 'POST',
   165→        body: JSON.stringify({ ticketTypeId: selectedTicket.id, quantity }),
   166→      });
   167→      toast.success('Booking confirmed!');
   168→      setBookDialog(false);
   169→      // Refresh event data
   170→      const data = await apiFetch<{ event: Event }>(`/api/events/${selectedEventId}`);
   171→      setEvent(data.event);
   172→    } catch (err: any) {
   173→      toast.error(err.message || 'Booking failed');
   174→    } finally {
   175→      setBooking(false);
   176→    }
   177→  };
   178→
   179→  const avgRating = event?.reviews?.length
   180→    ? event.reviews.reduce((s, r) => s + r.rating, 0) / event.reviews.length
   181→    : 0;
   182→
   183→  const submitReview = async () => {
   184→    if (!selectedEventId) return;
   185→    setSubmittingReview(true);
   186→    try {
   187→      await apiFetch(`/api/events/${selectedEventId}/reviews`, {
   188→        method: 'POST',
   189→        body: JSON.stringify({ rating: reviewRating, comment: reviewComment || undefined }),
   190→      });
   191→      toast.success('Review submitted!');
   192→      setReviewDialog(false);
   193→      setReviewRating(5);
   194→      setReviewComment('');
   195→      setUserHasReviewed(true);
   196→      // Refresh event to show new review
   197→      const data = await apiFetch<{ event: Event }>(`/api/events/${selectedEventId}`);
   198→      setEvent(data.event);
   199→    } catch (err: any) {
   200→      toast.error(err.message || 'Failed to submit review');
   201→    } finally {
   202→      setSubmittingReview(false);
   203→    }
   204→  };
   205→
   206→  if (loading) {
   207→    return (
   208→      <div className="space-y-6">
   209→        <Skeleton className="h-8 w-24" />
   210→        <Skeleton className="h-72 w-full rounded-xl" />
   211→        <div className="grid md:grid-cols-3 gap-6">
   212→          <div className="md:col-span-2 space-y-4">
   213→            <Skeleton className="h-8 w-3/4" />
   214→            <Skeleton className="h-4 w-1/2" />
   215→            <Skeleton className="h-40 w-full" />
   216→          </div>
   217→          <Skeleton className="h-80 rounded-xl" />
   218→        </div>
   219→      </div>
   220→    );
   221→  }
   222→
   223→  if (!event) {
   224→    return (
   225→      <div className="text-center py-20">
   226→        <CalendarDays className="h-12 w-12 text-muted-foreground mx-auto mb-4" />
   227→        <h2 className="text-xl font-semibold mb-2">Event not found</h2>
   228→        <Button variant="outline" onClick={goBack}>Go Back</Button>
   229→      </div>
   230→    );
   231→  }
   232→
   233→  return (
   234→    <motion.div
   235→      initial={{ opacity: 0, y: 10 }}
   236→      animate={{ opacity: 1, y: 0 }}
   237→      transition={{ duration: 0.3 }}
   238→      className="space-y-6"
   239→    >
   240→      {/* Back button */}
   241→      <Button variant="ghost" size="sm" onClick={goBack} className="gap-2">
   242→        <ArrowLeft className="h-4 w-4" /> Back
   243→      </Button>
   244→
   245→      {/* Hero Image */}
   246→      <div className="relative h-64 md:h-80 rounded-xl overflow-hidden">
   247→        {event.coverImage ? (
   248→          <img src={event.coverImage} alt={event.title} className="w-full h-full object-cover" />
   249→        ) : (
   250→          <div className="w-full h-full bg-gradient-to-br from-emerald-200 to-teal-200 flex items-center justify-center">
   251→            <CalendarDays className="h-16 w-16 text-emerald-300" />
   252→          </div>
   253→        )}
   254→        <div className="absolute inset-0 bg-gradient-to-t from-black/60 to-transparent" />
   255→        <div className="absolute bottom-4 left-6 right-6 text-white">
   256→          <Badge className="mb-2 bg-emerald-600">{event.category?.name}</Badge>
   257→          <h1 className="text-2xl md:text-4xl font-bold">{event.title}</h1>
   258→        </div>
   259→      </div>
   260→
   261→      <div className="grid md:grid-cols-3 gap-6">
   262→        {/* Main Content */}
   263→        <div className="md:col-span-2 space-y-6">
   264→          {/* Date / Location / Type */}
   265→          <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   266→            <Card className="p-4">
   267→              <div className="flex items-center gap-3">
   268→                <div className="p-2 rounded-lg bg-emerald-50 text-emerald-600">
   269→                  <CalendarDays className="h-5 w-5" />
   270→                </div>
   271→                <div>
   272→                  <p className="text-sm text-muted-foreground">Date</p>
   273→                  <p className="font-medium">{format(new Date(event.startDate), 'EEEE, MMM d, yyyy')}</p>
   274→                  {event.endDate && event.endDate !== event.startDate && (
   275→                    <p className="text-sm text-muted-foreground">to {format(new Date(event.endDate), 'MMM d, yyyy')}</p>
   276→                  )}
   277→                </div>
   278→              </div>
   279→            </Card>
   280→            <Card className="p-4">
   281→              <div className="flex items-center gap-3">
   282→                <div className="p-2 rounded-lg bg-teal-50 text-teal-600">
   283→                  <Clock className="h-5 w-5" />
   284→                </div>
   285→                <div>
   286→                  <p className="text-sm text-muted-foreground">Time</p>
   287→                  <p className="font-medium">{event.startTime || 'TBD'} {event.endTime ? `- ${event.endTime}` : ''}</p>
   288→                </div>
   289→              </div>
   290→            </Card>
   291→            <Card className="p-4">
   292→              <div className="flex items-center gap-3">
   293→                <div className={`p-2 rounded-lg ${event.isVirtual ? 'bg-cyan-50 text-cyan-600' : 'bg-amber-50 text-amber-600'}`}>
   294→                  {event.isVirtual ? <Globe className="h-5 w-5" /> : <MapPin className="h-5 w-5" />}
   295→                </div>
   296→                <div>
   297→                  <p className="text-sm text-muted-foreground">{event.isVirtual ? 'Virtual Event' : 'Location'}</p>
   298→                  {event.isVirtual ? (
   299→                    <p className="font-medium text-cyan-600">{event.virtualUrl || 'Online'}</p>
   300→                  ) : (
   301→                    <p className="font-medium">{event.venueName}{event.venueCity ? `, ${event.venueCity}` : ''}</p>
   302→                  )}
   303→                </div>
   304→              </div>
   305→            </Card>
   306→            <Card className="p-4">
   307→              <div className="flex items-center gap-3">
   308→                <div className="p-2 rounded-lg bg-purple-50 text-purple-600">
   309→                  <Users className="h-5 w-5" />
   310→                </div>
   311→                <div>
   312→                  <p className="text-sm text-muted-foreground">Capacity</p>
   313→                  <p className="font-medium">{event.capacity || 'Unlimited'} attendees</p>
   314→                </div>
   315→              </div>
   316→            </Card>
   317→          </div>
   318→
   319→          {/* Tags */}
   320→          {event.tags?.length > 0 && (
   321→            <div className="flex flex-wrap gap-2">
   322→              {event.tags.map(tag => (
   323→                <Badge key={tag.id} variant="outline" className="gap-1">
   324→                  <Tag className="h-3 w-3" /> {tag.name}
   325→                </Badge>
   326→              ))}
   327→            </div>
   328→          )}
   329→
   330→          {/* Description */}
   331→          <Card>
   332→            <CardHeader><CardTitle>Description</CardTitle></CardHeader>
   333→            <CardContent>
   334→              <div className="prose prose-sm max-w-none text-muted-foreground whitespace-pre-wrap">{event.description}</div>
   335→            </CardContent>
   336→          </Card>
   337→
   338→          {/* Program / Sessions */}
   339→          {sessions.length > 0 && (
   340→            <Card>
   341→              <CardHeader>
   342→                <CardTitle className="flex items-center gap-2">
   343→                  <CalendarDays className="h-5 w-5 text-emerald-600" />
   344→                  Program
   345→                </CardTitle>
   346→              </CardHeader>
   347→              <CardContent>
   348→                <div className="space-y-3">
   349→                  {sessions.map(session => (
   350→                    <div key={session.id} className="flex items-start gap-3 p-3 rounded-lg border">
   351→                      <div className="text-center min-w-[60px]">
   352→                        <p className="text-sm font-semibold">{session.startTime}</p>
   353→                        <p className="text-xs text-muted-foreground">{session.endTime}</p>
   354→                      </div>
   355→                      <div className="flex-1 min-w-0">
   356→                        <div className="flex items-center gap-2 flex-wrap">
   357→                          <h4 className="font-medium text-sm">{session.title}</h4>
   358→                          <Badge variant="outline" className="text-xs">{session.sessionType}</Badge>
   359→                        </div>
   360→                        {session.participant && (
   361→                          <p className="text-xs text-muted-foreground mt-1 flex items-center gap-1">
   362→                            <Mic className="h-3 w-3" /> {session.participant.name}
   363→                          </p>
   364→                        )}
   365→                        {session.venueName && (
   366→                          <p className="text-xs text-muted-foreground mt-0.5">{session.venueName}</p>
   367→                        )}
   368→                      </div>
   369→                    </div>
   370→                  ))}
   371→                </div>
   372→              </CardContent>
   373→            </Card>
   374→          )}
   375→
   376→          {/* Speakers & Participants */}
   377→          {participants.length > 0 && (
   378→            <Card>
   379→              <CardHeader>
   380→                <CardTitle className="flex items-center gap-2">
   381→                  <Users className="h-5 w-5 text-emerald-600" />
   382→                  Speakers & Participants
   383→                </CardTitle>
   384→              </CardHeader>
   385→              <CardContent>
   386→                <div className="grid sm:grid-cols-2 gap-4">
   387→                  {participants.map(p => (
   388→                    <div key={p.id} className="flex items-start gap-3 p-3 rounded-lg border">
   389→                      {p.image ? (
   390→                        <img src={p.image} alt={p.name} className="h-12 w-12 rounded-full object-cover shrink-0" />
   391→                      ) : (
   392→                        <div className="h-12 w-12 rounded-full bg-emerald-100 dark:bg-emerald-900 flex items-center justify-center shrink-0">
   393→                          <span className="text-emerald-700 dark:text-emerald-300 font-semibold">{p.name.charAt(0).toUpperCase()}</span>
   394→                        </div>
   395→                      )}
   396→                      <div className="min-w-0">
   397→                        <div className="flex items-center gap-2">
   398→                          <h4 className="font-medium text-sm">{p.name}</h4>
   399→                          {p.isFeatured && <Star className="h-3.5 w-3.5 fill-amber-400 text-amber-400" />}
   400→                        </div>
   401→                        <Badge variant="outline" className="text-xs mt-0.5">{p.role}</Badge>
   402→                        {p.title && (
   403→                          <p className="text-xs text-muted-foreground mt-1">{p.title}</p>
   404→                        )}
   405→                        {p.organization && (
   406→                          <p className="text-xs text-muted-foreground">{p.organization}</p>
   407→                        )}
   408→                        {p.bio && (
   409→                          <p className="text-xs text-muted-foreground mt-1 line-clamp-2">{p.bio}</p>
   410→                        )}
   411→                      </div>
   412→                    </div>
   413→                  ))}
   414→                </div>
   415→              </CardContent>
   416→            </Card>
   417→          )}
   418→
   419→          {/* Gallery / Media */}
   420→          {media.length > 0 && (
   421→            <Card>
   422→              <CardHeader>
   423→                <CardTitle className="flex items-center gap-2">
   424→                  <ImageIcon className="h-5 w-5 text-emerald-600" />
   425→                  Gallery
   426→                </CardTitle>
   427→              </CardHeader>
   428→              <CardContent>
   429→                <div className="grid grid-cols-2 sm:grid-cols-3 gap-3">
   430→                  {media.map(m => (
   431→                    <div key={m.id} className="relative rounded-lg overflow-hidden border aspect-video">
   432→                      {m.type === 'IMAGE' ? (
   433→                        <img src={m.url} alt={m.caption || 'Gallery image'} className="w-full h-full object-cover" />
   434→                      ) : m.type === 'VIDEO' ? (
   435→                        <div className="w-full h-full bg-zinc-100 dark:bg-zinc-800 flex items-center justify-center">
   436→                          <Video className="h-8 w-8 text-zinc-400" />
   437→                          <div className="absolute inset-0 flex items-center justify-center">
   438→                            <div className="h-10 w-10 rounded-full bg-black/30 flex items-center justify-center">
   439→                              <Play className="h-5 w-5 text-white ml-0.5" />
   440→                            </div>
   441→                          </div>
   442→                        </div>
   443→                      ) : (
   444→                        <div className="w-full h-full bg-zinc-50 dark:bg-zinc-800 flex items-center justify-center">
   445→                          <FileText className="h-8 w-8 text-zinc-400" />
   446→                        </div>
   447→                      )}
   448→                      {m.caption && (
   449→                        <div className="absolute bottom-0 left-0 right-0 bg-gradient-to-t from-black/60 to-transparent p-2">
   450→                          <p className="text-xs text-white truncate">{m.caption}</p>
   451→                        </div>
   452→                      )}
   453→                    </div>
   454→                  ))}
   455→                </div>
   456→              </CardContent>
   457→            </Card>
   458→          )}
   459→
   460→          {/* Reviews */}
   461→          <Card>
   462→            <CardHeader>
   463→              <CardTitle className="flex items-center justify-between">
   464→                <span className="flex items-center gap-2">
   465→                  <Star className="h-5 w-5 fill-amber-400 text-amber-400" />
   466→                  Reviews ({event._count?.reviews || 0})
   467→                  {avgRating > 0 && (
   468→                    <span className="text-sm font-normal text-muted-foreground">
   469→                      {avgRating.toFixed(1)} average
   470→                    </span>
   471→                  )}
   472→                </span>
   473→                {user && !userHasReviewed && (
   474→                  <Button
   475→                    variant="outline"
   476→                    size="sm"
   477→                    onClick={() => setReviewDialog(true)}
   478→                    className="gap-1"
   479→                  >
   480→                    <MessageSquarePlus className="h-4 w-4" />
   481→                    Write a Review
   482→                  </Button>
   483→                )}
   484→              </CardTitle>
   485→            </CardHeader>
   486→            <CardContent>
   487→              {event.reviews?.length === 0 ? (
   488→                <p className="text-muted-foreground text-sm">No reviews yet.</p>
   489→              ) : (
   490→                <div className="space-y-4 max-h-96 overflow-y-auto">
   491→                  {event.reviews.map(review => (
   492→                    <div key={review.id} className="border-b last:border-0 pb-4">
   493→                      <div className="flex items-center gap-2 mb-1">
   494→                        <span className="font-medium text-sm">{review.user?.name}</span>
   495→                        <div className="flex">
   496→                          {Array.from({ length: 5 }).map((_, i) => (
   497→                            <Star key={i} className={`h-3 w-3 ${i < review.rating ? 'fill-amber-400 text-amber-400' : 'text-gray-200'}`} />
   498→                          ))}
   499→                        </div>
   500→                        <span className="text-xs text-muted-foreground">{formatDistanceToNow(new Date(review.createdAt))}</span>
   501→                      </div>
   502→                      <p className="text-sm text-muted-foreground">{review.comment}</p>
   503→                    </div>
   504→                  ))}
   505→                </div>
   506→              )}
   507→            </CardContent>
   508→          </Card>
   509→
   510→          {/* Related Events */}
   511→          {relatedEvents.length > 0 && (
   512→            <Card>
   513→              <CardHeader><CardTitle>Related Events</CardTitle></CardHeader>
   514→              <CardContent>
   515→                <div className="grid sm:grid-cols-2 lg:grid-cols-3 gap-4">
   516→                  {relatedEvents.map(re => (
   517→                    <Card key={re.id} className="overflow-hidden cursor-pointer hover:shadow-md transition-shadow" onClick={() => navigate('event-detail', re.id)}>
   518→                      <div className="h-32 bg-gradient-to-br from-emerald-100 to-teal-100 flex items-center justify-center">

   531→        </div>
   532→
   533→        {/* Sidebar */}

   552→              <Separator />

   558→                  const available = tt.quantity - tt.soldCount;
   559→                  return (
   560→                    <div key={tt.id} className="border rounded-lg p-3 space-y-2">
   561→                      <div className="flex items-center justify-between">
   562→                        <span className="font-medium text-sm">{tt.name}</span>

   564→                          {tt.price > 0 ? `$${tt.price.toFixed(2)}` : 'Free'}

   570→                        className="w-full bg-emerald-600 hover:bg-emerald-700"
   571→                        disabled={available <= 0}
   572→                        onClick={() => handleBook(tt)}
   573→                      >
   574→                        {available <= 0 ? 'Sold Out' : event.isPaid ? 'Book Now' : 'Register Free'}
   575→                      </Button>
   576→                    </div>
   577→                  );
   578→                })}
   579→              </div>
   580→            </CardContent>
   581→          </Card>
   582→
   583→          {/* Organizer Card */}
   584→          {event.organizer && (
   585→            <Card>
   586→              <CardContent className="p-4">
   587→                <div className="flex items-center gap-3">
   588→                  <div className="h-10 w-10 rounded-full bg-emerald-100 flex items-center justify-center">
   589→                    <User className="h-5 w-5 text-emerald-600" />
   590→                  </div>
   591→                  <div>
   592→                    <p className="font-medium text-sm">{event.organizer.name}</p>
   593→                    <p className="text-xs text-muted-foreground">Organizer</p>
   594→                  </div>
   595→                </div>
   596→                {event.organizer.bio && (
   597→                  <p className="text-sm text-muted-foreground mt-2">{event.organizer.bio}</p>
   598→                )}
   599→              </CardContent>
   600→            </Card>
   601→          )}
   602→        </div>
   603→      </div>
   604→
   605→      {/* Booking Dialog */}
   606→      <Dialog open={bookDialog} onOpenChange={setBookDialog}>
   607→        <DialogContent>
   608→          <DialogHeader>
   609→            <DialogTitle>Book Tickets</DialogTitle>
   610→            <DialogDescription>{selectedTicket?.name} - {event.title}</DialogDescription>
   611→          </DialogHeader>
   612→          <div className="space-y-4 py-4">
   613→            <div className="flex items-center justify-between">
   614→              <span>Price per ticket</span>

   621→                <span className="w-8 text-center font-medium">{quantity}</span>
   622→                <Button variant="outline" size="icon" className="h-8 w-8" onClick={() => setQuantity(q => Math.min(selectedTicket ? selectedTicket.quantity - selectedTicket.soldCount : 10, q + 1))}><Plus className="h-4 w-4" /></Button>
   623→              </div>
   624→            </div>
   625→            <Separator />
   626→            <div className="flex items-center justify-between text-lg font-bold">
   627→              <span>Total</span>
   628→              <span className="text-emerald-600">{selectedTicket && selectedTicket.price > 0 ? `$${(selectedTicket.price * quantity).toFixed(2)}` : 'Free'}</span>
   629→            </div>
   630→          </div>
   631→          <DialogFooter>
   632→            <Button variant="outline" onClick={() => setBookDialog(false)}>Cancel</Button>
   633→            <Button className="bg-emerald-600 hover:bg-emerald-700" onClick={confirmBooking} disabled={booking}>
   634→              {booking && <Loader2 className="mr-2 h-4 w-4 animate-spin" />}
   635→              Confirm Booking
   636→            </Button>
   637→          </DialogFooter>
   638→        </DialogContent>
   639→      </Dialog>
   640→
   641→      {/* Review Dialog */}
   642→      <Dialog open={reviewDialog} onOpenChange={setReviewDialog}>
   643→        <DialogContent>
   644→          <DialogHeader>
   645→            <DialogTitle>Write a Review</DialogTitle>
   646→            <DialogDescription>Share your experience for {event.title}</DialogDescription>
   647→          </DialogHeader>
   648→          <div className="space-y-4 py-4">
   649→            <div className="space-y-2">
   650→              <Label className="text-sm font-medium">Rating</Label>
   651→              <div className="flex gap-1">
   652→                {Array.from({ length: 5 }).map((_, i) => (
   653→                  <button
   654→                    key={i}
   655→                    type="button"
   656→                    onClick={() => setReviewRating(i + 1)}
   657→                    className="p-1"
   658→                  >
   659→                    <Star
   660→                      className={`h-6 w-6 transition-colors ${
   661→                        i < reviewRating
   662→                          ? 'fill-amber-400 text-amber-400'
   663→                          : 'text-gray-200 hover:text-amber-300'
   664→                      }`}
   665→                    />
   666→                  </button>
   667→                ))}
   668→              </div>
   669→            </div>
   670→            <div className="space-y-2">
   671→              <Label className="text-sm font-medium">Comment (optional)</Label>
   672→              <Textarea
   673→                placeholder="Share your thoughts about this event..."
   674→                value={reviewComment}
   675→                onChange={(e) => setReviewComment(e.target.value)}
   676→                rows={3}
   677→                maxLength={1000}
   678→              />
   679→            </div>
   680→          </div>
   681→          <DialogFooter>
   682→            <Button variant="outline" onClick={() => setReviewDialog(false)}>
   683→              Cancel
   684→            </Button>
   685→            <Button
   686→              className="bg-emerald-600 hover:bg-emerald-700"
   687→              onClick={submitReview}
   688→              disabled={submittingReview}
   689→            >
   690→              {submittingReview && <Loader2 className="mr-2 h-4 w-4 animate-spin" />}
   691→              Submit Review
   692→            </Button>
   693→          </DialogFooter>
   694→        </DialogContent>
   695→      </Dialog>
   696→    </motion.div>
   697→  );
   698→}

   700→function formatDistanceToNow(date: Date) {
   701→  const seconds = Math.floor((Date.now() - date.getTime()) / 1000);
   702→  if (seconds < 60) return 'just now';
   703→  const minutes = Math.floor(seconds / 60);
   704→  if (minutes < 60) return `${minutes}m ago`;
   705→  const hours = Math.floor(minutes / 60);
   706→  if (hours < 24) return `${hours}h ago`;
   707→  const days = Math.floor(hours / 24);
   708→  if (days < 30) return `${days}d ago`;
   709→  const months = Math.floor(days / 30);
   710→  return `${months}mo ago`;
   711→}
