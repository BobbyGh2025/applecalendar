   700→   700→function formatDistanceToNow(date: Date) {
   701→   701→  const seconds = Math.floor((Date.now() - date.getTime()) / 1000);
   702→   702→  if (seconds < 60) return 'just now';
   703→   703→  const minutes = Math.floor(seconds / 60);
   704→   704→  if (minutes < 60) return `${minutes}m ago`;
   705→   705→  const hours = Math.floor(minutes / 60);
   706→   706→  if (hours < 24) return `${hours}h ago`;
   707→   707→  const days = Math.floor(hours / 24);
   708→   708→  if (days < 30) return `${days}d ago`;
   709→   709→  const months = Math.floor(days / 30);
   710→   710→  return `${months}mo ago`;
   711→   711→}
