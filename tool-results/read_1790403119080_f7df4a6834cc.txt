    65→    65→  DRAFT: 'bg-gray-100 text-gray-700',
    66→    66→  PENDING: 'bg-yellow-100 text-yellow-700',

    71→    71→
    72→    72→export function OrganizerEditEvent() {
    73→    73→  const { selectedEventId, navigate, user } = useAppStore();
    74→    74→  const [loading, setLoading] = useState(true);
    75→    75→  const [saving, setSaving] = useState(false);
    76→    76→  const [categories, setCategories] = useState<Category[]>([]);
    77→    77→  const [eventStatus, setEventStatus] = useState<string>('DRAFT');
    78→    78→  const [statusDialog, setStatusDialog] = useState<{ open: boolean; target: string }>({
    79→    79→    open: false,
    80→    80→    target: '',
    81→    81→  });
    82→    82→
    83→    83→  const [form, setForm] = useState({
    84→    84→    title: '',
    85→    85→    description: '',
    86→    86→    shortDescription: '',
    87→    87→    categoryId: '',
    88→    88→    coverImage: '',
    89→    89→    startDate: '',
    90→    90→    endDate: '',
    91→    91→    startTime: '',
    92→    92→    endTime: '',
    93→    93→    timezone: 'UTC',
    94→    94→    venueName: '',
    95→    95→    venueAddress: '',
    96→    96→    venueCity: '',
    97→    97→    venueState: '',
    98→    98→    venueCountry: '',
    99→    99→    isVirtual: false,
   100→   100→    virtualUrl: '',
   101→   101→    isPaid: false,
   102→   102→    currency: 'USD',
   103→   103→    capacity: '',
   104→   104→    tags: '',
   105→   105→  });
   106→   106→
   107→   107→  const [ticketTypes, setTicketTypes] = useState<TicketTypeForm[]>([]);
   108→   108→
   109→   109→  // Load event data and categories
   110→   110→  useEffect(() => {
   111→   111→    if (!selectedEventId) return;
   112→   112→    setLoading(true);
   113→   113→    Promise.all([
   114→   114→      apiFetch<{
   115→   115→        event: {
   116→   116→          title: string;
   117→   117→          description: string;
   118→   118→          shortDescription: string | null;
   119→   119→          coverImage: string | null;
   120→   120→          startDate: string;
   121→   121→          endDate: string | null;
   122→   122→          startTime: string | null;
   123→   123→          endTime: string | null;
   124→   124→          timezone: string;
   125→   125→          venueName: string | null;
   126→   126→          venueAddress: string | null;
   127→   127→          venueCity: string | null;
   128→   128→          venueState: string | null;
   129→   129→          venueCountry: string | null;
   130→   130→          isVirtual: boolean;
   131→   131→          virtualUrl: string | null;
   132→   132→          isPaid: boolean;
   133→   133→          currency: string;
   134→   134→          capacity: number | null;
   135→   135→          status: string;
   136→   136→          categoryId: string | null;
   137→   137→          category: { id: string; name: string; slug: string } | null;
   138→   138→          tags: { tag: { name: string } }[];
   139→   139→          ticketTypes: {
   140→   140→            id: string;
   141→   141→            name: string;
   142→   142→            price: number;
   143→   143→            quantity: number;
   144→   144→            soldCount: number;
   145→   145→          }[];
   146→   146→        };
   147→   147→      }>(`/api/events/${selectedEventId}`),
   148→   148→      apiFetch<{ categories: Category[] }>('/api/categories'),
   149→   149→    ])
   150→   150→      .then(([eventData, catData]) => {
   151→   151→        const e = eventData.event;
   152→   152→        setForm({
   153→   153→          title: e.title,
   154→   154→          description: e.description,
   155→   155→          shortDescription: e.shortDescription || '',
   156→   156→          categoryId: e.category?.id || '',
   157→   157→          coverImage: e.coverImage || '',
   158→   158→          startDate: e.startDate ? new Date(e.startDate).toISOString().split('T')[0] : '',
   159→   159→          endDate: e.endDate ? new Date(e.endDate).toISOString().split('T')[0] : '',
   160→   160→          startTime: e.startTime || '',
   161→   161→          endTime: e.endTime || '',
   162→   162→          timezone: e.timezone || 'UTC',
   163→   163→          venueName: e.venueName || '',
   164→   164→          venueAddress: e.venueAddress || '',
   165→   165→          venueCity: e.venueCity || '',
   166→   166→          venueState: e.venueState || '',
   167→   167→          venueCountry: e.venueCountry || '',
   168→   168→          isVirtual: e.isVirtual,
   169→   169→          virtualUrl: e.virtualUrl || '',
   170→   170→          isPaid: e.isPaid,
   171→   171→          currency: e.currency || 'USD',
   172→   172→          capacity: e.capacity ? String(e.capacity) : '',
   173→   173→          tags: e.tags?.map((t: { tag: { name: string } }) => t.tag.name).join(', ') || '',
   174→   174→        });
   175→   175→        setEventStatus(e.status);
   176→   176→        setTicketTypes(
   177→   177→          e.ticketTypes.map((tt) => ({
   178→   178→            id: tt.id,
   179→   179→            name: tt.name,
   180→   180→            price: String(tt.price),
   181→   181→            quantity: String(tt.quantity),
   182→   182→          }))
   183→   183→        );
   184→   184→        setCategories(catData.categories || []);
   185→   185→      })
   186→   186→      .catch(() => toast.error('Failed to load event'))
   187→   187→      .finally(() => setLoading(false));
   188→   188→  }, [selectedEventId]);
   189→   189→
   190→   190→  const updateForm = (key: string, value: string | boolean) =>
   191→   191→    setForm((prev) => ({ ...prev, [key]: value }));
   192→   192→
   193→   193→  const addTicketType = () =>
   194→   194→    setTicketTypes((prev) => [
   195→   195→      ...prev,
   196→   196→      { name: '', price: '0', quantity: '50', isNew: true },
   197→   197→    ]);
   198→   198→  const removeTicketType = (i: number) => {
   199→   199→    const tt = ticketTypes[i];
   200→   200→    if (tt.isNew) {
   201→   201→      setTicketTypes((prev) => prev.filter((_, idx) => idx !== i));
   202→   202→    } else {
   203→   203→      // Mark existing for deletion
   204→   204→      setTicketTypes((prev) =>
   205→   205→        prev.map((t, idx) => (idx === i ? { ...t, toDelete: true } : t))
   206→   206→      );
   207→   207→    }
   208→   208→  };
   209→   209→  const restoreTicketType = (i: number) => {
   210→   210→    setTicketTypes((prev) =>
   211→   211→      prev.map((t, idx) => (idx === i ? { ...t, toDelete: false } : t))
   212→   212→    );
   213→   213→  };
   214→   214→  const updateTicketType = (i: number, key: keyof TicketTypeForm, value: string) => {
   215→   215→    setTicketTypes((prev) =>
   216→   216→      prev.map((t, idx) => (idx === i ? { ...t, [key]: value } : t))
   217→   217→    );
   218→   218→  };
   219→   219→
   220→   220→  const handleSubmit = async (e: React.FormEvent) => {
   221→   221→    e.preventDefault();
   222→   222→    if (!form.title || !form.startDate) {
   223→   223→      toast.error('Please fill in required fields (title, date)');
   224→   224→      return;
   225→   225→    }
   226→   226→    if (form.isVirtual && !form.virtualUrl) {
   227→   227→      toast.error('Please provide a virtual event URL');
   228→   228→      return;
   229→   229→    }
   230→   230→
   231→   231→    setSaving(true);
   232→   232→    try {
   233→   233→      // Build PATCH body (only send changed fields)
   234→   234→      const body: Record<string, unknown> = {
   235→   235→        title: form.title,
   236→   236→        description: form.description,
   237→   237→        shortDescription: form.shortDescription || undefined,
   238→   238→        coverImage: form.coverImage || undefined,
   239→   239→        startDate: form.startDate,
   240→   240→        endDate: form.endDate || undefined,
   241→   241→        startTime: form.startTime || undefined,
   242→   242→        endTime: form.endTime || undefined,
   243→   243→        timezone: form.timezone,
   244→   244→        venueName: form.isVirtual ? undefined : form.venueName,
   245→   245→        venueAddress: form.isVirtual ? undefined : form.venueAddress,
   246→   246→        venueCity: form.isVirtual ? undefined : form.venueCity,
   247→   247→        venueState: form.isVirtual ? undefined : form.venueState,
   248→   248→        venueCountry: form.isVirtual ? undefined : form.venueCountry,
   249→   249→        isVirtual: form.isVirtual,
   250→   250→        virtualUrl: form.isVirtual ? form.virtualUrl : undefined,
   251→   251→        isPaid: form.isPaid,
   252→   252→        currency: form.currency,
   253→   253→        capacity: form.capacity ? parseInt(form.capacity) : undefined,
   254→   254→        categoryId: form.categoryId || undefined,
   255→   255→        tags: form.tags
   256→   256→          ? form.tags.split(',').map((t: string) => t.trim()).filter(Boolean)
   257→   257→          : [],
   258→   258→      };
   259→   259→
   260→   260→      await apiFetch(`/api/events/${selectedEventId}`, {
   261→   261→        method: 'PATCH',
   262→   262→        body: JSON.stringify(body),
   263→   263→      });
   264→   264→
   265→   265→      // Handle ticket type changes
   266→   266→      // Delete marked ticket types
   267→   267→      for (const tt of ticketTypes.filter((t) => t.toDelete && t.id)) {
   268→   268→        await apiFetch(
   269→   269→          `/api/events/${selectedEventId}/ticket-types/${tt.id}`,
   270→   270→          { method: 'DELETE' }
   271→   271→        );
   272→   272→      }
   273→   273→
   274→   274→      // Add new ticket types
   275→   275→      for (const tt of ticketTypes.filter((t) => t.isNew)) {
   276→   276→        await apiFetch(`/api/events/${selectedEventId}/ticket-types`, {
   277→   277→          method: 'POST',
   278→   278→          body: JSON.stringify({
   279→   279→            name: tt.name,
   280→   280→            price: parseFloat(tt.price) || 0,
   281→   281→            quantity: parseInt(tt.quantity) || 50,
   282→   282→            currency: form.currency,
   283→   283→          }),
   284→   284→        });
   285→   285→      }
   286→   286→
   287→   287→      // Update existing ticket types (changed values)
   288→   288→      for (const tt of ticketTypes.filter((t) => t.id && !t.isNew && !t.toDelete)) {
   289→   289→        await apiFetch(
   290→   290→          `/api/events/${selectedEventId}/ticket-types/${tt.id}`,
   291→   291→          {
   292→   292→            method: 'PATCH',
   293→   293→            body: JSON.stringify({
   294→   294→              name: tt.name,
   295→   295→              price: parseFloat(tt.price),
   296→   296→              quantity: parseInt(tt.quantity),
   297→   297→            }),
   298→   298→          }
   299→   299→        );
   300→   300→      }
   301→   301→
   302→   302→      toast.success('Event updated successfully!');
   303→   303→      navigate('organizer-events');
   304→   304→    } catch (err: unknown) {
   305→   305→      const message = err instanceof Error ? err.message : 'Failed to update event';
   306→   306→      toast.error(message);
   307→   307→    } finally {
   308→   308→      setSaving(false);
   309→   309→    }
   310→   310→  };
   311→   311→
   312→   312→  // Status transition handlers
   313→   313→  const handleStatusChange = async (targetStatus: string) => {
   314→   314→    setStatusDialog({ open: false, target: '' });
   315→   315→    setSaving(true);
   316→   316→    try {
   317→   317→      await apiFetch(`/api/events/${selectedEventId}`, {
   318→   318→        method: 'PATCH',
   319→   319→        body: JSON.stringify({ status: targetStatus }),
   320→   320→      });
   321→   321→      setEventStatus(targetStatus);
   322→   322→      toast.success(`Event status changed to ${targetStatus}`);
   323→   323→    } catch (err: unknown) {
   324→   324→      const message = err instanceof Error ? err.message : 'Failed to change status';
   325→   325→      toast.error(message);
   326→   326→    } finally {
   327→   327→      setSaving(false);
   328→   328→    }
   329→   329→  };
   330→   330→
   331→   331→  if (loading) {
   332→   332→    return (
   333→   333→      <div className="space-y-6 max-w-3xl">
   334→   334→        <div className="h-8 w-48 bg-muted animate-pulse rounded" />
   335→   335→        <div className="h-64 bg-muted animate-pulse rounded-xl" />
   336→   336→        <div className="h-48 bg-muted animate-pulse rounded-xl" />
   337→   337→      </div>
   338→   338→    );
   339→   339→  }
   340→   340→
   341→   341→  // Determine available status transitions based on current status
   342→   342→  const getStatusActions = () => {
   343→   343→    if (!user) return [];
   344→   344→    const actions: { target: string; label: string; icon: React.ReactNode; variant: string }[] = [];
   345→   345→
   346→   346→    if (user.role === 'ORGANIZER' || user.role === 'SUPER_ADMIN') {
   347→   347→      if (eventStatus === 'DRAFT') {
   348→   348→        actions.push({
   349→   349→          target: 'PENDING',
   350→   350→          label: 'Submit for Review',
   351→   351→          icon: <Send className="h-4 w-4" />,
   352→   352→          variant: 'default',
   353→   353→        });
   354→   354→      }
   355→   355→      if (eventStatus === 'PENDING') {
   356→   356→        actions.push({
   357→   357→          target: 'DRAFT',
   358→   358→          label: 'Withdraw from Review',
   359→   359→          icon: <XCircle className="h-4 w-4" />,
   360→   360→          variant: 'outline',
   361→   361→        });
   362→   362→      }
   363→   363→      if (eventStatus === 'PUBLISHED') {
   364→   364→        actions.push({
   365→   365→          target: 'CANCELLED',
   366→   366→          label: 'Cancel Event',
   367→   367→          icon: <XCircle className="h-4 w-4" />,
   368→   368→          variant: 'destructive',
   369→   369→        });
   370→   370→      }
   371→   371→    }
   372→   372→
   373→   373→    // SUPER_ADMIN can additionally publish and reject directly
   374→   374→    if (user.role === 'SUPER_ADMIN') {
   375→   375→      if (eventStatus === 'PENDING') {
   376→   376→        actions.push({
   377→   377→          target: 'PUBLISHED',
   378→   378→          label: 'Approve & Publish',
   379→   379→          icon: <Send className="h-4 w-4" />,
   380→   380→          variant: 'default',
   381→   381→        });
   382→   382→        actions.push({
   383→   383→          target: 'REJECTED',
   384→   384→          label: 'Reject Event',
   385→   385→          icon: <XCircle className="h-4 w-4" />,
   386→   386→          variant: 'destructive',
   387→   387→        });
   388→   388→      }
   389→   389→      if (eventStatus === 'DRAFT') {
   390→   390→        actions.push({
   391→   391→          target: 'PUBLISHED',
   392→   392→          label: 'Publish Directly',
   393→   393→          icon: <Send className="h-4 w-4" />,
   394→   394→          variant: 'default',
   395→   395→        });
   396→   396→      }
   397→   397→    }
   398→   398→
   399→   399→    return actions;
   400→   400→  };
   401→   401→
   402→   402→  return (
   403→   403→    <form onSubmit={handleSubmit} className="space-y-6 max-w-3xl">
   404→   404→      <div className="flex items-center justify-between">
   405→   405→        <h1 className="text-2xl font-bold">Edit Event</h1>
   406→   406→        <Badge className={STATUS_COLORS[eventStatus] || ''}>{eventStatus}</Badge>
   407→   407→      </div>
   408→   408→
   409→   409→      {/* Status Workflow Actions */}
   410→   410→      {getStatusActions().length > 0 && (
   411→   411→        <Card>
   412→   412→          <CardHeader>
   413→   413→            <CardTitle className="text-base">Event Status</CardTitle>
   414→   414→            <CardDescription>Change the event status to control visibility</CardDescription>
   415→   415→          </CardHeader>
   416→   416→          <CardContent>
   417→   417→            <div className="flex flex-wrap gap-2">
   418→   418→              {getStatusActions().map((action) => (
   419→   419→                <Button
   420→   420→                  key={action.target}
   421→   421→                  type="button"
   422→   422→                  variant={action.variant === 'destructive' ? 'destructive' : action.variant === 'default' ? 'default' : 'outline'}
   423→   423→                  className={action.variant === 'default' ? 'bg-emerald-600 hover:bg-emerald-700' : ''}
   424→   424→                  onClick={() =>
   425→   425→                    setStatusDialog({ open: true, target: action.target })
   426→   426→                  }
   427→   427→                  disabled={saving}
   428→   428→                >
   429→   429→                  {action.icon}
   430→   430→                  <span className="ml-2">{action.label}</span>
   431→   431→                </Button>
   432→   432→              ))}
   433→   433→            </div>
   434→   434→          </CardContent>
   435→   435→        </Card>
   436→   436→      )}
   437→   437→
   438→   438→      {/* Basic Info */}
   439→   439→      <Card>
   440→   440→        <CardHeader>
   441→   441→          <CardTitle className="flex items-center gap-2">
   442→   442→            <Info className="h-5 w-5 text-emerald-500" /> Basic Information
   443→   443→          </CardTitle>
   444→   444→        </CardHeader>
   445→   445→        <CardContent className="space-y-4">
   446→   446→          <div className="space-y-2">
   447→   447→            <Label htmlFor="title">Title *</Label>
   448→   448→            <Input
   449→   449→              id="title"
   450→   450→              placeholder="Event title"
   451→   451→              value={form.title}
   452→   452→              onChange={(e) => updateForm('title', e.target.value)}
   453→   453→              required
   454→   454→            />
   455→   455→          </div>
   456→   456→          <div className="space-y-2">
   457→   457→            <Label htmlFor="shortDesc">Short Description</Label>
   458→   458→            <Input
   459→   459→              id="shortDesc"
   460→   460→              placeholder="Brief summary (1-2 sentences)"
   461→   461→              value={form.shortDescription}
   462→   462→              onChange={(e) => updateForm('shortDescription', e.target.value)}
   463→   463→            />
   464→   464→          </div>
   465→   465→          <div className="space-y-2">
   466→   466→            <Label htmlFor="desc">Full Description</Label>
   467→   467→            <Textarea
   468→   468→              id="desc"
   469→   469→              placeholder="Detailed event description..."
   470→   470→              rows={5}
   471→   471→              value={form.description}
   472→   472→              onChange={(e) => updateForm('description', e.target.value)}
   473→   473→            />
   474→   474→          </div>
   475→   475→          <div className="grid sm:grid-cols-2 gap-4">
   476→   476→            <div className="space-y-2">
   477→   477→              <Label>Category</Label>
   478→   478→              <Select
   479→   479→                value={form.categoryId}
   480→   480→                onValueChange={(v) => updateForm('categoryId', v)}
   481→   481→              >
   482→   482→                <SelectTrigger>
   483→   483→                  <SelectValue placeholder="Select category" />
   484→   484→                </SelectTrigger>
   485→   485→                <SelectContent>
   486→   486→                  {categories.map((c) => (
   487→   487→                    <SelectItem key={c.id} value={c.id}>
   488→   488→                      {c.name}
   489→   489→                    </SelectItem>
   490→   490→                  ))}
   491→   491→                </SelectContent>
   492→   492→              </Select>
   493→   493→            </div>
   494→   494→            <div className="space-y-2">
   495→   495→              <Label htmlFor="coverImg">Cover Image URL</Label>
   496→   496→              <Input
   497→   497→                id="coverImg"
   498→   498→                placeholder="https://..."
   499→   499→                value={form.coverImage}
   500→   500→                onChange={(e) => updateForm('coverImage', e.target.value)}
   501→   501→              />
