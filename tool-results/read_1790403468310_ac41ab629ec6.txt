     1→'use client';
     2→
     3→import { useState, useEffect } from 'react';
     4→import { Card, CardContent, CardHeader, CardTitle, CardDescription } from '@/components/ui/card';
     5→import { Button } from '@/components/ui/button';
     6→import { Input } from '@/components/ui/input';
     7→import { Label } from '@/components/ui/label';
     8→import { Textarea } from '@/components/ui/textarea';
     9→import { Switch } from '@/components/ui/switch';
    10→import { Badge } from '@/components/ui/badge';
    11→import { Separator } from '@/components/ui/separator';
    12→import {
    13→  Select,
    14→  SelectContent,
    15→  SelectItem,
    16→  SelectTrigger,
    17→  SelectValue,
    18→} from '@/components/ui/select';
    19→import {
    20→  Dialog,
    21→  DialogContent,
    22→  DialogHeader,
    23→  DialogTitle,
    24→  DialogDescription,
    25→  DialogFooter,
    26→} from '@/components/ui/dialog';
    27→import { useAppStore } from '@/stores/app-store';
    28→import { apiFetch } from '@/lib/api';
    29→import { toast } from 'sonner';
    30→import {
    31→  Loader2,
    32→  Plus,
    33→  Trash2,
    34→  Globe,
    35→  MapPin,
    36→  DollarSign,
    37→  Info,
    38→  Clock,
    39→  Send,
    40→  XCircle,
    41→  Save,
    42→  AlertTriangle,
    43→  Calendar,
    44→  Users,
    45→  Image as ImageIcon,
    46→} from 'lucide-react';
    47→
    48→interface Category {
    49→  id: string;
    50→  name: string;
    51→  slug: string;
    52→}
    53→
    54→interface TicketTypeForm {
    55→  id?: string;
    56→  name: string;
    57→  price: string;
    58→  quantity: string;
    59→  isNew?: boolean;
    60→  toDelete?: boolean;
    61→}
    62→
    63→const STATUS_COLORS: Record<string, string> = {
    64→  PUBLISHED: 'bg-green-100 text-green-700',
    65→  DRAFT: 'bg-gray-100 text-gray-700',
    66→  PENDING: 'bg-yellow-100 text-yellow-700',
    67→  CANCELLED: 'bg-red-100 text-red-700',
    68→  COMPLETED: 'bg-blue-100 text-blue-700',
    69→  REJECTED: 'bg-red-100 text-red-700',
    70→};
    71→
    72→export function OrganizerEditEvent() {
    73→  const { selectedEventId, navigate, user } = useAppStore();
    74→  const [loading, setLoading] = useState(true);
    75→  const [saving, setSaving] = useState(false);
    76→  const [categories, setCategories] = useState<Category[]>([]);
    77→  const [eventStatus, setEventStatus] = useState<string>('DRAFT');
    78→  const [statusDialog, setStatusDialog] = useState<{ open: boolean; target: string }>({
    79→    open: false,
    80→    target: '',
    81→  });
    82→
    83→  const [form, setForm] = useState({
    84→    title: '',
    85→    description: '',
    86→    shortDescription: '',
    87→    categoryId: '',
    88→    coverImage: '',
    89→    startDate: '',
    90→    endDate: '',
    91→    startTime: '',
    92→    endTime: '',
    93→    timezone: 'UTC',
    94→    venueName: '',
    95→    venueAddress: '',
    96→    venueCity: '',
    97→    venueState: '',
    98→    venueCountry: '',
    99→    isVirtual: false,
   100→    virtualUrl: '',
   101→    isPaid: false,
   102→    currency: 'USD',
   103→    capacity: '',
   104→    tags: '',
   105→  });
   106→
   107→  const [ticketTypes, setTicketTypes] = useState<TicketTypeForm[]>([]);
   108→
   109→  // Load event data and categories
   110→  useEffect(() => {
   111→    if (!selectedEventId) return;
   112→    setLoading(true);
   113→    Promise.all([
   114→      apiFetch<{
   115→        event: {
   116→          title: string;
   117→          description: string;
   118→          shortDescription: string | null;
   119→          coverImage: string | null;
   120→          startDate: string;
   121→          endDate: string | null;
   122→          startTime: string | null;
   123→          endTime: string | null;
   124→          timezone: string;
   125→          venueName: string | null;
   126→          venueAddress: string | null;
   127→          venueCity: string | null;
   128→          venueState: string | null;
   129→          venueCountry: string | null;
   130→          isVirtual: boolean;
   131→          virtualUrl: string | null;
   132→          isPaid: boolean;
   133→          currency: string;
   134→          capacity: number | null;
   135→          status: string;
   136→          categoryId: string | null;
   137→          category: { id: string; name: string; slug: string } | null;
   138→          tags: { tag: { name: string } }[];
   139→          ticketTypes: {
   140→            id: string;
   141→            name: string;
   142→            price: number;
   143→            quantity: number;
   144→            soldCount: number;
   145→          }[];
   146→        };
   147→      }>(`/api/events/${selectedEventId}`),
   148→      apiFetch<{ categories: Category[] }>('/api/categories'),
   149→    ])
   150→      .then(([eventData, catData]) => {
   151→        const e = eventData.event;
   152→        setForm({
   153→          title: e.title,
   154→          description: e.description,
   155→          shortDescription: e.shortDescription || '',
   156→          categoryId: e.category?.id || '',
   157→          coverImage: e.coverImage || '',
   158→          startDate: e.startDate ? new Date(e.startDate).toISOString().split('T')[0] : '',
   159→          endDate: e.endDate ? new Date(e.endDate).toISOString().split('T')[0] : '',
   160→          startTime: e.startTime || '',
   161→          endTime: e.endTime || '',
   162→          timezone: e.timezone || 'UTC',
   163→          venueName: e.venueName || '',
   164→          venueAddress: e.venueAddress || '',
   165→          venueCity: e.venueCity || '',
   166→          venueState: e.venueState || '',
   167→          venueCountry: e.venueCountry || '',
   168→          isVirtual: e.isVirtual,
   169→          virtualUrl: e.virtualUrl || '',
   170→          isPaid: e.isPaid,
   171→          currency: e.currency || 'USD',
   172→          capacity: e.capacity ? String(e.capacity) : '',
   173→          tags: e.tags?.map((t: { tag: { name: string } }) => t.tag.name).join(', ') || '',
   174→        });
   175→        setEventStatus(e.status);
   176→        setTicketTypes(
   177→          e.ticketTypes.map((tt) => ({
   178→            id: tt.id,
   179→            name: tt.name,
   180→            price: String(tt.price),
   181→            quantity: String(tt.quantity),
   182→          }))
   183→        );
   184→        setCategories(catData.categories || []);
   185→      })
   186→      .catch(() => toast.error('Failed to load event'))
   187→      .finally(() => setLoading(false));
   188→  }, [selectedEventId]);
   189→
   190→  const updateForm = (key: string, value: string | boolean) =>
   191→    setForm((prev) => ({ ...prev, [key]: value }));
   192→
   193→  const addTicketType = () =>
   194→    setTicketTypes((prev) => [
   195→      ...prev,
   196→      { name: '', price: '0', quantity: '50', isNew: true },
   197→    ]);
   198→  const removeTicketType = (i: number) => {
   199→    const tt = ticketTypes[i];
   200→    if (tt.isNew) {
   201→      setTicketTypes((prev) => prev.filter((_, idx) => idx !== i));
   202→    } else {
   203→      // Mark existing for deletion
   204→      setTicketTypes((prev) =>
   205→        prev.map((t, idx) => (idx === i ? { ...t, toDelete: true } : t))
   206→      );
   207→    }
   208→  };
   209→  const restoreTicketType = (i: number) => {
   210→    setTicketTypes((prev) =>
   211→      prev.map((t, idx) => (idx === i ? { ...t, toDelete: false } : t))
   212→    );
   213→  };
   214→  const updateTicketType = (i: number, key: keyof TicketTypeForm, value: string) => {
   215→    setTicketTypes((prev) =>
   216→      prev.map((t, idx) => (idx === i ? { ...t, [key]: value } : t))
   217→    );
   218→  };
   219→
   220→  const handleSubmit = async (e: React.FormEvent) => {
   221→    e.preventDefault();
   222→    if (!form.title || !form.startDate) {
   223→      toast.error('Please fill in required fields (title, date)');
   224→      return;
   225→    }
   226→    if (form.isVirtual && !form.virtualUrl) {
   227→      toast.error('Please provide a virtual event URL');
   228→      return;
   229→    }
   230→
   231→    setSaving(true);
   232→    try {
   233→      // Build PATCH body (only send changed fields)
   234→      const body: Record<string, unknown> = {
   235→        title: form.title,
   236→        description: form.description,
   237→        shortDescription: form.shortDescription || undefined,
   238→        coverImage: form.coverImage || undefined,
   239→        startDate: form.startDate,
   240→        endDate: form.endDate || undefined,
   241→        startTime: form.startTime || undefined,
   242→        endTime: form.endTime || undefined,
   243→        timezone: form.timezone,
   244→        venueName: form.isVirtual ? undefined : form.venueName,
   245→        venueAddress: form.isVirtual ? undefined : form.venueAddress,
   246→        venueCity: form.isVirtual ? undefined : form.venueCity,
   247→        venueState: form.isVirtual ? undefined : form.venueState,
   248→        venueCountry: form.isVirtual ? undefined : form.venueCountry,
   249→        isVirtual: form.isVirtual,
   250→        virtualUrl: form.isVirtual ? form.virtualUrl : undefined,
   251→        isPaid: form.isPaid,
   252→        currency: form.currency,
   253→        capacity: form.capacity ? parseInt(form.capacity) : undefined,
   254→        categoryId: form.categoryId || undefined,
   255→        tags: form.tags
   256→          ? form.tags.split(',').map((t: string) => t.trim()).filter(Boolean)
   257→          : [],
   258→      };
   259→
   260→      await apiFetch(`/api/events/${selectedEventId}`, {
   261→        method: 'PATCH',
   262→        body: JSON.stringify(body),
   263→      });
   264→
   265→      // Handle ticket type changes
   266→      // Delete marked ticket types
   267→      for (const tt of ticketTypes.filter((t) => t.toDelete && t.id)) {
   268→        await apiFetch(
   269→          `/api/events/${selectedEventId}/ticket-types/${tt.id}`,
   270→          { method: 'DELETE' }
   271→        );
   272→      }
   273→
   274→      // Add new ticket types
   275→      for (const tt of ticketTypes.filter((t) => t.isNew)) {
   276→        await apiFetch(`/api/events/${selectedEventId}/ticket-types`, {
   277→          method: 'POST',
   278→          body: JSON.stringify({
   279→            name: tt.name,
   280→            price: parseFloat(tt.price) || 0,
   281→            quantity: parseInt(tt.quantity) || 50,
   282→            currency: form.currency,
   283→          }),
   284→        });
   285→      }
   286→
   287→      // Update existing ticket types (changed values)
   288→      for (const tt of ticketTypes.filter((t) => t.id && !t.isNew && !t.toDelete)) {
   289→        await apiFetch(
   290→          `/api/events/${selectedEventId}/ticket-types/${tt.id}`,
   291→          {
   292→            method: 'PATCH',
   293→            body: JSON.stringify({
   294→              name: tt.name,
   295→              price: parseFloat(tt.price),
   296→              quantity: parseInt(tt.quantity),
   297→            }),
   298→          }
   299→        );
   300→      }
   301→
   302→      toast.success('Event updated successfully!');
   303→      navigate('organizer-events');
   304→    } catch (err: unknown) {
   305→      const message = err instanceof Error ? err.message : 'Failed to update event';
   306→      toast.error(message);
   307→    } finally {
   308→      setSaving(false);
   309→    }
   310→  };
   311→
   312→  // Status transition handlers
   313→  const handleStatusChange = async (targetStatus: string) => {
   314→    setStatusDialog({ open: false, target: '' });
   315→    setSaving(true);
   316→    try {
   317→      await apiFetch(`/api/events/${selectedEventId}`, {
   318→        method: 'PATCH',
   319→        body: JSON.stringify({ status: targetStatus }),
   320→      });
   321→      setEventStatus(targetStatus);
   322→      toast.success(`Event status changed to ${targetStatus}`);
   323→    } catch (err: unknown) {
   324→      const message = err instanceof Error ? err.message : 'Failed to change status';
   325→      toast.error(message);
   326→    } finally {
   327→      setSaving(false);
   328→    }
   329→  };
   330→
   331→  if (loading) {
   332→    return (
   333→      <div className="space-y-6 max-w-3xl">
   334→        <div className="h-8 w-48 bg-muted animate-pulse rounded" />
   335→        <div className="h-64 bg-muted animate-pulse rounded-xl" />
   336→        <div className="h-48 bg-muted animate-pulse rounded-xl" />
   337→      </div>
   338→    );
   339→  }
   340→
   341→  // Determine available status transitions based on current status
   342→  const getStatusActions = () => {
   343→    if (!user) return [];
   344→    const actions: { target: string; label: string; icon: React.ReactNode; variant: string }[] = [];
   345→
   346→    if (user.role === 'ORGANIZER' || user.role === 'SUPER_ADMIN') {
   347→      if (eventStatus === 'DRAFT') {
   348→        actions.push({
   349→          target: 'PENDING',
   350→          label: 'Submit for Review',
   351→          icon: <Send className="h-4 w-4" />,
   352→          variant: 'default',
   353→        });
   354→      }
   355→      if (eventStatus === 'PENDING') {
   356→        actions.push({
   357→          target: 'DRAFT',
   358→          label: 'Withdraw from Review',
   359→          icon: <XCircle className="h-4 w-4" />,
   360→          variant: 'outline',
   361→        });
   362→      }
   363→      if (eventStatus === 'PUBLISHED') {
   364→        actions.push({
   365→          target: 'CANCELLED',
   366→          label: 'Cancel Event',
   367→          icon: <XCircle className="h-4 w-4" />,
   368→          variant: 'destructive',
   369→        });
   370→      }
   371→    }
   372→
   373→    // SUPER_ADMIN can additionally publish and reject directly
   374→    if (user.role === 'SUPER_ADMIN') {
   375→      if (eventStatus === 'PENDING') {
   376→        actions.push({
   377→          target: 'PUBLISHED',
   378→          label: 'Approve & Publish',
   379→          icon: <Send className="h-4 w-4" />,
   380→          variant: 'default',
   381→        });
   382→        actions.push({
   383→          target: 'REJECTED',
   384→          label: 'Reject Event',
   385→          icon: <XCircle className="h-4 w-4" />,
   386→          variant: 'destructive',
   387→        });
   388→      }
   389→      if (eventStatus === 'DRAFT') {
   390→        actions.push({
   391→          target: 'PUBLISHED',
   392→          label: 'Publish Directly',
   393→          icon: <Send className="h-4 w-4" />,
   394→          variant: 'default',
   395→        });
   396→      }
   397→    }
   398→
   399→    return actions;
   400→  };
   401→
   402→  return (
   403→    <form onSubmit={handleSubmit} className="space-y-6 max-w-3xl">
   404→      <div className="flex items-center justify-between">
   405→        <h1 className="text-2xl font-bold">Edit Event</h1>
   406→        <Badge className={STATUS_COLORS[eventStatus] || ''}>{eventStatus}</Badge>
   407→      </div>
   408→
   409→      {/* Status Workflow Actions */}
   410→      {getStatusActions().length > 0 && (
   411→        <Card>
   412→          <CardHeader>
   413→            <CardTitle className="text-base">Event Status</CardTitle>
   414→            <CardDescription>Change the event status to control visibility</CardDescription>
   415→          </CardHeader>
   416→          <CardContent>
   417→            <div className="flex flex-wrap gap-2">
   418→              {getStatusActions().map((action) => (
   419→                <Button
   420→                  key={action.target}
   421→                  type="button"
   422→                  variant={action.variant === 'destructive' ? 'destructive' : action.variant === 'default' ? 'default' : 'outline'}
   423→                  className={action.variant === 'default' ? 'bg-emerald-600 hover:bg-emerald-700' : ''}
   424→                  onClick={() =>
   425→                    setStatusDialog({ open: true, target: action.target })
   426→                  }
   427→                  disabled={saving}
   428→                >
   429→                  {action.icon}
   430→                  <span className="ml-2">{action.label}</span>
   431→                </Button>
   432→              ))}
   433→            </div>
   434→          </CardContent>
   435→        </Card>
   436→      )}
   437→
   438→      {/* Basic Info */}
   439→      <Card>
   440→        <CardHeader>
   441→          <CardTitle className="flex items-center gap-2">
   442→            <Info className="h-5 w-5 text-emerald-500" /> Basic Information
   443→          </CardTitle>
   444→        </CardHeader>
   445→        <CardContent className="space-y-4">
   446→          <div className="space-y-2">
   447→            <Label htmlFor="title">Title *</Label>
   448→            <Input
   449→              id="title"
   450→              placeholder="Event title"
   451→              value={form.title}
   452→              onChange={(e) => updateForm('title', e.target.value)}
   453→              required
   454→            />
   455→          </div>
   456→          <div className="space-y-2">
   457→            <Label htmlFor="shortDesc">Short Description</Label>
   458→            <Input
   459→              id="shortDesc"
   460→              placeholder="Brief summary (1-2 sentences)"
   461→              value={form.shortDescription}
   462→              onChange={(e) => updateForm('shortDescription', e.target.value)}
   463→            />
   464→          </div>
   465→          <div className="space-y-2">
   466→            <Label htmlFor="desc">Full Description</Label>
   467→            <Textarea
   468→              id="desc"
   469→              placeholder="Detailed event description..."
   470→              rows={5}
   471→              value={form.description}
   472→              onChange={(e) => updateForm('description', e.target.value)}
   473→            />
   474→          </div>
   475→          <div className="grid sm:grid-cols-2 gap-4">
   476→            <div className="space-y-2">
   477→              <Label>Category</Label>
   478→              <Select
   479→                value={form.categoryId}
   480→                onValueChange={(v) => updateForm('categoryId', v)}
   481→              >
   482→                <SelectTrigger>
   483→                  <SelectValue placeholder="Select category" />
   484→                </SelectTrigger>
   485→                <SelectContent>
   486→                  {categories.map((c) => (
   487→                    <SelectItem key={c.id} value={c.id}>
   488→                      {c.name}
   489→                    </SelectItem>
   490→                  ))}
   491→                </SelectContent>
   492→              </Select>
   493→            </div>
   494→            <div className="space-y-2">
   495→              <Label htmlFor="coverImg">Cover Image URL</Label>
   496→              <Input
   497→                id="coverImg"
   498→                placeholder="https://..."
   499→                value={form.coverImage}
   500→                onChange={(e) => updateForm('coverImage', e.target.value)}
   501→              />
   502→            </div>
   503→          </div>
   504→          <div className="space-y-2">
   505→            <Label htmlFor="tags">Tags (comma-separated)</Label>
   506→            <Input
   507→              id="tags"
   508→              placeholder="music, tech, networking"
   509→              value={form.tags}
   510→              onChange={(e) => updateForm('tags', e.target.value)}
   511→            />
   512→          </div>
   513→        </CardContent>
   514→      </Card>
   515→
   516→      {/* Date & Time */}
   517→      <Card>
   518→        <CardHeader>
   519→          <CardTitle className="flex items-center gap-2">
   520→            <Clock className="h-5 w-5 text-teal-500" /> Date & Time
   521→          </CardTitle>
   522→        </CardHeader>
   523→        <CardContent className="space-y-4">
   524→          <div className="grid sm:grid-cols-2 gap-4">
   525→            <div className="space-y-2">
   526→              <Label htmlFor="startDate">Start Date *</Label>
   527→              <Input
   528→                id="startDate"
   529→                type="date"
   530→                value={form.startDate}
   531→                onChange={(e) => updateForm('startDate', e.target.value)}
   532→                required
   533→              />
   534→            </div>
   535→            <div className="space-y-2">
   536→              <Label htmlFor="endDate">End Date</Label>
   537→              <Input
   538→                id="endDate"
   539→                type="date"
   540→                value={form.endDate}
   541→                onChange={(e) => updateForm('endDate', e.target.value)}
   542→              />
   543→            </div>
   544→          </div>
   545→          <div className="grid sm:grid-cols-2 gap-4">
   546→            <div className="space-y-2">
   547→              <Label htmlFor="startTime">Start Time</Label>
   548→              <Input
   549→                id="startTime"
   550→                type="time"
   551→                value={form.startTime}
   552→                onChange={(e) => updateForm('startTime', e.target.value)}
   553→              />
   554→            </div>
   555→            <div className="space-y-2">
   556→              <Label htmlFor="endTime">End Time</Label>
   557→              <Input
   558→                id="endTime"
   559→                type="time"
   560→                value={form.endTime}
   561→                onChange={(e) => updateForm('endTime', e.target.value)}
   562→              />
   563→            </div>
   564→          </div>
   565→          <div className="space-y-2">
   566→            <Label>Timezone</Label>
   567→            <Select
   568→              value={form.timezone}
   569→              onValueChange={(v) => updateForm('timezone', v)}
   570→            >
   571→              <SelectTrigger>
   572→                <SelectValue />
   573→              </SelectTrigger>
   574→              <SelectContent>
   575→                <SelectItem value="UTC">UTC</SelectItem>
   576→                <SelectItem value="America/New_York">Eastern (ET)</SelectItem>
   577→                <SelectItem value="America/Chicago">Central (CT)</SelectItem>
   578→                <SelectItem value="America/Denver">Mountain (MT)</SelectItem>
   579→                <SelectItem value="America/Los_Angeles">Pacific (PT)</SelectItem>
   580→                <SelectItem value="Europe/London">London (GMT)</SelectItem>
   581→              </SelectContent>
   582→            </Select>
   583→          </div>
   584→        </CardContent>
   585→      </Card>
   586→
   587→      {/* Venue */}
   588→      <Card>
   589→        <CardHeader>
   590→          <CardTitle className="flex items-center gap-2">
   591→            {form.isVirtual ? (
   592→              <Globe className="h-5 w-5 text-cyan-500" />
   593→            ) : (
   594→              <MapPin className="h-5 w-5 text-amber-500" />
   595→            )}
   596→            {form.isVirtual ? 'Virtual Event' : 'Venue'}
   597→          </CardTitle>
   598→        </CardHeader>
   599→        <CardContent className="space-y-4">
   600→          <div className="flex items-center gap-3">
   601→            <Switch
   602→              checked={form.isVirtual}
   603→              onCheckedChange={(v) => updateForm('isVirtual', v)}
   604→            />
   605→            <Label>This is a virtual/online event</Label>
   606→          </div>
   607→          {form.isVirtual ? (
   608→            <div className="space-y-2">
   609→              <Label htmlFor="virtualUrl">Virtual Event URL *</Label>
   610→              <Input
   611→                id="virtualUrl"
   612→                placeholder="https://zoom.us/..."
   613→                value={form.virtualUrl}
   614→                onChange={(e) => updateForm('virtualUrl', e.target.value)}
   615→              />
   616→            </div>
   617→          ) : (
   618→            <div className="grid sm:grid-cols-2 gap-4">
   619→              <div className="space-y-2">
   620→                <Label htmlFor="venueName">Venue Name</Label>
   621→                <Input
   622→                  id="venueName"
   623→                  placeholder="Convention Center"
   624→                  value={form.venueName}
   625→                  onChange={(e) => updateForm('venueName', e.target.value)}
   626→                />
   627→              </div>
   628→              <div className="space-y-2">
   629→                <Label htmlFor="venueAddr">Address</Label>
   630→                <Input
   631→                  id="venueAddr"
   632→                  placeholder="123 Main St"
   633→                  value={form.venueAddress}
   634→                  onChange={(e) => updateForm('venueAddress', e.target.value)}
   635→                />
   636→              </div>
   637→              <div className="space-y-2">
   638→                <Label htmlFor="venueCity">City</Label>
   639→                <Input
   640→                  id="venueCity"
   641→                  placeholder="New York"
   642→                  value={form.venueCity}
   643→                  onChange={(e) => updateForm('venueCity', e.target.value)}
   644→                />
   645→              </div>
   646→              <div className="space-y-2">
   647→                <Label htmlFor="venueState">State</Label>
   648→                <Input
   649→                  id="venueState"
   650→                  placeholder="NY"
   651→                  value={form.venueState}
   652→                  onChange={(e) => updateForm('venueState', e.target.value)}
   653→                />
   654→              </div>
   655→              <div className="space-y-2 sm:col-span-2">
   656→                <Label htmlFor="venueCountry">Country</Label>
   657→                <Input
   658→                  id="venueCountry"
   659→                  placeholder="United States"
   660→                  value={form.venueCountry}
   661→                  onChange={(e) => updateForm('venueCountry', e.target.value)}
   662→                />
   663→              </div>
   664→            </div>
   665→          )}
   666→        </CardContent>
   667→      </Card>
   668→
   669→      {/* Ticketing */}
   670→      <Card>
   671→        <CardHeader>
   672→          <CardTitle className="flex items-center gap-2">
   673→            <DollarSign className="h-5 w-5 text-emerald-500" /> Ticketing
   674→          </CardTitle>
   675→        </CardHeader>
   676→        <CardContent className="space-y-4">
   677→          <div className="flex items-center gap-3">
   678→            <Switch
   679→              checked={form.isPaid}
   680→              onCheckedChange={(v) => updateForm('isPaid', v)}
   681→            />
   682→            <Label>This is a paid event</Label>
   683→          </div>
   684→          {form.isPaid && (
   685→            <>
   686→              <div className="grid sm:grid-cols-2 gap-4">
   687→                <div className="space-y-2">
   688→                  <Label>Currency</Label>
   689→                  <Select
   690→                    value={form.currency}
   691→                    onValueChange={(v) => updateForm('currency', v)}
   692→                  >
   693→                    <SelectTrigger>
   694→                      <SelectValue />
   695→                    </SelectTrigger>
   696→                    <SelectContent>
   697→                      <SelectItem value="USD">USD ($)</SelectItem>
   698→                      <SelectItem value="EUR">EUR</SelectItem>
   699→                      <SelectItem value="GBP">GBP</SelectItem>
   700→                    </SelectContent>
   701→                  </Select>
   702→                </div>
   703→                <div className="space-y-2">
   704→                  <Label htmlFor="capacity">Total Capacity</Label>
   705→                  <Input
   706→                    id="capacity"
   707→                    type="number"
   708→                    placeholder="100"
   709→                    value={form.capacity}
   710→                    onChange={(e) => updateForm('capacity', e.target.value)}
   711→                  />
   712→                </div>
   713→              </div>
   714→              <Separator />
   715→              <div className="space-y-3">
   716→                <div className="flex items-center justify-between">
   717→                  <Label>Ticket Types</Label>
   718→                  <Button
   719→                    type="button"
   720→                    variant="outline"
   721→                    size="sm"
   722→                    onClick={addTicketType}
   723→                  >
   724→                    <Plus className="h-4 w-4 mr-1" /> Add Type
   725→                  </Button>
   726→                </div>
   727→                {ticketTypes.map((tt, i) => (
   728→                  <div
   729→                    key={tt.id || i}
   730→                    className={`grid grid-cols-[1fr_100px_100px_40px] gap-2 items-end ${tt.toDelete ? 'opacity-50' : ''}`}
   731→                  >
   732→                    <div className="space-y-1">
   733→                      {i === 0 && <Label className="text-xs">Name</Label>}
   734→                      <Input
   735→                        placeholder="VIP, General..."
   736→                        value={tt.name}
   737→                        onChange={(e) => updateTicketType(i, 'name', e.target.value)}
   738→                        disabled={tt.toDelete}
   739→                      />
   740→                    </div>
   741→                    <div className="space-y-1">
   742→                      {i === 0 && <Label className="text-xs">Price</Label>}
   743→                      <Input
   744→                        type="number"
   745→                        placeholder="0"
   746→                        value={tt.price}
   747→                        onChange={(e) => updateTicketType(i, 'price', e.target.value)}
   748→                        disabled={tt.toDelete}
   749→                      />
   750→                    </div>
   751→                    <div className="space-y-1">
   752→                      {i === 0 && <Label className="text-xs">Qty</Label>}
   753→                      <Input
   754→                        type="number"
   755→                        placeholder="50"
   756→                        value={tt.quantity}
   757→                        onChange={(e) => updateTicketType(i, 'quantity', e.target.value)}
   758→                        disabled={tt.toDelete}
   759→                      />
   760→                    </div>
   761→                    {tt.toDelete ? (
   762→                      <Button
   763→                        type="button"
   764→                        variant="ghost"
   765→                        size="icon"
   766→                        className="text-emerald-500"
   767→                        onClick={() => restoreTicketType(i)}
   768→                      >
   769→                        <AlertTriangle className="h-4 w-4" />
   770→                      </Button>
   771→                    ) : (
   772→                      <Button
   773→                        type="button"
   774→                        variant="ghost"
   775→                        size="icon"
   776→                        className="text-red-500"
   777→                        onClick={() => removeTicketType(i)}
   778→                        disabled={ticketTypes.filter((t) => !t.toDelete).length <= 1}
   779→                      >
   780→                        <Trash2 className="h-4 w-4" />
   781→                      </Button>
   782→                    )}
   783→                  </div>
   784→                ))}
   785→              </div>
   786→            </>
   787→          )}
   788→        </CardContent>
   789→      </Card>
   790→
   791→      {/* Manage Event Content */}
   792→      <Card>
   793→        <CardHeader>
   794→          <CardTitle className="flex items-center gap-2">
   795→            <Calendar className="h-5 w-5 text-emerald-500" /> Manage Event Content
   796→          </CardTitle>
   797→          <CardDescription>Manage sessions, participants, and media for this event</CardDescription>
   798→        </CardHeader>
   799→        <CardContent>
   800→          <div className="grid sm:grid-cols-3 gap-3">
   801→            <Button
   802→              type="button"
   803→              variant="outline"
   804→              className="h-auto py-4 flex-col gap-2"
   805→              onClick={() => navigate('organizer-event-content')}
   806→            >
   807→              <Calendar className="h-6 w-6 text-emerald-600" />
   808→              <span className="font-medium">Program</span>
   809→              <span className="text-xs text-muted-foreground">Sessions & Schedule</span>
   810→            </Button>
   811→            <Button
   812→              type="button"
   813→              variant="outline"
   814→              className="h-auto py-4 flex-col gap-2"
   815→              onClick={() => navigate('organizer-event-content')}
   816→            >
   817→              <Users className="h-6 w-6 text-emerald-600" />
   818→              <span className="font-medium">Participants</span>
   819→              <span className="text-xs text-muted-foreground">Speakers & Artists</span>
   820→            </Button>
   821→            <Button
   822→              type="button"
   823→              variant="outline"
   824→              className="h-auto py-4 flex-col gap-2"
   825→              onClick={() => navigate('organizer-event-content')}
   826→            >
   827→              <ImageIcon className="h-6 w-6 text-emerald-600" />
   828→              <span className="font-medium">Media</span>
   829→              <span className="text-xs text-muted-foreground">Images & Videos</span>
   830→            </Button>
   831→          </div>
   832→        </CardContent>
   833→      </Card>
   834→
   835→      <div className="flex gap-4">
   836→        <Button
   837→          type="button"
   838→          variant="outline"
   839→          onClick={() => navigate('organizer-events')}
   840→        >
   841→          Cancel
   842→        </Button>
   843→        <Button
   844→          type="submit"
   845→          className="bg-emerald-600 hover:bg-emerald-700"
   846→          disabled={saving}
   847→        >
   848→          {saving ? (
   849→            <Loader2 className="mr-2 h-4 w-4 animate-spin" />
   850→          ) : (
   851→            <Save className="mr-2 h-4 w-4" />
   852→          )}
   853→          Save Changes
   854→        </Button>
   855→      </div>
   856→
   857→      {/* Status Change Confirmation Dialog */}
   858→      <Dialog
   859→        open={statusDialog.open}
   860→        onOpenChange={(open) => setStatusDialog({ open, target: statusDialog.target })}
   861→      >
   862→        <DialogContent>
   863→          <DialogHeader>
   864→            <DialogTitle>Confirm Status Change</DialogTitle>
   865→            <DialogDescription>
   866→              Are you sure you want to change the event status from{' '}
   867→              <Badge className={STATUS_COLORS[eventStatus]}>{eventStatus}</Badge> to{' '}
   868→              <Badge className={STATUS_COLORS[statusDialog.target]}>
   869→                {statusDialog.target}
   870→              </Badge>
   871→              ?
   872→              {statusDialog.target === 'CANCELLED' &&
   873→                ' This will make the event unavailable for booking.'}
   874→              {statusDialog.target === 'PENDING' &&
   875→                ' This will submit the event for admin review.'}
   876→              {statusDialog.target === 'PUBLISHED' &&
   877→                ' This will make the event publicly visible.'}
   878→            </DialogDescription>
   879→          </DialogHeader>
   880→          <DialogFooter>
   881→            <Button
   882→              variant="outline"
   883→              onClick={() => setStatusDialog({ open: false, target: '' })}
   884→            >
   885→              Cancel
   886→            </Button>
   887→            <Button
   888→              className={
   889→                statusDialog.target === 'CANCELLED' || statusDialog.target === 'REJECTED'
   890→                  ? 'bg-red-600 hover:bg-red-700'
   891→                  : 'bg-emerald-600 hover:bg-emerald-700'
   892→              }
   893→              onClick={() => handleStatusChange(statusDialog.target)}
   894→              disabled={saving}
   895→            >
   896→              {saving && <Loader2 className="mr-2 h-4 w-4 animate-spin" />}
   897→              Confirm
   898→            </Button>
   899→          </DialogFooter>
   900→        </DialogContent>
   901→      </Dialog>
   902→    </form>
   903→  );
