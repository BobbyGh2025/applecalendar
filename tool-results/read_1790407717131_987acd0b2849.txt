     1→'use client';
     2→
     3→import { useState, useEffect } from 'react';
     4→import { Card, CardContent, CardHeader, CardTitle, CardDescription } from '@/components/ui/card';
     5→import { Button } from '@/components/ui/button';
     6→import { Input } from '@/components/ui/input';
     7→import { Label } from '@/components/ui/label';
     8→import { Textarea } from '@/components/ui/textarea';
     9→import { Switch } from '@/components/ui/switch';
    10→import { Badge } from '@/components/ui/badge';
    11→import { Separator } from '@/components/ui/separator';
    12→import {
    13→  Select,
    14→  SelectContent,
    15→  SelectItem,
    16→  SelectTrigger,
    17→  SelectValue,
    18→} from '@/components/ui/select';
    19→import {
    20→  Dialog,
    21→  DialogContent,
    22→  DialogHeader,
    23→  DialogTitle,
    24→  DialogDescription,
    25→  DialogFooter,
    26→} from '@/components/ui/dialog';
    27→import { useAppStore } from '@/stores/app-store';
    28→import { apiFetch } from '@/lib/api';
    29→import { toast } from 'sonner';
    30→import {
    31→  Loader2,
    32→  Plus,
    33→  Trash2,
    34→  Globe,
    35→  MapPin,
    36→  DollarSign,
    37→  Info,
    38→  Clock,
    39→  Send,
    40→  XCircle,
    41→  Save,
    42→  AlertTriangle,
    43→  Calendar,
    44→  Users,
    45→  Image as ImageIcon,
    46→  Building2,
    47→} from 'lucide-react';
    48→
    49→interface Category {
    50→  id: string;
    51→  name: string;
    52→  slug: string;
    53→}
    54→
    55→interface VenueOption {
    56→  id: string;
    57→  name: string;
    58→  address: string;
    59→  city: string;
    60→  state: string | null;
    61→  country: string;
    62→  lat: number | null;
    63→  lng: number | null;
    64→}
    65→
    66→interface TicketTypeForm {
    67→  id?: string;
    68→  name: string;
    69→  price: string;
    70→  quantity: string;
    71→  isNew?: boolean;
    72→  toDelete?: boolean;
    73→}
    74→
    75→const STATUS_COLORS: Record<string, string> = {
    76→  PUBLISHED: 'bg-green-100 text-green-700',
    77→  DRAFT: 'bg-gray-100 text-gray-700',
    78→  PENDING: 'bg-yellow-100 text-yellow-700',
    79→  CANCELLED: 'bg-red-100 text-red-700',
    80→  COMPLETED: 'bg-blue-100 text-blue-700',
    81→  REJECTED: 'bg-red-100 text-red-700',
    82→};
    83→
    84→export function OrganizerEditEvent() {
    85→  const { selectedEventId, navigate, user } = useAppStore();
    86→  const [loading, setLoading] = useState(true);
    87→  const [saving, setSaving] = useState(false);
    88→  const [categories, setCategories] = useState<Category[]>([]);
    89→  const [venues, setVenues] = useState<VenueOption[]>([]);
    90→  const [eventStatus, setEventStatus] = useState<string>('DRAFT');
    91→  const [statusDialog, setStatusDialog] = useState<{ open: boolean; target: string }>({
    92→    open: false,
    93→    target: '',
    94→  });
    95→
    96→  const [form, setForm] = useState({
    97→    title: '',
    98→    description: '',
    99→    shortDescription: '',
   100→    categoryId: '',
   101→    coverImage: '',
   102→    startDate: '',
   103→    endDate: '',
   104→    startTime: '',
   105→    endTime: '',
   106→    timezone: 'UTC',
   107→    venueId: '',
   108→    venueName: '',
   109→    venueAddress: '',
   110→    venueCity: '',
   111→    venueState: '',
   112→    venueCountry: '',
   113→    venueLat: '',
   114→    venueLng: '',
   115→    isVirtual: false,
   116→    virtualUrl: '',
   117→    isPaid: false,
   118→    currency: 'USD',
   119→    capacity: '',
   120→    tags: '',
   121→  });
   122→
   123→  const [ticketTypes, setTicketTypes] = useState<TicketTypeForm[]>([]);
   124→
   125→  // Load event data and categories
   126→  useEffect(() => {
   127→    if (!selectedEventId) return;
   128→    setLoading(true);
   129→    Promise.all([
   130→      apiFetch<{
   131→        event: {
   132→          title: string;
   133→          description: string;
   134→          shortDescription: string | null;
   135→          coverImage: string | null;
   136→          startDate: string;
   137→          endDate: string | null;
   138→          startTime: string | null;
   139→          endTime: string | null;
   140→          timezone: string;
   141→          venueId: string | null;
   142→          venueName: string | null;
   143→          venueAddress: string | null;
   144→          venueCity: string | null;
   145→          venueState: string | null;
   146→          venueCountry: string | null;
   147→          venueLat: number | null;
   148→          venueLng: number | null;
   149→          isVirtual: boolean;
   150→          virtualUrl: string | null;
   151→          isPaid: boolean;
   152→          currency: string;
   153→          capacity: number | null;
   154→          status: string;
   155→          categoryId: string | null;
   156→          category: { id: string; name: string; slug: string } | null;
   157→          tags: { tag: { name: string } }[];
   158→          ticketTypes: {
   159→            id: string;
   160→            name: string;
   161→            price: number;
   162→            quantity: number;
   163→            soldCount: number;
   164→          }[];
   165→        };
   166→      }>(`/api/events/${selectedEventId}`),
   167→      apiFetch<{ categories: Category[] }>('/api/categories'),
   168→      apiFetch<{ venues: VenueOption[] }>('/api/organizer/venues').catch(() => ({ venues: [] })),
   169→    ])
   170→      .then(([eventData, catData, venueData]) => {
   171→        const e = eventData.event;
   172→        if (venueData?.venues) setVenues(venueData.venues);
   173→        setForm({
   174→          title: e.title,
   175→          description: e.description,
   176→          shortDescription: e.shortDescription || '',
   177→          categoryId: e.category?.id || '',
   178→          coverImage: e.coverImage || '',
   179→          startDate: e.startDate ? new Date(e.startDate).toISOString().split('T')[0] : '',
   180→          endDate: e.endDate ? new Date(e.endDate).toISOString().split('T')[0] : '',
   181→          startTime: e.startTime || '',
   182→          endTime: e.endTime || '',
   183→          timezone: e.timezone || 'UTC',
   184→          venueId: e.venueId || '',
   185→          venueName: e.venueName || '',
   186→          venueAddress: e.venueAddress || '',
   187→          venueCity: e.venueCity || '',
   188→          venueState: e.venueState || '',
   189→          venueCountry: e.venueCountry || '',
   190→          venueLat: e.venueLat != null ? String(e.venueLat) : '',
   191→          venueLng: e.venueLng != null ? String(e.venueLng) : '',
   192→          isVirtual: e.isVirtual,
   193→          virtualUrl: e.virtualUrl || '',
   194→          isPaid: e.isPaid,
   195→          currency: e.currency || 'USD',
   196→          capacity: e.capacity ? String(e.capacity) : '',
   197→          tags: e.tags?.map((t: { tag: { name: string } }) => t.tag.name).join(', ') || '',
   198→        });
   199→        setEventStatus(e.status);
   200→        setTicketTypes(
   201→          e.ticketTypes.map((tt) => ({
   202→            id: tt.id,
   203→            name: tt.name,
   204→            price: String(tt.price),
   205→            quantity: String(tt.quantity),
   206→          }))
   207→        );
   208→        setCategories(catData.categories || []);
   209→      })
   210→      .catch(() => toast.error('Failed to load event'))
   211→      .finally(() => setLoading(false));
   212→  }, [selectedEventId]);
   213→
   214→  const updateForm = (key: string, value: string | boolean) =>
   215→    setForm((prev) => ({ ...prev, [key]: value }));
   216→
   217→  const addTicketType = () =>
   218→    setTicketTypes((prev) => [
   219→      ...prev,
   220→      { name: '', price: '0', quantity: '50', isNew: true },
   221→    ]);
   222→  const removeTicketType = (i: number) => {
   223→    const tt = ticketTypes[i];
   224→    if (tt.isNew) {
   225→      setTicketTypes((prev) => prev.filter((_, idx) => idx !== i));
   226→    } else {
   227→      // Mark existing for deletion
   228→      setTicketTypes((prev) =>
   229→        prev.map((t, idx) => (idx === i ? { ...t, toDelete: true } : t))
   230→      );
   231→    }
   232→  };
   233→  const restoreTicketType = (i: number) => {
   234→    setTicketTypes((prev) =>
   235→      prev.map((t, idx) => (idx === i ? { ...t, toDelete: false } : t))
   236→    );
   237→  };
   238→  const updateTicketType = (i: number, key: keyof TicketTypeForm, value: string) => {
   239→    setTicketTypes((prev) =>
   240→      prev.map((t, idx) => (idx === i ? { ...t, [key]: value } : t))
   241→    );
   242→  };
   243→
   244→  const handleSubmit = async (e: React.FormEvent) => {
   245→    e.preventDefault();
   246→    if (!form.title || !form.startDate) {
   247→      toast.error('Please fill in required fields (title, date)');
   248→      return;
   249→    }
   250→    if (form.isVirtual && !form.virtualUrl) {
   251→      toast.error('Please provide a virtual event URL');
   252→      return;
   253→    }
   254→
   255→    setSaving(true);
   256→    try {
   257→      // Build PATCH body (only send changed fields)
   258→      const body: Record<string, unknown> = {
   259→        title: form.title,
   260→        description: form.description,
   261→        shortDescription: form.shortDescription || undefined,
   262→        coverImage: form.coverImage || undefined,
   263→        startDate: form.startDate,
   264→        endDate: form.endDate || undefined,
   265→        startTime: form.startTime || undefined,
   266→        endTime: form.endTime || undefined,
   267→        timezone: form.timezone,
   268→        venueId: form.venueId || undefined,
   269→        venueName: form.isVirtual ? undefined : form.venueName,
   270→        venueAddress: form.isVirtual ? undefined : form.venueAddress,
   271→        venueCity: form.isVirtual ? undefined : form.venueCity,
   272→        venueState: form.isVirtual ? undefined : form.venueState,
   273→        venueCountry: form.isVirtual ? undefined : form.venueCountry,
   274→        venueLat: form.isVirtual ? undefined : (form.venueLat ? Number(form.venueLat) : undefined),
   275→        venueLng: form.isVirtual ? undefined : (form.venueLng ? Number(form.venueLng) : undefined),
   276→        isVirtual: form.isVirtual,
   277→        virtualUrl: form.isVirtual ? form.virtualUrl : undefined,
   278→        isPaid: form.isPaid,
   279→        currency: form.currency,
   280→        capacity: form.capacity ? parseInt(form.capacity) : undefined,
   281→        categoryId: form.categoryId || undefined,
   282→        tags: form.tags
   283→          ? form.tags.split(',').map((t: string) => t.trim()).filter(Boolean)
   284→          : [],
   285→      };
   286→
   287→      await apiFetch(`/api/events/${selectedEventId}`, {
   288→        method: 'PATCH',
   289→        body: JSON.stringify(body),
   290→      });
   291→
   292→      // Handle ticket type changes
   293→      // Delete marked ticket types
   294→      for (const tt of ticketTypes.filter((t) => t.toDelete && t.id)) {
   295→        await apiFetch(
   296→          `/api/events/${selectedEventId}/ticket-types/${tt.id}`,
   297→          { method: 'DELETE' }
   298→        );
   299→      }
   300→
   301→      // Add new ticket types
   302→      for (const tt of ticketTypes.filter((t) => t.isNew)) {
   303→        await apiFetch(`/api/events/${selectedEventId}/ticket-types`, {
   304→          method: 'POST',
   305→          body: JSON.stringify({
   306→            name: tt.name,
   307→            price: parseFloat(tt.price) || 0,
   308→            quantity: parseInt(tt.quantity) || 50,
   309→            currency: form.currency,
   310→          }),
   311→        });
   312→      }
   313→
   314→      // Update existing ticket types (changed values)
   315→      for (const tt of ticketTypes.filter((t) => t.id && !t.isNew && !t.toDelete)) {
   316→        await apiFetch(
   317→          `/api/events/${selectedEventId}/ticket-types/${tt.id}`,
   318→          {
   319→            method: 'PATCH',
   320→            body: JSON.stringify({
   321→              name: tt.name,
   322→              price: parseFloat(tt.price),
   323→              quantity: parseInt(tt.quantity),
   324→            }),
   325→          }
   326→        );
   327→      }
   328→
   329→      toast.success('Event updated successfully!');
   330→      navigate('organizer-events');
   331→    } catch (err: unknown) {
   332→      const message = err instanceof Error ? err.message : 'Failed to update event';
   333→      toast.error(message);
   334→    } finally {
   335→      setSaving(false);
   336→    }
   337→  };
   338→
   339→  // Status transition handlers
   340→  const handleStatusChange = async (targetStatus: string) => {
   341→    setStatusDialog({ open: false, target: '' });
   342→    setSaving(true);
   343→    try {
   344→      await apiFetch(`/api/events/${selectedEventId}`, {
   345→        method: 'PATCH',
   346→        body: JSON.stringify({ status: targetStatus }),
   347→      });
   348→      setEventStatus(targetStatus);
   349→      toast.success(`Event status changed to ${targetStatus}`);
   350→    } catch (err: unknown) {
   351→      const message = err instanceof Error ? err.message : 'Failed to change status';
   352→      toast.error(message);
   353→    } finally {
   354→      setSaving(false);
   355→    }
   356→  };
   357→
   358→  if (loading) {
   359→    return (
   360→      <div className="space-y-6 max-w-3xl">
   361→        <div className="h-8 w-48 bg-muted animate-pulse rounded" />
   362→        <div className="h-64 bg-muted animate-pulse rounded-xl" />
   363→        <div className="h-48 bg-muted animate-pulse rounded-xl" />
   364→      </div>
   365→    );
   366→  }
   367→
   368→  // Determine available status transitions based on current status
   369→  const getStatusActions = () => {
   370→    if (!user) return [];
   371→    const actions: { target: string; label: string; icon: React.ReactNode; variant: string }[] = [];
   372→
   373→    if (user.role === 'ORGANIZER' || user.role === 'SUPER_ADMIN') {
   374→      if (eventStatus === 'DRAFT') {
   375→        actions.push({
   376→          target: 'PENDING',
   377→          label: 'Submit for Review',
   378→          icon: <Send className="h-4 w-4" />,
   379→          variant: 'default',
   380→        });
   381→      }
   382→      if (eventStatus === 'PENDING') {
   383→        actions.push({
   384→          target: 'DRAFT',
   385→          label: 'Withdraw from Review',
   386→          icon: <XCircle className="h-4 w-4" />,
   387→          variant: 'outline',
   388→        });
   389→      }
   390→      if (eventStatus === 'PUBLISHED') {
   391→        actions.push({
   392→          target: 'CANCELLED',
   393→          label: 'Cancel Event',
   394→          icon: <XCircle className="h-4 w-4" />,
   395→          variant: 'destructive',
   396→        });
   397→      }
   398→    }
   399→
   400→    // SUPER_ADMIN can additionally publish and reject directly
   401→    if (user.role === 'SUPER_ADMIN') {
   402→      if (eventStatus === 'PENDING') {
   403→        actions.push({
   404→          target: 'PUBLISHED',
   405→          label: 'Approve & Publish',
   406→          icon: <Send className="h-4 w-4" />,
   407→          variant: 'default',
   408→        });
   409→        actions.push({
   410→          target: 'REJECTED',
   411→          label: 'Reject Event',
   412→          icon: <XCircle className="h-4 w-4" />,
   413→          variant: 'destructive',
   414→        });
   415→      }
   416→      if (eventStatus === 'DRAFT') {
   417→        actions.push({
   418→          target: 'PUBLISHED',
   419→          label: 'Publish Directly',
   420→          icon: <Send className="h-4 w-4" />,
   421→          variant: 'default',
   422→        });
   423→      }
   424→    }
   425→
   426→    return actions;
   427→  };
   428→
   429→  return (
   430→    <form onSubmit={handleSubmit} className="space-y-6 max-w-3xl">
   431→      <div className="flex items-center justify-between">
   432→        <h1 className="text-2xl font-bold">Edit Event</h1>
   433→        <Badge className={STATUS_COLORS[eventStatus] || ''}>{eventStatus}</Badge>
   434→      </div>
   435→
   436→      {/* Status Workflow Actions */}
   437→      {getStatusActions().length > 0 && (
   438→        <Card>
   439→          <CardHeader>
   440→            <CardTitle className="text-base">Event Status</CardTitle>
   441→            <CardDescription>Change the event status to control visibility</CardDescription>
   442→          </CardHeader>
   443→          <CardContent>
   444→            <div className="flex flex-wrap gap-2">
   445→              {getStatusActions().map((action) => (
   446→                <Button
   447→                  key={action.target}
   448→                  type="button"
   449→                  variant={action.variant === 'destructive' ? 'destructive' : action.variant === 'default' ? 'default' : 'outline'}
   450→                  className={action.variant === 'default' ? 'bg-emerald-600 hover:bg-emerald-700' : ''}
   451→                  onClick={() =>
   452→                    setStatusDialog({ open: true, target: action.target })
   453→                  }
   454→                  disabled={saving}
   455→                >
   456→                  {action.icon}
   457→                  <span className="ml-2">{action.label}</span>
   458→                </Button>
   459→              ))}
   460→            </div>
   461→          </CardContent>
   462→        </Card>
   463→      )}
   464→
   465→      {/* Basic Info */}
   466→      <Card>
   467→        <CardHeader>
   468→          <CardTitle className="flex items-center gap-2">
   469→            <Info className="h-5 w-5 text-emerald-500" /> Basic Information
   470→          </CardTitle>
   471→        </CardHeader>
   472→        <CardContent className="space-y-4">
   473→          <div className="space-y-2">
   474→            <Label htmlFor="title">Title *</Label>
   475→            <Input
   476→              id="title"
   477→              placeholder="Event title"
   478→              value={form.title}
   479→              onChange={(e) => updateForm('title', e.target.value)}
   480→              required
   481→            />
   482→          </div>
   483→          <div className="space-y-2">
   484→            <Label htmlFor="shortDesc">Short Description</Label>
   485→            <Input
   486→              id="shortDesc"
   487→              placeholder="Brief summary (1-2 sentences)"
   488→              value={form.shortDescription}
   489→              onChange={(e) => updateForm('shortDescription', e.target.value)}
   490→            />
   491→          </div>
   492→          <div className="space-y-2">
   493→            <Label htmlFor="desc">Full Description</Label>
   494→            <Textarea
   495→              id="desc"
   496→              placeholder="Detailed event description..."
   497→              rows={5}
   498→              value={form.description}
   499→              onChange={(e) => updateForm('description', e.target.value)}
   500→            />
   501→          </div>
   502→          <div className="grid sm:grid-cols-2 gap-4">
   503→            <div className="space-y-2">
   504→              <Label>Category</Label>
   505→              <Select
   506→                value={form.categoryId}
   507→                onValueChange={(v) => updateForm('categoryId', v)}
   508→              >
   509→                <SelectTrigger>
   510→                  <SelectValue placeholder="Select category" />
   511→                </SelectTrigger>
   512→                <SelectContent>
   513→                  {categories.map((c) => (
   514→                    <SelectItem key={c.id} value={c.id}>
   515→                      {c.name}
   516→                    </SelectItem>
   517→                  ))}
   518→                </SelectContent>
   519→              </Select>
   520→            </div>
   521→            <div className="space-y-2">
   522→              <Label htmlFor="coverImg">Cover Image URL</Label>
   523→              <Input
   524→                id="coverImg"
   525→                placeholder="https://..."
   526→                value={form.coverImage}
   527→                onChange={(e) => updateForm('coverImage', e.target.value)}
   528→              />
   529→            </div>
   530→          </div>
   531→          <div className="space-y-2">
   532→            <Label htmlFor="tags">Tags (comma-separated)</Label>
   533→            <Input
   534→              id="tags"
   535→              placeholder="music, tech, networking"
   536→              value={form.tags}
   537→              onChange={(e) => updateForm('tags', e.target.value)}
   538→            />
   539→          </div>
   540→        </CardContent>
   541→      </Card>
   542→
   543→      {/* Date & Time */}
   544→      <Card>
   545→        <CardHeader>
   546→          <CardTitle className="flex items-center gap-2">
   547→            <Clock className="h-5 w-5 text-teal-500" /> Date & Time
   548→          </CardTitle>
   549→        </CardHeader>
   550→        <CardContent className="space-y-4">
   551→          <div className="grid sm:grid-cols-2 gap-4">
   552→            <div className="space-y-2">
   553→              <Label htmlFor="startDate">Start Date *</Label>
   554→              <Input
   555→                id="startDate"
   556→                type="date"
   557→                value={form.startDate}
   558→                onChange={(e) => updateForm('startDate', e.target.value)}
   559→                required
   560→              />
   561→            </div>
   562→            <div className="space-y-2">
   563→              <Label htmlFor="endDate">End Date</Label>
   564→              <Input
   565→                id="endDate"
   566→                type="date"
   567→                value={form.endDate}
   568→                onChange={(e) => updateForm('endDate', e.target.value)}
   569→              />
   570→            </div>
   571→          </div>
   572→          <div className="grid sm:grid-cols-2 gap-4">
   573→            <div className="space-y-2">
   574→              <Label htmlFor="startTime">Start Time</Label>
   575→              <Input
   576→                id="startTime"
   577→                type="time"
   578→                value={form.startTime}
   579→                onChange={(e) => updateForm('startTime', e.target.value)}
   580→              />
   581→            </div>
   582→            <div className="space-y-2">
   583→              <Label htmlFor="endTime">End Time</Label>
   584→              <Input
   585→                id="endTime"
   586→                type="time"
   587→                value={form.endTime}
   588→                onChange={(e) => updateForm('endTime', e.target.value)}
   589→              />
   590→            </div>
   591→          </div>
   592→          <div className="space-y-2">
   593→            <Label>Timezone</Label>
   594→            <Select
   595→              value={form.timezone}
   596→              onValueChange={(v) => updateForm('timezone', v)}
   597→            >
   598→              <SelectTrigger>
   599→                <SelectValue />
   600→              </SelectTrigger>
   601→              <SelectContent>
   602→                <SelectItem value="UTC">UTC</SelectItem>
   603→                <SelectItem value="America/New_York">Eastern (ET)</SelectItem>
   604→                <SelectItem value="America/Chicago">Central (CT)</SelectItem>
   605→                <SelectItem value="America/Denver">Mountain (MT)</SelectItem>
   606→                <SelectItem value="America/Los_Angeles">Pacific (PT)</SelectItem>
   607→                <SelectItem value="Europe/London">London (GMT)</SelectItem>
   608→              </SelectContent>
   609→            </Select>

   614→      {/* Venue */}

   619→              <Globe className="h-5 w-5 text-cyan-500" />

   621→              <MapPin className="h-5 w-5 text-amber-500" />

   627→          <div className="flex items-center gap-3">
   628→            <Switch
   629→              checked={form.isVirtual}
   630→              onCheckedChange={(v) => updateForm('isVirtual', v)}
   631→            />
   632→            <Label>This is a virtual/online event</Label>
   633→          </div>
   634→          {form.isVirtual ? (
   635→            <div className="space-y-2">
   636→              <Label htmlFor="virtualUrl">Virtual Event URL *</Label>

   638→                id="virtualUrl"
   639→                placeholder="https://zoom.us/..."
   640→                value={form.virtualUrl}
   641→                onChange={(e) => updateForm('virtualUrl', e.target.value)}

   645→            <>
   646→              {/* Venue Selector */}

   658→                          ...prev,
   659→                          venueId: venue.id,
   660→                          venueName: venue.name,
   661→                          venueAddress: venue.address,
   662→                          venueCity: venue.city,
   663→                          venueState: venue.state || '',
   664→                          venueCountry: venue.country,
   665→                          venueLat: venue.lat != null ? String(venue.lat) : '',
   666→                          venueLng: venue.lng != null ? String(venue.lng) : '',

   671→                >
   672→                  <SelectTrigger><SelectValue placeholder="No venue selected" /></SelectTrigger>
   673→                  <SelectContent>
   674→                    <SelectItem value="__none__">No venue (custom)</SelectItem>
   675→                    {venues.map(v => <SelectItem key={v.id} value={v.id}>{v.name} — {v.city}</SelectItem>)}

   677→                </Select>
   678→              </div>
   679→
   680→              {/* Inline Venue Fields */}
   681→              <div className="grid sm:grid-cols-2 gap-4">
   682→                <div className="space-y-2">
   683→                  <Label htmlFor="venueName">Venue Name</Label>
   684→                  <Input
   685→                    id="venueName"
   686→                    placeholder="Convention Center"
   687→                    value={form.venueName}
   688→                    onChange={(e) => updateForm('venueName', e.target.value)}
   689→                    disabled={!!form.venueId}
   690→                  />
   691→                </div>
   692→                <div className="space-y-2">
   693→                  <Label htmlFor="venueAddr">Address</Label>
   694→                  <Input
   695→                    id="venueAddr"
   696→                    placeholder="123 Main St"
   697→                    value={form.venueAddress}
   698→                    onChange={(e) => updateForm('venueAddress', e.target.value)}
   699→                    disabled={!!form.venueId}
   700→                  />
   701→                </div>
   702→                <div className="space-y-2">
   703→                  <Label htmlFor="venueCity">City</Label>
   704→                  <Input
   705→                    id="venueCity"
   706→                    placeholder="New York"
   707→                    value={form.venueCity}
   708→                    onChange={(e) => updateForm('venueCity', e.target.value)}
   709→                    disabled={!!form.venueId}
   710→                  />
   711→                </div>
   712→                <div className="space-y-2">
   713→                  <Label htmlFor="venueState">State</Label>
   714→                  <Input
   715→                    id="venueState"
   716→                    placeholder="NY"
   717→                    value={form.venueState}
   718→                    onChange={(e) => updateForm('venueState', e.target.value)}
   719→                    disabled={!!form.venueId}
   720→                  />
   721→                </div>
   722→                <div className="space-y-2 sm:col-span-2">
   723→                  <Label htmlFor="venueCountry">Country</Label>
   724→                  <Input
   725→                    id="venueCountry"
   726→                    placeholder="United States"
   727→                    value={form.venueCountry}
   728→                    onChange={(e) => updateForm('venueCountry', e.target.value)}
   729→                    disabled={!!form.venueId}
   730→                  />
   731→                </div>
   732→              </div>
   733→            </>
   734→          )}
   735→        </CardContent>
   736→      </Card>
   737→
   738→      {/* Ticketing */}
   739→      <Card>
   740→        <CardHeader>
   741→          <CardTitle className="flex items-center gap-2">
   742→            <DollarSign className="h-5 w-5 text-emerald-500" /> Ticketing
   743→          </CardTitle>
   744→        </CardHeader>
   745→        <CardContent className="space-y-4">
   746→          <div className="flex items-center gap-3">
   747→            <Switch
   748→              checked={form.isPaid}
   749→              onCheckedChange={(v) => updateForm('isPaid', v)}
   750→            />
   751→            <Label>This is a paid event</Label>
   752→          </div>
   753→          {form.isPaid && (
   754→            <>
   755→              <div className="grid sm:grid-cols-2 gap-4">
   756→                <div className="space-y-2">
   757→                  <Label>Currency</Label>
   758→                  <Select
   759→                    value={form.currency}
   760→                    onValueChange={(v) => updateForm('currency', v)}
   761→                  >
   762→                    <SelectTrigger>
   763→                      <SelectValue />
   764→                    </SelectTrigger>
   765→                    <SelectContent>
   766→                      <SelectItem value="USD">USD ($)</SelectItem>
   767→                      <SelectItem value="EUR">EUR</SelectItem>
   768→                      <SelectItem value="GBP">GBP</SelectItem>
   769→                    </SelectContent>
   770→                  </Select>
   771→                </div>
   772→                <div className="space-y-2">
   773→                  <Label htmlFor="capacity">Total Capacity</Label>
   774→                  <Input
   775→                    id="capacity"
   776→                    type="number"
   777→                    placeholder="100"
   778→                    value={form.capacity}
   779→                    onChange={(e) => updateForm('capacity', e.target.value)}
   780→                  />
   781→                </div>

   784→              <div className="space-y-3">
   785→                <div className="flex items-center justify-between">
   786→                  <Label>Ticket Types</Label>
   787→                  <Button
   788→                    type="button"
   789→                    variant="outline"
   790→                    size="sm"
   791→                    onClick={addTicketType}
   792→                  >
   793→                    <Plus className="h-4 w-4 mr-1" /> Add Type
   794→                  </Button>
   795→                </div>
   796→                {ticketTypes.map((tt, i) => (
   797→                  <div
   798→                    key={tt.id || i}
   799→                    className={`grid grid-cols-[1fr_100px_100px_40px] gap-2 items-end ${tt.toDelete ? 'opacity-50' : ''}`}

   802→                      {i === 0 && <Label className="text-xs">Name</Label>}

   804→                        placeholder="VIP, General..."

   808→                      />
   809→                    </div>
   810→                    <div className="space-y-1">
   811→                      {i === 0 && <Label className="text-xs">Price</Label>}
   812→                      <Input
   813→                        type="number"
   814→                        placeholder="0"
   815→                        value={tt.price}
   816→                        onChange={(e) => updateTicketType(i, 'price', e.target.value)}

   818→                      />
   819→                    </div>
   820→                    <div className="space-y-1">
   821→                      {i === 0 && <Label className="text-xs">Qty</Label>}
   822→                      <Input
   823→                        type="number"
   824→                        placeholder="50"
   825→                        value={tt.quantity}
   826→                        onChange={(e) => updateTicketType(i, 'quantity', e.target.value)}
   827→                        disabled={tt.toDelete}
   828→                      />
   829→                    </div>
   830→                    {tt.toDelete ? (
   831→                      <Button
   832→                        type="button"
   833→                        variant="ghost"
   834→                        size="icon"
   835→                        className="text-emerald-500"
   836→                        onClick={() => restoreTicketType(i)}
   837→                      >
   838→                        <AlertTriangle className="h-4 w-4" />

   840→                    ) : (
   841→                      <Button
   842→                        type="button"
   843→                        variant="ghost"
   844→                        size="icon"
   845→                        className="text-red-500"

   847→                        disabled={ticketTypes.filter((t) => !t.toDelete).length <= 1}
   848→                      >
   849→                        <Trash2 className="h-4 w-4" />
   850→                      </Button>
   851→                    )}
   852→                  </div>
   853→                ))}
   854→              </div>
   855→            </>
   856→          )}
   857→        </CardContent>
   858→      </Card>
   859→
   860→      {/* Manage Event Content */}
   861→      <Card>
   862→        <CardHeader>
   863→          <CardTitle className="flex items-center gap-2">
   864→            <Calendar className="h-5 w-5 text-emerald-500" /> Manage Event Content
   865→          </CardTitle>
   866→          <CardDescription>Manage sessions, participants, and media for this event</CardDescription>
   867→        </CardHeader>
   868→        <CardContent>
   869→          <div className="grid sm:grid-cols-3 gap-3">
   870→            <Button
   871→              type="button"
   872→              variant="outline"
   873→              className="h-auto py-4 flex-col gap-2"
   874→              onClick={() => navigate('organizer-event-content')}
   875→            >
   876→              <Calendar className="h-6 w-6 text-emerald-600" />
   877→              <span className="font-medium">Program</span>
   878→              <span className="text-xs text-muted-foreground">Sessions & Schedule</span>
   879→            </Button>
   880→            <Button
   881→              type="button"
   882→              variant="outline"
   883→              className="h-auto py-4 flex-col gap-2"
   884→              onClick={() => navigate('organizer-event-content')}
   885→            >
   886→              <Users className="h-6 w-6 text-emerald-600" />
   887→              <span className="font-medium">Participants</span>
   888→              <span className="text-xs text-muted-foreground">Speakers & Artists</span>
   889→            </Button>
   890→            <Button
   891→              type="button"
   892→              variant="outline"
   893→              className="h-auto py-4 flex-col gap-2"
   894→              onClick={() => navigate('organizer-event-content')}
   895→            >
   896→              <ImageIcon className="h-6 w-6 text-emerald-600" />
   897→              <span className="font-medium">Media</span>
   898→              <span className="text-xs text-muted-foreground">Images & Videos</span>
   899→            </Button>
   900→          </div>
   901→        </CardContent>
   902→      </Card>
   903→
   904→      <div className="flex gap-4">
   905→        <Button
   906→          type="button"
   907→          variant="outline"
   908→          onClick={() => navigate('organizer-events')}
   909→        >
   910→          Cancel
   911→        </Button>
   912→        <Button
   913→          type="submit"
   914→          className="bg-emerald-600 hover:bg-emerald-700"
   915→          disabled={saving}
   916→        >
   917→          {saving ? (
   918→            <Loader2 className="mr-2 h-4 w-4 animate-spin" />
   919→          ) : (
   920→            <Save className="mr-2 h-4 w-4" />
   921→          )}
   922→          Save Changes
   923→        </Button>
   924→      </div>
   925→
   926→      {/* Status Change Confirmation Dialog */}
   927→      <Dialog
   928→        open={statusDialog.open}
   929→        onOpenChange={(open) => setStatusDialog({ open, target: statusDialog.target })}
   930→      >
   931→        <DialogContent>
   932→          <DialogHeader>
   933→            <DialogTitle>Confirm Status Change</DialogTitle>
   934→            <DialogDescription>
   935→              Are you sure you want to change the event status from{' '}
   936→              <Badge className={STATUS_COLORS[eventStatus]}>{eventStatus}</Badge> to{' '}
   937→              <Badge className={STATUS_COLORS[statusDialog.target]}>
   938→                {statusDialog.target}
   939→              </Badge>
   940→              ?
   941→              {statusDialog.target === 'CANCELLED' &&
   942→                ' This will make the event unavailable for booking.'}
   943→              {statusDialog.target === 'PENDING' &&
   944→                ' This will submit the event for admin review.'}
   945→              {statusDialog.target === 'PUBLISHED' &&
   946→                ' This will make the event publicly visible.'}
   947→            </DialogDescription>
   948→          </DialogHeader>
   949→          <DialogFooter>
   950→            <Button
   951→              variant="outline"
   952→              onClick={() => setStatusDialog({ open: false, target: '' })}
   953→            >
   954→              Cancel
   955→            </Button>
   956→            <Button
   957→              className={
   958→                statusDialog.target === 'CANCELLED' || statusDialog.target === 'REJECTED'
   959→                  ? 'bg-red-600 hover:bg-red-700'
   960→                  : 'bg-emerald-600 hover:bg-emerald-700'
   961→              }
   962→              onClick={() => handleStatusChange(statusDialog.target)}
   963→              disabled={saving}
   964→            >
   965→              {saving && <Loader2 className="mr-2 h-4 w-4 animate-spin" />}
   966→              Confirm
   967→            </Button>
   968→          </DialogFooter>
   969→        </DialogContent>

   971→    </form>
