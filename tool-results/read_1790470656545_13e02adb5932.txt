   742→   742→function formatDistanceToNow(date: Date) {
   743→   743→  const seconds = Math.floor((Date.now() - date.getTime()) / 1000);
   744→   744→  if (seconds < 60) return 'just now';
   745→   745→  const minutes = Math.floor(seconds / 60);
   746→   746→  if (minutes < 60) return `${minutes}m ago`;
   747→   747→  const hours = Math.floor(minutes / 60);
   748→   748→  if (hours < 24) return `${hours}h ago`;
   749→   749→  const days = Math.floor(hours / 24);
   750→   750→  if (days < 30) return `${days}d ago`;
   751→   751→  const months = Math.floor(days / 30);
   752→   752→  return `${months}mo ago`;
   753→   753→}
