   159→export function EventDetail() {
   160→  const { selectedEventId, goBack, navigate, user } = useAppStore();
   161→  const [event, setEvent] = useState<Event | null>(null);
   162→  const [loading, setLoading] = useState(true);
   163→  const [bookDialog, setBookDialog] = useState(false);
   164→  const [selectedTicket, setSelectedTicket] = useState<TicketType | null>(null);
   165→  const [quantity, setQuantity] = useState(1);
   166→  const [booking, setBooking] = useState(false);
   167→  const [relatedEvents, setRelatedEvents] = useState<any[]>([]);
   168→  const [reviewDialog, setReviewDialog] = useState(false);
   169→  const [reviewRating, setReviewRating] = useState(5);
   170→  const [reviewComment, setReviewComment] = useState('');
   171→  const [submittingReview, setSubmittingReview] = useState(false);
   172→  const [userHasReviewed, setUserHasReviewed] = useState(false);
   173→
   174→  // Phase 4F: Use single API response — event already includes sessions/participants/media
   175→  // No separate N+1 fetches needed
   176→
   177→  useEffect(() => {
   178→    if (!selectedEventId) return;
   179→    setLoading(true);
   180→    apiFetch<{ event: Event }>(`/api/events/${selectedEventId}`)
   181→      .then(data => {
   182→        setEvent(data.event);
   183→        // Check if current user has already reviewed
   184→        if (user && data.event.reviews) {
   185→          setUserHasReviewed(data.event.reviews.some((r: Review) => r.user?.id === user.id));
   186→        }
   187→        // Phase 4F: Set document title for SEO
   188→        if (data.event.title) {
   189→          document.title = `${data.event.title} — AppleCalendar`;
   190→        }
   191→      })
   192→      .catch(() => toast.error('Failed to load event'))
   193→      .finally(() => setLoading(false));
   194→  }, [selectedEventId, user?.id]);
   195→
   196→  // Cleanup: restore document title on unmount
   197→  useEffect(() => {
   198→    return () => { document.title = 'AppleCalendar — Discover Amazing Events'; };
   199→  }, []);
   200→
   201→  useEffect(() => {
   202→    if (!event?.category?.slug) return;
   203→    apiFetch<{ events: any[] }>(`/api/events?category=${event.category.slug}&limit=4&featured=false`)
   204→      .then(data => setRelatedEvents(data.events.filter((e: any) => e.id !== event.id).slice(0, 3)))
   205→      .catch(() => {});
   206→  }, [event?.category?.slug]);
   207→
   208→  // Derived data from single API response
   209→  const sessions = event?.sessions || [];
   210→  const participants = event?.participants || [];
   211→  const media = event?.media || [];
   212→
   213→  // Phase 4F: Bookability check
   214→  const canBook = event?.status === 'PUBLISHED' && event.isBookable !== false;
   215→  const isCancelled = event?.status === 'CANCELLED';
   216→  const isCompleted = event?.status === 'COMPLETED';
   217→
   218→  const handleBook = (ticket: TicketType) => {
   219→    if (!user) {
   220→      navigate('login');
   221→      toast.error('Please login to book tickets');
   222→      return;
   223→    }
   224→    setSelectedTicket(ticket);
   225→    setQuantity(1);
   226→    setBookDialog(true);
   227→  };
   228→
   229→  const confirmBooking = async () => {
   230→    if (!selectedTicket || !selectedEventId) return;
   231→    setBooking(true);
   232→    try {
   233→      await apiFetch('/api/events/' + selectedEventId + '/book', {
   234→        method: 'POST',
   235→        body: JSON.stringify({ ticketTypeId: selectedTicket.id, quantity }),
   236→      });
   237→      toast.success('Booking confirmed!');
   238→      setBookDialog(false);
   239→      const data = await apiFetch<{ event: Event }>(`/api/events/${selectedEventId}`);
   240→      setEvent(data.event);
   241→    } catch (err: any) {
   242→      toast.error(err.message || 'Booking failed');
   243→    } finally {
   244→      setBooking(false);
   245→    }
   246→  };
   247→
   248→  const avgRating = event?.reviews?.length
   249→    ? event.reviews.reduce((s, r) => s + r.rating, 0) / event.reviews.length
   250→    : 0;
   251→
   252→  const submitReview = async () => {
   253→    if (!selectedEventId) return;
   254→    setSubmittingReview(true);
   255→    try {
   256→      await apiFetch(`/api/events/${selectedEventId}/reviews`, {
   257→        method: 'POST',
   258→        body: JSON.stringify({ rating: reviewRating, comment: reviewComment || undefined }),
   259→      });
   260→      toast.success('Review submitted!');
   261→      setReviewDialog(false);
   262→      setReviewRating(5);
   263→      setReviewComment('');
   264→      setUserHasReviewed(true);
   265→      const data = await apiFetch<{ event: Event }>(`/api/events/${selectedEventId}`);
   266→      setEvent(data.event);
   267→    } catch (err: any) {
   268→      toast.error(err.message || 'Failed to submit review');
   269→    } finally {
   270→      setSubmittingReview(false);
   271→    }
   272→  };
   273→
   274→  // Session date grouping
   275→  const sessionsByDate = sessions.reduce<Record<string, EventSession[]>>((acc, session) => {
   276→    const dateKey = session.date ? format(new Date(session.date), 'yyyy-MM-dd') : 'no-date';
   277→    if (!acc[dateKey]) acc[dateKey] = [];
   278→    acc[dateKey].push(session);
   279→    return acc;
   280→  }, {});
   281→
   282→  // Media grouping by category
   283→  const posterMedia = media.filter(m => m.category === 'POSTER' || m.category === 'COVER');
   284→  const galleryMedia = media.filter(m => m.category === 'GALLERY');
   285→  const videoMedia = media.filter(m => m.category === 'PROMOTIONAL_VIDEO' || m.type === 'VIDEO');
   286→  const otherMedia = media.filter(m =>
   287→    !['POSTER', 'COVER', 'GALLERY', 'PROMOTIONAL_VIDEO'].includes(m.category) && m.type !== 'VIDEO'
   288→  );
   289→
   290→  if (loading) {
   291→    return (
   292→      <div className="space-y-6">
   293→        <Skeleton className="h-8 w-24" />
   294→        <Skeleton className="h-72 w-full rounded-xl" />
   295→        <div className="grid md:grid-cols-3 gap-6">
   296→          <div className="md:col-span-2 space-y-4">
   297→            <Skeleton className="h-8 w-3/4" />
   298→            <Skeleton className="h-4 w-1/2" />
   299→            <Skeleton className="h-40 w-full" />
   300→          </div>
   301→          <Skeleton className="h-80 rounded-xl" />
   302→        </div>
   303→      </div>
   304→    );
   305→  }
   306→
   307→  if (!event) {
   308→    return (
   309→      <div className="text-center py-20">
   310→        <CalendarDays className="h-12 w-12 text-muted-foreground mx-auto mb-4" />
   311→        <h2 className="text-xl font-semibold mb-2">Event not found</h2>
   312→        <p className="text-muted-foreground text-sm mb-4">This event may not exist or is not publicly available.</p>
   313→        <Button variant="outline" onClick={goBack}>Go Back</Button>
   314→      </div>
   315→    );
   316→  }
   317→
   318→  // Get tag name (handles both {tag: {name}} and {name} shapes)
   319→  const getTagName = (tag: any): string => tag.tag?.name || tag.name || '';
   320→
   321→  return (
   322→    <motion.div
   323→      initial={{ opacity: 0, y: 10 }}
   324→      animate={{ opacity: 1, y: 0 }}
   325→      transition={{ duration: 0.3 }}
   326→      className="space-y-6"
   327→    >
   328→      {/* Back button */}
   329→      <Button variant="ghost" size="sm" onClick={goBack} className="gap-2" aria-label="Go back to events">
   330→        <ArrowLeft className="h-4 w-4" /> Back
   331→      </Button>
   332→
   333→      {/* Phase 4F: Event Status Banner */}
   334→      {isCancelled && (
   335→        <div className="flex items-center gap-3 p-4 rounded-lg bg-red-50 dark:bg-red-950 border border-red-200 dark:border-red-800" role="alert">
   336→          <XCircle className="h-6 w-6 text-red-500 flex-shrink-0" />
   337→          <div>
   338→            <h3 className="font-semibold text-red-700 dark:text-red-300">This event has been cancelled</h3>
   339→            <p className="text-sm text-red-600 dark:text-red-400">Bookings are no longer available for this event.</p>
   340→          </div>
   341→        </div>
   342→      )}
   343→      {isCompleted && (
   344→        <div className="flex items-center gap-3 p-4 rounded-lg bg-blue-50 dark:bg-blue-950 border border-blue-200 dark:border-blue-800">
   345→          <CheckCircle2 className="h-6 w-6 text-blue-500 flex-shrink-0" />
   346→          <div>
   347→            <h3 className="font-semibold text-blue-700 dark:text-blue-300">This event has taken place</h3>
   348→            <p className="text-sm text-blue-600 dark:text-blue-400">You can still view event details and leave a review.</p>
   349→          </div>
   350→        </div>
   351→      )}
   352→
   353→      {/* Hero Image */}
   354→      <div className="relative h-64 md:h-80 lg:h-96 rounded-xl overflow-hidden">
   355→        {event.coverImage ? (
   356→          <img src={event.coverImage} alt={`${event.title} event cover`} className="w-full h-full object-cover" />
   357→        ) : (
   358→          <div className="w-full h-full bg-gradient-to-br from-emerald-200 to-teal-200 flex items-center justify-center">
   359→            <CalendarDays className="h-16 w-16 text-emerald-300" />
   360→          </div>
   361→        )}
   362→        <div className="absolute inset-0 bg-gradient-to-t from-black/60 to-transparent" />
   363→        <div className="absolute bottom-4 left-6 right-6 text-white">
   364→          <div className="flex items-center gap-2 mb-2 flex-wrap">
   365→            <Badge className="bg-emerald-600">{event.category?.name}</Badge>
   366→            {event.isFeatured && <Badge className="bg-amber-500">Featured</Badge>}
   367→            {event.isVirtual && <Badge className="bg-cyan-600">Virtual</Badge>}
   368→          </div>
   369→          <h1 className="text-2xl md:text-3xl lg:text-4xl font-bold">{event.title}</h1>
   370→          {event.shortDescription && (
   371→            <p className="text-white/80 text-sm md:text-base mt-1 line-clamp-2">{event.shortDescription}</p>
   372→          )}
   373→        </div>
   374→      </div>
   375→
   376→      <div className="grid md:grid-cols-3 gap-6">
   377→        {/* Main Content */}
   378→        <div className="md:col-span-2 space-y-6">
   379→          {/* Date / Location / Type Info Cards */}
   380→          <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   381→            <Card className="p-4">
   382→              <div className="flex items-center gap-3">
   383→                <div className="p-2 rounded-lg bg-emerald-50 text-emerald-600" aria-hidden="true">
   384→                  <CalendarDays className="h-5 w-5" />
   385→                </div>
   386→                <div>
   387→                  <p className="text-sm text-muted-foreground">Date</p>
   388→                  <p className="font-medium">{format(new Date(event.startDate), 'EEEE, MMM d, yyyy')}</p>
   389→                  {event.endDate && event.endDate !== event.startDate && (
   390→                    <p className="text-sm text-muted-foreground">to {format(new Date(event.endDate), 'EEE, MMM d, yyyy')}</p>
   391→                  )}
   392→                </div>
   393→              </div>
   394→            </Card>
   395→            <Card className="p-4">
   396→              <div className="flex items-center gap-3">
   397→                <div className="p-2 rounded-lg bg-teal-50 text-teal-600" aria-hidden="true">
   398→                  <Clock className="h-5 w-5" />
   399→                </div>
   400→                <div>
   401→                  <p className="text-sm text-muted-foreground">Time</p>
   402→                  <p className="font-medium">{event.startTime || 'TBD'} {event.endTime ? `- ${event.endTime}` : ''}</p>
   403→                  {event.timezone && event.timezone !== 'UTC' && (
   404→                    <p className="text-xs text-muted-foreground">{event.timezone}</p>
   405→                  )}
   406→                </div>
   407→              </div>
   408→            </Card>
   409→            <Card className="p-4">
   410→              <div className="flex items-center gap-3">
   411→                <div className={`p-2 rounded-lg ${event.isVirtual ? 'bg-cyan-50 text-cyan-600' : 'bg-amber-50 text-amber-600'}`} aria-hidden="true">
   412→                  {event.isVirtual ? <Globe className="h-5 w-5" /> : <MapPin className="h-5 w-5" />}
   413→                </div>
   414→                <div className="min-w-0">
   415→                  <p className="text-sm text-muted-foreground">{event.isVirtual ? 'Virtual Event' : 'Location'}</p>
   416→                  {event.isVirtual ? (
   417→                    event.virtualUrl ? (
   418→                      <a
   419→                        href={event.virtualUrl}
   420→                        target="_blank"
   421→                        rel="noopener noreferrer"
   422→                        className="font-medium text-cyan-600 hover:underline inline-flex items-center gap-1"
   423→                      >
   424→                        Join Online <ExternalLink className="h-3 w-3" />
   425→                      </a>
   426→                    ) : (
   427→                      <p className="font-medium text-cyan-600">Online</p>
   428→                    )
   429→                  ) : event.venue ? (
   430→                    <div>
   431→                      <p className="font-medium">{event.venue.name}</p>
   432→                      <p className="text-xs text-muted-foreground truncate">
   433→                        {event.venue.address}{event.venue.city ? `, ${event.venue.city}` : ''}{event.venue.state ? `, ${event.venue.state}` : ''}
   434→                      </p>
   435→                      {(event.venue.googleMapsUrl || (event.venue.lat != null && event.venue.lng != null)) && (
   436→                        <a
   437→                          href={event.venue.googleMapsUrl || `https://www.google.com/maps?q=${event.venue.lat},${event.venue.lng}`}
   438→                          target="_blank"
   439→                          rel="noopener noreferrer"
   440→                          className="text-xs text-emerald-600 hover:underline inline-flex items-center gap-1 mt-0.5"
   441→                        >
   442→                          <Globe className="h-3 w-3" /> View on Google Maps
   443→                        </a>
   444→                      )}
   445→                      {event.venue.website && (
   446→                        <a
   447→                          href={event.venue.website}
   448→                          target="_blank"
   449→                          rel="noopener noreferrer"
   450→                          className="text-xs text-emerald-600 hover:underline inline-flex items-center gap-1 mt-0.5 ml-2"
   451→                        >
   452→                          <ExternalLink className="h-3 w-3" /> Venue Website
   453→                        </a>
   454→                      )}
   455→                    </div>
   456→                  ) : (
   457→                    <div>
   458→                      <p className="font-medium">{event.venueName || 'Location TBD'}</p>
   459→                      {event.venueCity && (
   460→                        <p className="text-xs text-muted-foreground">{event.venueCity}{event.venueState ? `, ${event.venueState}` : ''}{event.venueCountry ? `, ${event.venueCountry}` : ''}</p>
   461→                      )}
   462→                    </div>
   463→                  )}
   464→                </div>
   465→              </div>
   466→            </Card>
   467→            <Card className="p-4">
   468→              <div className="flex items-center gap-3">
   469→                <div className="p-2 rounded-lg bg-purple-50 text-purple-600" aria-hidden="true">
   470→                  <Users className="h-5 w-5" />
   471→                </div>
   472→                <div>
   473→                  <p className="text-sm text-muted-foreground">Capacity</p>
   474→                  <p className="font-medium">{event.capacity || 'Unlimited'}{event.capacity ? ' attendees' : ''}</p>
   475→                  {event.ageRestriction && (
   476→                    <p className="text-xs text-muted-foreground">{event.ageRestriction}</p>
   477→                  )}
   478→                </div>
   479→              </div>
   480→            </Card>
   481→          </div>
   482→
   483→          {/* Tags */}
   484→          {event.tags?.length > 0 && (
   485→            <div className="flex flex-wrap gap-2" role="list" aria-label="Event tags">
   486→              {event.tags.map((tag, i) => (
   487→                <Badge key={tag.id || i} variant="outline" className="gap-1" role="listitem">
   488→                  <Tag className="h-3 w-3" /> {getTagName(tag)}
   489→                </Badge>
   490→              ))}
   491→            </div>
   492→          )}
   493→
   494→          {/* Description */}
   495→          <Card>
   496→            <CardHeader><CardTitle>About This Event</CardTitle></CardHeader>
   497→            <CardContent>
   498→              <div className="prose prose-sm max-w-none text-muted-foreground whitespace-pre-wrap">{event.description}</div>
   499→            </CardContent>
   500→          </Card>
   501→
   502→          {/* Program / Sessions */}
   503→          {sessions.length > 0 && (
   504→            <Card>
   505→              <CardHeader>
   506→                <CardTitle className="flex items-center gap-2">
   507→                  <CalendarDays className="h-5 w-5 text-emerald-600" aria-hidden="true" />
   508→                  Program
   509→                  <span className="text-sm font-normal text-muted-foreground">({sessions.length} sessions)</span>
   510→                </CardTitle>
   511→              </CardHeader>
   512→              <CardContent>
   513→                <div className="space-y-6">
   514→                  {Object.entries(sessionsByDate).map(([dateKey, dateSessions]) => (
   515→                    <div key={dateKey}>
   516→                      {dateKey !== 'no-date' && (
   517→                        <h3 className="text-sm font-semibold text-muted-foreground mb-3 flex items-center gap-2">
   518→                          <CalendarDays className="h-4 w-4" />

   955→
   956→function formatDistanceToNow(date: Date) {
   957→  const seconds = Math.floor((Date.now() - date.getTime()) / 1000);
   958→  if (seconds < 60) return 'just now';
   959→  const minutes = Math.floor(seconds / 60);
   960→  if (minutes < 60) return `${minutes}m ago`;
   961→  const hours = Math.floor(minutes / 60);
   962→  if (hours < 24) return `${hours}h ago`;
   963→  const days = Math.floor(hours / 24);
   964→  if (days < 30) return `${days}d ago`;
   965→  const months = Math.floor(days / 30);
   966→  return `${months}mo ago`;
   967→}
