   160→export function EventDetail() {
   161→  const { selectedEventId, goBack, navigate, user } = useAppStore();
   162→  const [event, setEvent] = useState<Event | null>(null);
   163→  const [loading, setLoading] = useState(true);
   164→  const [bookDialog, setBookDialog] = useState(false);
   165→  const [selectedTicket, setSelectedTicket] = useState<TicketType | null>(null);
   166→  const [quantity, setQuantity] = useState(1);
   167→  const [booking, setBooking] = useState(false);
   168→  const [relatedEvents, setRelatedEvents] = useState<any[]>([]);
   169→  const [reviewDialog, setReviewDialog] = useState(false);
   170→  const [reviewRating, setReviewRating] = useState(5);
   171→  const [reviewComment, setReviewComment] = useState('');
   172→  const [submittingReview, setSubmittingReview] = useState(false);
   173→  const [userHasReviewed, setUserHasReviewed] = useState(false);
   174→
   175→  // Phase 4F: Use single API response — event already includes sessions/participants/media
   176→  // No separate N+1 fetches needed
   177→
   178→  useEffect(() => {
   179→    if (!selectedEventId) return;
   180→    setLoading(true);
   181→    apiFetch<{ event: Event }>(`/api/events/${selectedEventId}`)
   182→      .then(data => {
   183→        setEvent(data.event);
   184→        // Check if current user has already reviewed
   185→        if (user && data.event.reviews) {
   186→          setUserHasReviewed(data.event.reviews.some((r: Review) => r.user?.id === user.id));
   187→        }
   188→        // Phase 4F: Set document title for SEO
   189→        if (data.event.title) {
   190→          document.title = `${data.event.title} — AppleCalendar`;
   191→        }
   192→      })
   193→      .catch(() => toast.error('Failed to load event'))
   194→      .finally(() => setLoading(false));
   195→  }, [selectedEventId, user?.id]);
   196→
   197→  // Cleanup: restore document title on unmount
   198→  useEffect(() => {
   199→    return () => { document.title = 'AppleCalendar — Discover Amazing Events'; };
   200→  }, []);
   201→
   202→  useEffect(() => {
   203→    if (!event?.category?.slug) return;
   204→    apiFetch<{ events: any[] }>(`/api/events?category=${event.category.slug}&limit=4&featured=false`)
   205→      .then(data => setRelatedEvents(data.events.filter((e: any) => e.id !== event.id).slice(0, 3)))
   206→      .catch(() => {});
   207→  }, [event?.category?.slug]);
   208→
   209→  // Derived data from single API response
   210→  const sessions = event?.sessions || [];
   211→  const participants = event?.participants || [];
   212→  const media = event?.media || [];
   213→
   214→  // Phase 4F: Bookability check
   215→  const canBook = event?.status === 'PUBLISHED' && event.isBookable !== false;
   216→  const isCancelled = event?.status === 'CANCELLED';
   217→  const isCompleted = event?.status === 'COMPLETED';
   218→
   219→  const handleBook = (ticket: TicketType) => {
   220→    if (!user) {
   221→      navigate('login');
   222→      toast.error('Please login to book tickets');
   223→      return;
   224→    }
   225→    setSelectedTicket(ticket);
   226→    setQuantity(1);
   227→    setBookDialog(true);
   228→  };
   229→
   230→  const confirmBooking = async () => {
   231→    if (!selectedTicket || !selectedEventId) return;
   232→    setBooking(true);
   233→    try {
   234→      await apiFetch('/api/events/' + selectedEventId + '/book', {
   235→        method: 'POST',
   236→        body: JSON.stringify({ ticketTypeId: selectedTicket.id, quantity }),
   237→      });
   238→      toast.success('Booking confirmed!');
   239→      setBookDialog(false);
   240→      const data = await apiFetch<{ event: Event }>(`/api/events/${selectedEventId}`);
   241→      setEvent(data.event);
   242→    } catch (err: any) {
   243→      toast.error(err.message || 'Booking failed');
   244→    } finally {
   245→      setBooking(false);
   246→    }
   247→  };
   248→
   249→  const avgRating = event?.reviews?.length
   250→    ? event.reviews.reduce((s, r) => s + r.rating, 0) / event.reviews.length
   251→    : 0;
   252→
   253→  const submitReview = async () => {
   254→    if (!selectedEventId) return;
   255→    setSubmittingReview(true);
   256→    try {
   257→      await apiFetch(`/api/events/${selectedEventId}/reviews`, {
   258→        method: 'POST',
   259→        body: JSON.stringify({ rating: reviewRating, comment: reviewComment || undefined }),
   260→      });
   261→      toast.success('Review submitted!');
   262→      setReviewDialog(false);
   263→      setReviewRating(5);
   264→      setReviewComment('');
   265→      setUserHasReviewed(true);
   266→      const data = await apiFetch<{ event: Event }>(`/api/events/${selectedEventId}`);
   267→      setEvent(data.event);
   268→    } catch (err: any) {
   269→      toast.error(err.message || 'Failed to submit review');
   270→    } finally {
   271→      setSubmittingReview(false);
   272→    }
   273→  };
   274→
   275→  // Session date grouping
   276→  const sessionsByDate = sessions.reduce<Record<string, EventSession[]>>((acc, session) => {
   277→    const dateKey = session.date ? format(new Date(session.date), 'yyyy-MM-dd') : 'no-date';
   278→    if (!acc[dateKey]) acc[dateKey] = [];
   279→    acc[dateKey].push(session);
   280→    return acc;
   281→  }, {});
   282→
   283→  // Media grouping by category
   284→  const posterMedia = media.filter(m => m.category === 'POSTER' || m.category === 'COVER');
   285→  const galleryMedia = media.filter(m => m.category === 'GALLERY');
   286→  const videoMedia = media.filter(m => m.category === 'PROMOTIONAL_VIDEO' || m.type === 'VIDEO');
   287→  const otherMedia = media.filter(m =>
   288→    !['POSTER', 'COVER', 'GALLERY', 'PROMOTIONAL_VIDEO'].includes(m.category) && m.type !== 'VIDEO'
   289→  );
   290→
   291→  if (loading) {
   292→    return (
   293→      <div className="space-y-6">
   294→        <Skeleton className="h-8 w-24" />
   295→        <Skeleton className="h-72 w-full rounded-xl" />
   296→        <div className="grid md:grid-cols-3 gap-6">
   297→          <div className="md:col-span-2 space-y-4">
   298→            <Skeleton className="h-8 w-3/4" />
   299→            <Skeleton className="h-4 w-1/2" />
   300→            <Skeleton className="h-40 w-full" />
   301→          </div>
   302→          <Skeleton className="h-80 rounded-xl" />
   303→        </div>
   304→      </div>
   305→    );
   306→  }
   307→
   308→  if (!event) {
   309→    return (
   310→      <div className="text-center py-20">
   311→        <CalendarDays className="h-12 w-12 text-muted-foreground mx-auto mb-4" />
   312→        <h2 className="text-xl font-semibold mb-2">Event not found</h2>
   313→        <p className="text-muted-foreground text-sm mb-4">This event may not exist or is not publicly available.</p>
   314→        <Button variant="outline" onClick={goBack}>Go Back</Button>
   315→      </div>
   316→    );
   317→  }
   318→
   319→  // Get tag name (handles both {tag: {name}} and {name} shapes)
   320→  const getTagName = (tag: any): string => tag.tag?.name || tag.name || '';
   321→
   322→  return (
   323→    <motion.div
   324→      initial={{ opacity: 0, y: 10 }}
   325→      animate={{ opacity: 1, y: 0 }}
   326→      transition={{ duration: 0.3 }}
   327→      className="space-y-6"
   328→    >
   329→      {/* Back button */}
   330→      <Button variant="ghost" size="sm" onClick={goBack} className="gap-2" aria-label="Go back to events">
   331→        <ArrowLeft className="h-4 w-4" /> Back
   332→      </Button>
   333→
   334→      {/* Phase 4F: Event Status Banner */}
   335→      {isCancelled && (
   336→        <div className="flex items-center gap-3 p-4 rounded-lg bg-red-50 dark:bg-red-950 border border-red-200 dark:border-red-800" role="alert">
   337→          <XCircle className="h-6 w-6 text-red-500 flex-shrink-0" />
   338→          <div>
   339→            <h3 className="font-semibold text-red-700 dark:text-red-300">This event has been cancelled</h3>
   340→            <p className="text-sm text-red-600 dark:text-red-400">Bookings are no longer available for this event.</p>
   341→          </div>
   342→        </div>
   343→      )}
   344→      {isCompleted && (
   345→        <div className="flex items-center gap-3 p-4 rounded-lg bg-blue-50 dark:bg-blue-950 border border-blue-200 dark:border-blue-800">
   346→          <CheckCircle2 className="h-6 w-6 text-blue-500 flex-shrink-0" />
   347→          <div>
   348→            <h3 className="font-semibold text-blue-700 dark:text-blue-300">This event has taken place</h3>
   349→            <p className="text-sm text-blue-600 dark:text-blue-400">You can still view event details and leave a review.</p>
   350→          </div>
   351→        </div>
   352→      )}
   353→
   354→      {/* Hero Image */}
   355→      <div className="relative h-64 md:h-80 lg:h-96 rounded-xl overflow-hidden">
   356→        {event.coverImage ? (
   357→          <img src={event.coverImage} alt={`${event.title} event cover`} className="w-full h-full object-cover" />
   358→        ) : (
   359→          <div className="w-full h-full bg-gradient-to-br from-emerald-200 to-teal-200 flex items-center justify-center">
   360→            <CalendarDays className="h-16 w-16 text-emerald-300" />
   361→          </div>
   362→        )}
   363→        <div className="absolute inset-0 bg-gradient-to-t from-black/60 to-transparent" />
   364→        <div className="absolute bottom-4 left-6 right-6 text-white">
   365→          <div className="flex items-center gap-2 mb-2 flex-wrap">
   366→            <Badge className="bg-emerald-600">{event.category?.name}</Badge>
   367→            {event.isFeatured && <Badge className="bg-amber-500">Featured</Badge>}
   368→            {event.isVirtual && <Badge className="bg-cyan-600">Virtual</Badge>}
   369→          </div>
   370→          <h1 className="text-2xl md:text-3xl lg:text-4xl font-bold">{event.title}</h1>
   371→          {event.shortDescription && (
   372→            <p className="text-white/80 text-sm md:text-base mt-1 line-clamp-2">{event.shortDescription}</p>
   373→          )}
   374→        </div>
   375→      </div>
   376→
   377→      <div className="grid md:grid-cols-3 gap-6">
   378→        {/* Main Content */}
   379→        <div className="md:col-span-2 space-y-6">
   380→          {/* Date / Location / Type Info Cards */}
   381→          <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   382→            <Card className="p-4">
   383→              <div className="flex items-center gap-3">
   384→                <div className="p-2 rounded-lg bg-emerald-50 text-emerald-600" aria-hidden="true">
   385→                  <CalendarDays className="h-5 w-5" />
   386→                </div>
   387→                <div>
   388→                  <p className="text-sm text-muted-foreground">Date</p>
   389→                  <p className="font-medium">{format(new Date(event.startDate), 'EEEE, MMM d, yyyy')}</p>
   390→                  {event.endDate && event.endDate !== event.startDate && (
   391→                    <p className="text-sm text-muted-foreground">to {format(new Date(event.endDate), 'EEE, MMM d, yyyy')}</p>
   392→                  )}
   393→                </div>
   394→              </div>
   395→            </Card>
   396→            <Card className="p-4">
   397→              <div className="flex items-center gap-3">
   398→                <div className="p-2 rounded-lg bg-teal-50 text-teal-600" aria-hidden="true">
   399→                  <Clock className="h-5 w-5" />
   400→                </div>
   401→                <div>
   402→                  <p className="text-sm text-muted-foreground">Time</p>
   403→                  <p className="font-medium">{event.startTime || 'TBD'} {event.endTime ? `- ${event.endTime}` : ''}</p>
   404→                  {event.timezone && event.timezone !== 'UTC' && (
   405→                    <p className="text-xs text-muted-foreground">{event.timezone}</p>
   406→                  )}
   407→                </div>
   408→              </div>
   409→            </Card>
   410→            <Card className="p-4">
   411→              <div className="flex items-center gap-3">
   412→                <div className={`p-2 rounded-lg ${event.isVirtual ? 'bg-cyan-50 text-cyan-600' : 'bg-amber-50 text-amber-600'}`} aria-hidden="true">
   413→                  {event.isVirtual ? <Globe className="h-5 w-5" /> : <MapPin className="h-5 w-5" />}
   414→                </div>
   415→                <div className="min-w-0">
   416→                  <p className="text-sm text-muted-foreground">{event.isVirtual ? 'Virtual Event' : 'Location'}</p>
   417→                  {event.isVirtual ? (
   418→                    event.virtualUrl ? (
   419→                      <a
   420→                        href={event.virtualUrl}
   421→                        target="_blank"
   422→                        rel="noopener noreferrer"
   423→                        className="font-medium text-cyan-600 hover:underline inline-flex items-center gap-1"
   424→                      >
   425→                        Join Online <ExternalLink className="h-3 w-3" />
   426→                      </a>
   427→                    ) : (
   428→                      <p className="font-medium text-cyan-600">Online</p>
   429→                    )
   430→                  ) : event.venue ? (
   431→                    <div>
   432→                      <p className="font-medium">{event.venue.name}</p>
   433→                      <p className="text-xs text-muted-foreground truncate">
   434→                        {event.venue.address}{event.venue.city ? `, ${event.venue.city}` : ''}{event.venue.state ? `, ${event.venue.state}` : ''}
   435→                      </p>
   436→                      {(event.venue.googleMapsUrl || (event.venue.lat != null && event.venue.lng != null)) && (
   437→                        <a
   438→                          href={event.venue.googleMapsUrl || `https://www.google.com/maps?q=${event.venue.lat},${event.venue.lng}`}
   439→                          target="_blank"
   440→                          rel="noopener noreferrer"
   441→                          className="text-xs text-emerald-600 hover:underline inline-flex items-center gap-1 mt-0.5"
   442→                        >
   443→                          <Globe className="h-3 w-3" /> View on Google Maps
   444→                        </a>
   445→                      )}
   446→                      {event.venue.website && (
   447→                        <a
   448→                          href={event.venue.website}
   449→                          target="_blank"
   450→                          rel="noopener noreferrer"
   451→                          className="text-xs text-emerald-600 hover:underline inline-flex items-center gap-1 mt-0.5 ml-2"
   452→                        >
   453→                          <ExternalLink className="h-3 w-3" /> Venue Website
   454→                        </a>
   455→                      )}
   456→                    </div>
   457→                  ) : (
   458→                    <div>
   459→                      <p className="font-medium">{event.venueName || 'Location TBD'}</p>
   460→                      {event.venueCity && (
   461→                        <p className="text-xs text-muted-foreground">{event.venueCity}{event.venueState ? `, ${event.venueState}` : ''}{event.venueCountry ? `, ${event.venueCountry}` : ''}</p>
   462→                      )}
   463→                    </div>
   464→                  )}
   465→                </div>
   466→              </div>
   467→            </Card>
   468→            <Card className="p-4">
   469→              <div className="flex items-center gap-3">
   470→                <div className="p-2 rounded-lg bg-purple-50 text-purple-600" aria-hidden="true">
   471→                  <Users className="h-5 w-5" />
   472→                </div>
   473→                <div>
   474→                  <p className="text-sm text-muted-foreground">Capacity</p>
   475→                  <p className="font-medium">{event.capacity || 'Unlimited'}{event.capacity ? ' attendees' : ''}</p>
   476→                  {event.ageRestriction && (
   477→                    <p className="text-xs text-muted-foreground">{event.ageRestriction}</p>
   478→                  )}
   479→                </div>
   480→              </div>
   481→            </Card>
   482→          </div>
   483→
   484→          {/* Tags */}
   485→          {event.tags?.length > 0 && (
   486→            <div className="flex flex-wrap gap-2" role="list" aria-label="Event tags">
   487→              {event.tags.map((tag, i) => (
   488→                <Badge key={tag.id || i} variant="outline" className="gap-1" role="listitem">
   489→                  <Tag className="h-3 w-3" /> {getTagName(tag)}
   490→                </Badge>
   491→              ))}
   492→            </div>
   493→          )}
   494→
   495→          {/* Description */}
   496→          <Card>
   497→            <CardHeader><CardTitle>About This Event</CardTitle></CardHeader>
   498→            <CardContent>
   499→              <div className="prose prose-sm max-w-none text-muted-foreground whitespace-pre-wrap">{event.description}</div>
   500→            </CardContent>
   501→          </Card>
   502→
   503→          {/* Program / Sessions */}
   504→          {sessions.length > 0 && (
   505→            <Card>
   506→              <CardHeader>
   507→                <CardTitle className="flex items-center gap-2">
   508→                  <CalendarDays className="h-5 w-5 text-emerald-600" aria-hidden="true" />
   509→                  Program
   510→                  <span className="text-sm font-normal text-muted-foreground">({sessions.length} sessions)</span>
   511→                </CardTitle>
   512→              </CardHeader>
   513→              <CardContent>
   514→                <div className="space-y-6">
   515→                  {Object.entries(sessionsByDate).map(([dateKey, dateSessions]) => (
   516→                    <div key={dateKey}>
   517→                      {dateKey !== 'no-date' && (

   956→
   957→function formatDistanceToNow(date: Date) {
   958→  const seconds = Math.floor((Date.now() - date.getTime()) / 1000);
   959→  if (seconds < 60) return 'just now';
   960→  const minutes = Math.floor(seconds / 60);
   961→  if (minutes < 60) return `${minutes}m ago`;
   962→  const hours = Math.floor(minutes / 60);
   963→  if (hours < 24) return `${hours}h ago`;
   964→  const days = Math.floor(hours / 24);
   965→  if (days < 30) return `${days}d ago`;
   966→  const months = Math.floor(days / 30);
   967→  return `${months}mo ago`;
   968→}
