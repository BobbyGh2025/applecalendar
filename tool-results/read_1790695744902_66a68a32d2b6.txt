     1→'use client';
     2→
     3→import { useState, useEffect } from 'react';
     4→import { Card, CardContent, CardHeader, CardTitle, CardDescription } from '@/components/ui/card';
     5→import { Button } from '@/components/ui/button';
     6→import { Input } from '@/components/ui/input';
     7→import { Label } from '@/components/ui/label';
     8→import { Textarea } from '@/components/ui/textarea';
     9→import { Switch } from '@/components/ui/switch';
    10→import { Badge } from '@/components/ui/badge';
    11→import { Separator } from '@/components/ui/separator';
    12→import {
    13→  Select,
    14→  SelectContent,
    15→  SelectItem,
    16→  SelectTrigger,
    17→  SelectValue,
    18→} from '@/components/ui/select';
    19→import {
    20→  Dialog,
    21→  DialogContent,
    22→  DialogHeader,
    23→  DialogTitle,
    24→  DialogDescription,
    25→  DialogFooter,
    26→} from '@/components/ui/dialog';
    27→import { useAppStore } from '@/stores/app-store';
    28→import { apiFetch } from '@/lib/api';
    29→import { toast } from 'sonner';
    30→import {
    31→  Loader2,
    32→  Plus,
    33→  Trash2,
    34→  Globe,
    35→  MapPin,
    36→  DollarSign,
    37→  Info,
    38→  Clock,
    39→  Send,
    40→  XCircle,
    41→  Save,
    42→  AlertTriangle,
    43→  Calendar,
    44→  Users,
    45→  Image as ImageIcon,
    46→  Building2,
    47→} from 'lucide-react';
    48→import { formatMoney, parseMoneyOrThrow, DEFAULT_CURRENCY } from '@/lib/money';
    49→
    50→interface Category {
    51→  id: string;
    52→  name: string;
    53→  slug: string;
    54→}
    55→
    56→interface VenueOption {
    57→  id: string;
    58→  name: string;
    59→  address: string;
    60→  city: string;
    61→  state: string | null;
    62→  country: string;
    63→  lat: number | null;
    64→  lng: number | null;
    65→}
    66→
    67→interface TicketTypeForm {
    68→  id?: string;
    69→  name: string;
    70→  price: string;
    71→  quantity: string;
    72→  isNew?: boolean;
    73→  toDelete?: boolean;
    74→}
    75→
    76→const STATUS_COLORS: Record<string, string> = {
    77→  PUBLISHED: 'bg-green-100 text-green-700',
    78→  DRAFT: 'bg-gray-100 text-gray-700',
    79→  PENDING: 'bg-yellow-100 text-yellow-700',
    80→  CANCELLED: 'bg-red-100 text-red-700',
    81→  COMPLETED: 'bg-blue-100 text-blue-700',
    82→  REJECTED: 'bg-red-100 text-red-700',
    83→};
    84→
    85→export function OrganizerEditEvent() {
    86→  const { selectedEventId, navigate, user } = useAppStore();
    87→  const [loading, setLoading] = useState(true);
    88→  const [saving, setSaving] = useState(false);
    89→  const [categories, setCategories] = useState<Category[]>([]);
    90→  const [venues, setVenues] = useState<VenueOption[]>([]);
    91→  const [eventStatus, setEventStatus] = useState<string>('DRAFT');
    92→  const [statusDialog, setStatusDialog] = useState<{ open: boolean; target: string }>({
    93→    open: false,
    94→    target: '',
    95→  });
    96→
    97→  const [form, setForm] = useState({
    98→    title: '',
    99→    description: '',
   100→    shortDescription: '',
   101→    categoryId: '',
   102→    coverImage: '',
   103→    startDate: '',
   104→    endDate: '',
   105→    startTime: '',
   106→    endTime: '',
   107→    timezone: 'UTC',
   108→    venueId: '',
   109→    venueName: '',
   110→    venueAddress: '',
   111→    venueCity: '',
   112→    venueState: '',
   113→    venueCountry: '',
   114→    venueLat: '',
   115→    venueLng: '',
   116→    isVirtual: false,
   117→    virtualUrl: '',
   118→    isPaid: false,
   119→    currency: 'USD',
   120→    capacity: '',
   121→    tags: '',
   122→  });
   123→
   124→  const [ticketTypes, setTicketTypes] = useState<TicketTypeForm[]>([]);
   125→
   126→  // Load event data and categories
   127→  useEffect(() => {
   128→    if (!selectedEventId) return;
   129→    setLoading(true);
   130→    Promise.all([
   131→      apiFetch<{
   132→        event: {
   133→          title: string;
   134→          description: string;
   135→          shortDescription: string | null;
   136→          coverImage: string | null;
   137→          startDate: string;
   138→          endDate: string | null;
   139→          startTime: string | null;
   140→          endTime: string | null;
   141→          timezone: string;
   142→          venueId: string | null;
   143→          venueName: string | null;
   144→          venueAddress: string | null;
   145→          venueCity: string | null;
   146→          venueState: string | null;
   147→          venueCountry: string | null;
   148→          venueLat: number | null;
   149→          venueLng: number | null;
   150→          isVirtual: boolean;
   151→          virtualUrl: string | null;
   152→          isPaid: boolean;
   153→          currency: string;
   154→          capacity: number | null;
   155→          status: string;
   156→          categoryId: string | null;
   157→          category: { id: string; name: string; slug: string } | null;
   158→          tags: { tag: { name: string } }[];
   159→          ticketTypes: {
   160→            id: string;
   161→            name: string;
   162→            price: number;
   163→            quantity: number;
   164→            soldCount: number;
   165→          }[];
   166→        };
   167→      }>(`/api/events/${selectedEventId}`),
   168→      apiFetch<{ categories: Category[] }>('/api/categories'),
   169→      apiFetch<{ venues: VenueOption[] }>('/api/organizer/venues').catch(() => ({ venues: [] })),
   170→    ])
   171→      .then(([eventData, catData, venueData]) => {
   172→        const e = eventData.event;
   173→        if (venueData?.venues) setVenues(venueData.venues);
   174→        setForm({
   175→          title: e.title,
   176→          description: e.description,
   177→          shortDescription: e.shortDescription || '',
   178→          categoryId: e.category?.id || '',
   179→          coverImage: e.coverImage || '',
   180→          startDate: e.startDate ? new Date(e.startDate).toISOString().split('T')[0] : '',
   181→          endDate: e.endDate ? new Date(e.endDate).toISOString().split('T')[0] : '',
   182→          startTime: e.startTime || '',
   183→          endTime: e.endTime || '',
   184→          timezone: e.timezone || 'UTC',
   185→          venueId: e.venueId || '',
   186→          venueName: e.venueName || '',
   187→          venueAddress: e.venueAddress || '',
   188→          venueCity: e.venueCity || '',
   189→          venueState: e.venueState || '',
   190→          venueCountry: e.venueCountry || '',
   191→          venueLat: e.venueLat != null ? String(e.venueLat) : '',
   192→          venueLng: e.venueLng != null ? String(e.venueLng) : '',
   193→          isVirtual: e.isVirtual,
   194→          virtualUrl: e.virtualUrl || '',
   195→          isPaid: e.isPaid,
   196→          currency: e.currency || 'USD',
   197→          capacity: e.capacity ? String(e.capacity) : '',
   198→          tags: e.tags?.map((t: { tag: { name: string } }) => t.tag.name).join(', ') || '',
   199→        });
   200→        setEventStatus(e.status);
   201→        setTicketTypes(
   202→          e.ticketTypes.map((tt) => ({
   203→            id: tt.id,
   204→            name: tt.name,
   205→            price: formatMoney(tt.price, e.currency || DEFAULT_CURRENCY),
   206→            quantity: String(tt.quantity),
   207→          }))
   208→        );
   209→        setCategories(catData.categories || []);
   210→      })
   211→      .catch(() => toast.error('Failed to load event'))
   212→      .finally(() => setLoading(false));
   213→  }, [selectedEventId]);
   214→
   215→  const updateForm = (key: string, value: string | boolean) =>
   216→    setForm((prev) => ({ ...prev, [key]: value }));
   217→
   218→  const addTicketType = () =>
   219→    setTicketTypes((prev) => [
   220→      ...prev,
   221→      { name: '', price: '0', quantity: '50', isNew: true },
   222→    ]);
   223→  const removeTicketType = (i: number) => {
   224→    const tt = ticketTypes[i];
   225→    if (tt.isNew) {
   226→      setTicketTypes((prev) => prev.filter((_, idx) => idx !== i));
   227→    } else {
   228→      // Mark existing for deletion
   229→      setTicketTypes((prev) =>
   230→        prev.map((t, idx) => (idx === i ? { ...t, toDelete: true } : t))
   231→      );
   232→    }
   233→  };
   234→  const restoreTicketType = (i: number) => {
   235→    setTicketTypes((prev) =>
   236→      prev.map((t, idx) => (idx === i ? { ...t, toDelete: false } : t))
   237→    );
   238→  };
   239→  const updateTicketType = (i: number, key: keyof TicketTypeForm, value: string) => {
   240→    setTicketTypes((prev) =>
   241→      prev.map((t, idx) => (idx === i ? { ...t, [key]: value } : t))
   242→    );
   243→  };
   244→
   245→  const handleSubmit = async (e: React.FormEvent) => {
   246→    e.preventDefault();
   247→    if (!form.title || !form.startDate) {
   248→      toast.error('Please fill in required fields (title, date)');
   249→      return;
   250→    }
   251→    if (form.isVirtual && !form.virtualUrl) {
   252→      toast.error('Please provide a virtual event URL');
   253→      return;
   254→    }
   255→
   256→    setSaving(true);
   257→    try {
   258→      // Build PATCH body (only send changed fields)
   259→      const body: Record<string, unknown> = {
   260→        title: form.title,
   261→        description: form.description,
   262→        shortDescription: form.shortDescription || undefined,
   263→        coverImage: form.coverImage || undefined,
   264→        startDate: form.startDate,
   265→        endDate: form.endDate || undefined,
   266→        startTime: form.startTime || undefined,
   267→        endTime: form.endTime || undefined,
   268→        timezone: form.timezone,
   269→        venueId: form.venueId || undefined,
   270→        venueName: form.isVirtual ? undefined : form.venueName,
   271→        venueAddress: form.isVirtual ? undefined : form.venueAddress,
   272→        venueCity: form.isVirtual ? undefined : form.venueCity,
   273→        venueState: form.isVirtual ? undefined : form.venueState,
   274→        venueCountry: form.isVirtual ? undefined : form.venueCountry,
   275→        venueLat: form.isVirtual ? undefined : (form.venueLat ? Number(form.venueLat) : undefined),
   276→        venueLng: form.isVirtual ? undefined : (form.venueLng ? Number(form.venueLng) : undefined),
   277→        isVirtual: form.isVirtual,
   278→        virtualUrl: form.isVirtual ? form.virtualUrl : undefined,
   279→        isPaid: form.isPaid,
   280→        currency: form.currency,
   281→        capacity: form.capacity ? parseInt(form.capacity) : undefined,
   282→        categoryId: form.categoryId || undefined,
   283→        tags: form.tags
   284→          ? form.tags.split(',').map((t: string) => t.trim()).filter(Boolean)
   285→          : [],
   286→      };
   287→
   288→      await apiFetch(`/api/events/${selectedEventId}`, {
   289→        method: 'PATCH',
   290→        body: JSON.stringify(body),
   291→      });
   292→
   293→      // Handle ticket type changes
   294→      // Delete marked ticket types
   295→      for (const tt of ticketTypes.filter((t) => t.toDelete && t.id)) {
   296→        await apiFetch(
   297→          `/api/events/${selectedEventId}/ticket-types/${tt.id}`,
   298→          { method: 'DELETE' }
   299→        );
   300→      }
   301→
   302→      // Add new ticket types
   303→      for (const tt of ticketTypes.filter((t) => t.isNew)) {
   304→        await apiFetch(`/api/events/${selectedEventId}/ticket-types`, {
   305→          method: 'POST',
   306→          body: JSON.stringify({
   307→            name: tt.name,
   308→            price: parseMoneyOrThrow(tt.price, form.currency || DEFAULT_CURRENCY),
   309→            quantity: parseInt(tt.quantity) || 50,
   310→            currency: form.currency,
   311→          }),
   312→        });
   313→      }
   314→
   315→      // Update existing ticket types (changed values)
   316→      for (const tt of ticketTypes.filter((t) => t.id && !t.isNew && !t.toDelete)) {
   317→        await apiFetch(
   318→          `/api/events/${selectedEventId}/ticket-types/${tt.id}`,
   319→          {
   320→            method: 'PATCH',
   321→            body: JSON.stringify({
   322→              name: tt.name,
   323→              price: parseMoneyOrThrow(tt.price, form.currency || DEFAULT_CURRENCY),
   324→              quantity: parseInt(tt.quantity),
   325→            }),
   326→          }
   327→        );
   328→      }
   329→
   330→      toast.success('Event updated successfully!');
   331→      navigate('organizer-events');
   332→    } catch (err: unknown) {
   333→      const message = err instanceof Error ? err.message : 'Failed to update event';
   334→      toast.error(message);
   335→    } finally {
   336→      setSaving(false);
   337→    }
   338→  };
   339→
   340→  // Status transition handlers
   341→  const handleStatusChange = async (targetStatus: string) => {
   342→    setStatusDialog({ open: false, target: '' });
   343→    setSaving(true);
   344→    try {
   345→      await apiFetch(`/api/events/${selectedEventId}`, {
   346→        method: 'PATCH',
   347→        body: JSON.stringify({ status: targetStatus }),
   348→      });
   349→      setEventStatus(targetStatus);
   350→      toast.success(`Event status changed to ${targetStatus}`);
   351→    } catch (err: unknown) {
   352→      const message = err instanceof Error ? err.message : 'Failed to change status';
   353→      toast.error(message);
   354→    } finally {
   355→      setSaving(false);
   356→    }
   357→  };
   358→
   359→  if (loading) {
   360→    return (
   361→      <div className="space-y-6 max-w-3xl">
   362→        <div className="h-8 w-48 bg-muted animate-pulse rounded" />
   363→        <div className="h-64 bg-muted animate-pulse rounded-xl" />
   364→        <div className="h-48 bg-muted animate-pulse rounded-xl" />
   365→      </div>
   366→    );
   367→  }
   368→
   369→  // Determine available status transitions based on current status
   370→  const getStatusActions = () => {
   371→    if (!user) return [];
   372→    const actions: { target: string; label: string; icon: React.ReactNode; variant: string }[] = [];
   373→
   374→    if (user.role === 'ORGANIZER' || user.role === 'SUPER_ADMIN') {
   375→      if (eventStatus === 'DRAFT') {
   376→        actions.push({
   377→          target: 'PENDING',
   378→          label: 'Submit for Review',
   379→          icon: <Send className="h-4 w-4" />,
   380→          variant: 'default',
   381→        });
   382→      }
   383→      if (eventStatus === 'PENDING') {
   384→        actions.push({
   385→          target: 'DRAFT',
   386→          label: 'Withdraw from Review',
   387→          icon: <XCircle className="h-4 w-4" />,
   388→          variant: 'outline',
   389→        });
   390→      }
   391→      if (eventStatus === 'PUBLISHED') {
   392→        actions.push({
   393→          target: 'CANCELLED',
   394→          label: 'Cancel Event',
   395→          icon: <XCircle className="h-4 w-4" />,
   396→          variant: 'destructive',
   397→        });
   398→      }
   399→    }
   400→
   401→    // SUPER_ADMIN can additionally publish and reject directly
   402→    if (user.role === 'SUPER_ADMIN') {
   403→      if (eventStatus === 'PENDING') {
   404→        actions.push({
   405→          target: 'PUBLISHED',
   406→          label: 'Approve & Publish',
   407→          icon: <Send className="h-4 w-4" />,
   408→          variant: 'default',
   409→        });
   410→        actions.push({
   411→          target: 'REJECTED',
   412→          label: 'Reject Event',
   413→          icon: <XCircle className="h-4 w-4" />,
   414→          variant: 'destructive',
   415→        });
   416→      }
   417→      if (eventStatus === 'DRAFT') {
   418→        actions.push({
   419→          target: 'PUBLISHED',
   420→          label: 'Publish Directly',
   421→          icon: <Send className="h-4 w-4" />,
   422→          variant: 'default',
   423→        });
   424→      }
   425→    }
   426→
   427→    return actions;
   428→  };
   429→
   430→  return (
   431→    <form onSubmit={handleSubmit} className="space-y-6 max-w-3xl">
   432→      <div className="flex items-center justify-between">
   433→        <h1 className="text-2xl font-bold">Edit Event</h1>
   434→        <Badge className={STATUS_COLORS[eventStatus] || ''}>{eventStatus}</Badge>
   435→      </div>
   436→
   437→      {/* Status Workflow Actions */}
   438→      {getStatusActions().length > 0 && (
   439→        <Card>
   440→          <CardHeader>
   441→            <CardTitle className="text-base">Event Status</CardTitle>
   442→            <CardDescription>Change the event status to control visibility</CardDescription>
   443→          </CardHeader>
   444→          <CardContent>
   445→            <div className="flex flex-wrap gap-2">
   446→              {getStatusActions().map((action) => (
   447→                <Button
   448→                  key={action.target}
   449→                  type="button"
   450→                  variant={action.variant === 'destructive' ? 'destructive' : action.variant === 'default' ? 'default' : 'outline'}
   451→                  className={action.variant === 'default' ? 'bg-emerald-600 hover:bg-emerald-700' : ''}
   452→                  onClick={() =>
   453→                    setStatusDialog({ open: true, target: action.target })
   454→                  }
   455→                  disabled={saving}
   456→                >
   457→                  {action.icon}
   458→                  <span className="ml-2">{action.label}</span>
   459→                </Button>
   460→              ))}
   461→            </div>
   462→          </CardContent>
   463→        </Card>
   464→      )}
   465→
   466→      {/* Basic Info */}
   467→      <Card>
   468→        <CardHeader>
   469→          <CardTitle className="flex items-center gap-2">
   470→            <Info className="h-5 w-5 text-emerald-500" /> Basic Information
   471→          </CardTitle>
   472→        </CardHeader>
   473→        <CardContent className="space-y-4">
   474→          <div className="space-y-2">
   475→            <Label htmlFor="title">Title *</Label>
   476→            <Input
   477→              id="title"
   478→              placeholder="Event title"
   479→              value={form.title}
   480→              onChange={(e) => updateForm('title', e.target.value)}
   481→              required
   482→            />
   483→          </div>
   484→          <div className="space-y-2">
   485→            <Label htmlFor="shortDesc">Short Description</Label>
   486→            <Input
   487→              id="shortDesc"
   488→              placeholder="Brief summary (1-2 sentences)"
   489→              value={form.shortDescription}
   490→              onChange={(e) => updateForm('shortDescription', e.target.value)}
   491→            />
   492→          </div>
   493→          <div className="space-y-2">
   494→            <Label htmlFor="desc">Full Description</Label>
   495→            <Textarea
   496→              id="desc"
   497→              placeholder="Detailed event description..."
   498→              rows={5}
   499→              value={form.description}
   500→              onChange={(e) => updateForm('description', e.target.value)}
   501→            />
   502→          </div>
   503→          <div className="grid sm:grid-cols-2 gap-4">
   504→            <div className="space-y-2">
   505→              <Label>Category</Label>
   506→              <Select
   507→                value={form.categoryId}
   508→                onValueChange={(v) => updateForm('categoryId', v)}
   509→              >
   510→                <SelectTrigger>
   511→                  <SelectValue placeholder="Select category" />
   512→                </SelectTrigger>
   513→                <SelectContent>
   514→                  {categories.map((c) => (
   515→                    <SelectItem key={c.id} value={c.id}>
   516→                      {c.name}
   517→                    </SelectItem>
   518→                  ))}
   519→                </SelectContent>
   520→              </Select>
   521→            </div>
   522→            <div className="space-y-2">
   523→              <Label htmlFor="coverImg">Cover Image URL</Label>
   524→              <Input
   525→                id="coverImg"
   526→                placeholder="https://..."
   527→                value={form.coverImage}
   528→                onChange={(e) => updateForm('coverImage', e.target.value)}
   529→              />
   530→            </div>
   531→          </div>
   532→          <div className="space-y-2">
   533→            <Label htmlFor="tags">Tags (comma-separated)</Label>
   534→            <Input
   535→              id="tags"
   536→              placeholder="music, tech, networking"
   537→              value={form.tags}
   538→              onChange={(e) => updateForm('tags', e.target.value)}
   539→            />
   540→          </div>
   541→        </CardContent>
   542→      </Card>
   543→
   544→      {/* Date & Time */}
   545→      <Card>
   546→        <CardHeader>
   547→          <CardTitle className="flex items-center gap-2">
   548→            <Clock className="h-5 w-5 text-teal-500" /> Date & Time
   549→          </CardTitle>
   550→        </CardHeader>
   551→        <CardContent className="space-y-4">
   552→          <div className="grid sm:grid-cols-2 gap-4">
   553→            <div className="space-y-2">
   554→              <Label htmlFor="startDate">Start Date *</Label>
   555→              <Input
   556→                id="startDate"
   557→                type="date"
   558→                value={form.startDate}
   559→                onChange={(e) => updateForm('startDate', e.target.value)}
   560→                required
   561→              />
   562→            </div>
   563→            <div className="space-y-2">
   564→              <Label htmlFor="endDate">End Date</Label>
   565→              <Input
   566→                id="endDate"
   567→                type="date"
   568→                value={form.endDate}
   569→                onChange={(e) => updateForm('endDate', e.target.value)}
   570→              />
   571→            </div>
   572→          </div>
   573→          <div className="grid sm:grid-cols-2 gap-4">
   574→            <div className="space-y-2">
   575→              <Label htmlFor="startTime">Start Time</Label>
   576→              <Input
   577→                id="startTime"
   578→                type="time"
   579→                value={form.startTime}
   580→                onChange={(e) => updateForm('startTime', e.target.value)}
   581→              />
   582→            </div>
   583→            <div className="space-y-2">
   584→              <Label htmlFor="endTime">End Time</Label>
   585→              <Input
   586→                id="endTime"
   587→                type="time"
   588→                value={form.endTime}
   589→                onChange={(e) => updateForm('endTime', e.target.value)}
   590→              />
   591→            </div>
   592→          </div>
   593→          <div className="space-y-2">
   594→            <Label>Timezone</Label>
   595→            <Select
   596→              value={form.timezone}
   597→              onValueChange={(v) => updateForm('timezone', v)}
   598→            >
   599→              <SelectTrigger>
   600→                <SelectValue />
   601→              </SelectTrigger>
   602→              <SelectContent>
   603→                <SelectItem value="UTC">UTC</SelectItem>
   604→                <SelectItem value="America/New_York">Eastern (ET)</SelectItem>
   605→                <SelectItem value="America/Chicago">Central (CT)</SelectItem>
   606→                <SelectItem value="America/Denver">Mountain (MT)</SelectItem>

   693→                <div className="space-y-2">
   694→                  <Label htmlFor="venueAddr">Address</Label>
   695→                  <Input
   696→                    id="venueAddr"
   697→                    placeholder="123 Main St"
   698→                    value={form.venueAddress}
   699→                    onChange={(e) => updateForm('venueAddress', e.target.value)}
   700→                    disabled={!!form.venueId}
   701→                  />
   702→                </div>
   703→                <div className="space-y-2">

   705→                  <Input
   706→                    id="venueCity"
   707→                    placeholder="New York"
   708→                    value={form.venueCity}
   709→                    onChange={(e) => updateForm('venueCity', e.target.value)}
   710→                    disabled={!!form.venueId}
   711→                  />

   716→                    id="venueState"

   719→                    onChange={(e) => updateForm('venueState', e.target.value)}

   749→              checked={form.isPaid}
   750→              onCheckedChange={(v) => updateForm('isPaid', v)}

   870→          <div className="grid sm:grid-cols-3 gap-3">
   871→            <Button
   872→              type="button"
   873→              variant="outline"
   874→              className="h-auto py-4 flex-col gap-2"
   875→              onClick={() => navigate('organizer-event-content')}
   876→            >
   877→              <Calendar className="h-6 w-6 text-emerald-600" />
   878→              <span className="font-medium">Program</span>

   880→            </Button>
   881→            <Button
   882→              type="button"
   883→              variant="outline"
   884→              className="h-auto py-4 flex-col gap-2"
   885→              onClick={() => navigate('organizer-event-content')}
   886→            >
   887→              <Users className="h-6 w-6 text-emerald-600" />

   892→              type="button"
   893→              variant="outline"

   896→            >

   904→
   905→      <div className="flex gap-4">
   906→        <Button
   907→          type="button"
   908→          variant="outline"
   909→          onClick={() => navigate('organizer-events')}
   910→        >
   911→          Cancel
   912→        </Button>
   913→        <Button
   914→          type="submit"
   915→          className="bg-emerald-600 hover:bg-emerald-700"

   932→        <DialogContent>
