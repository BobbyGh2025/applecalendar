    89→    it('PROCESSING → COMPLETED', () => {
    90→      const result = validatePaymentTransition('PROCESSING', 'COMPLETED');
    91→      expect(result.to).toBe('COMPLETED');

   101→    it('PROCESSING → EXPIRED', () => {
   102→      const result = validatePaymentTransition('PROCESSING', 'EXPIRED');

   106→    it('COMPLETED → REFUNDED', () => {

   112→      const result = validatePaymentTransition('PENDING', 'PENDING');

   121→    });
   122→
   123→    it('FAILED → COMPLETED ❌', () => {
   124→      expect(() => validatePaymentTransition('FAILED', 'COMPLETED')).toThrow(InvalidPaymentTransition);

   126→

   129→    });

   131→    it('EXPIRED → COMPLETED ❌', () => {

   136→      expect(() => validatePaymentTransition('REFUNDED', 'COMPLETED')).toThrow(InvalidPaymentTransition);
   137→    });
   138→
   139→    it('REFUNDED → PENDING ❌', () => {
   140→      expect(() => validatePaymentTransition('REFUNDED', 'PENDING')).toThrow(InvalidPaymentTransition);

   308→describe('Inventory Reservation', () => {

   311→  let testOrgId: string;

   313→  beforeAll(async () => {

   317→    });
   318→    testOrgId = org.id;
   319→
   320→    const event = await prisma.event.create({
   321→      data: {
   322→        title: 'Inventory Test Event',
   323→        slug: `inv-test-${Date.now()}`,
   324→        description: 'Test',
   325→        startDate: new Date('2025-12-01'),
   326→        organizerId: org.id,
   327→        status: 'PUBLISHED',
   328→        isPaid: true,
   329→      },
   330→    });
   331→    testEventId = event.id;
   332→
   333→    const tt = await prisma.ticketType.create({
   334→      data: {
   335→        eventId: event.id,
   336→        name: 'General Admission',
   337→        price: 5000,
   338→        currency: 'GHS',
   339→        quantity: 10,
   340→        soldCount: 0,
   341→        reservedCount: 0,
   342→        minPerOrder: 1,
   343→        maxPerOrder: 5,
   344→        isActive: true,
   345→      },
   346→    });
   347→    testTicketTypeId = tt.id;
   348→  });
   349→
   350→  afterAll(async () => {
   351→    await prisma.ticketType.delete({ where: { id: testTicketTypeId } }).catch(() => {});
   352→    await prisma.event.delete({ where: { id: testEventId } }).catch(() => {});
   353→    await prisma.user.delete({ where: { id: testOrgId } }).catch(() => {});
   354→    await prisma.$disconnect();
   355→  });
   356→
   357→  it('checkInventory returns correct state', async () => {
   358→    const inv = await checkInventory(testTicketTypeId);
   359→    expect(inv.quantity).toBe(10);
   360→    expect(inv.soldCount).toBe(0);
   361→    expect(inv.reservedCount).toBe(0);
   362→    expect(inv.available).toBe(10);
   363→    expect(inv.canReserve).toBe(true);
   364→  });
   365→
   366→  it('reserveInventory increments reservedCount', async () => {
   367→    await reserveInventory({ ticketTypeId: testTicketTypeId, quantity: 3 });
   368→    const inv = await checkInventory(testTicketTypeId);
   369→    expect(inv.reservedCount).toBe(3);
   370→    expect(inv.available).toBe(7);
   371→  });
   372→

   374→    await expect(reserveInventory({ ticketTypeId: testTicketTypeId, quantity: 8 }))

   379→    await releaseReservation({ ticketTypeId: testTicketTypeId, quantity: 2 });
   380→    const inv = await checkInventory(testTicketTypeId);
   381→    expect(inv.reservedCount).toBe(1);
   382→  });

   389→    expect(inv.available).toBe(9);

   393→    const result = await verifyInventoryInvariant(testTicketTypeId);
   394→    expect(result).toBe(true);
   395→  });
   396→
   397→  it('concurrent reservations cannot oversell', async () => {
   398→    // Create a ticket type with only 2 tickets left
   399→    const tt = await prisma.ticketType.create({
   400→      data: {
   401→        eventId: testEventId,
   402→        name: 'Limited Tickets',
   403→        price: 1000,
   404→        currency: 'GHS',
   405→        quantity: 2,
   406→        soldCount: 0,
   407→        reservedCount: 0,
   408→        minPerOrder: 1,
   409→        maxPerOrder: 2,
   410→        isActive: true,
   411→      },
   412→    });
   413→
   414→    // Two concurrent reservations of 2 each — only one should succeed
   415→    const results = await Promise.allSettled([
   416→      reserveInventory({ ticketTypeId: tt.id, quantity: 2 }),
   417→      reserveInventory({ ticketTypeId: tt.id, quantity: 2 }),
   418→    ]);
   419→
   420→    const succeeded = results.filter(r => r.status === 'fulfilled').length;
   421→    const failed = results.filter(r => r.status === 'rejected').length;
   422→
   423→    // At least one should fail (both might fail if they truly race)
   424→    expect(succeeded + failed).toBe(2);
   425→
   426→    // Verify invariant
   427→    const inv = await checkInventory(tt.id);
   428→    expect(inv.soldCount + inv.reservedCount).toBeLessThanOrEqual(inv.quantity);
   429→
   430→    await prisma.ticketType.delete({ where: { id: tt.id } }).catch(() => {});
   431→  });
   432→});
