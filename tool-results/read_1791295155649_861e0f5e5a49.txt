   147→  return crypto.createHmac('sha512', secret).update(rawBody).digest('hex');

   283→describe('Stage 3: Webhook Signature Verification', () => {
   284→  it('valid signature is deterministic', () => {
   285→    const body = JSON.stringify({ event: 'charge.success', data: { id: 123, reference: 'ref-1', amount: 5000, currency: 'GHS' } });
   286→    const secret = 'test-webhook-secret';
   287→    const sig1 = computePaystackSignature(body, secret);
   288→    const sig2 = computePaystackSignature(body, secret);
   289→    expect(sig1).toBe(sig2);
   290→  });
   291→
   292→  it('wrong secret produces different signature', () => {
   293→    const body = JSON.stringify({ event: 'charge.success', data: { id: 123 } });
   294→    const sig1 = computePaystackSignature(body, 'correct-secret');
   295→    const sig2 = computePaystackSignature(body, 'wrong-secret');
   296→    expect(sig1).not.toBe(sig2);
   297→  });
   298→
   299→  it('tampered body produces different signature', () => {
   300→    const secret = 'test-webhook-secret';
   301→    const body1 = JSON.stringify({ event: 'charge.success', data: { amount: 5000 } });
   302→    const body2 = JSON.stringify({ event: 'charge.success', data: { amount: 9999 } });
   303→    expect(computePaystackSignature(body1, secret)).not.toBe(computePaystackSignature(body2, secret));
   304→  });
   305→
   306→  it('timing-safe comparison: different-length signatures handled safely', () => {
   307→    const buf1 = Buffer.from('abc123');
   308→    const buf2 = Buffer.from('abc12345');
   309→    // Direct timingSafeEqual with different lengths should throw
   310→    expect(() => crypto.timingSafeEqual(buf1, buf2)).toThrow();
   311→    // Route code should check length first (as it does)
   312→    expect(buf1.length !== buf2.length).toBe(true);
   313→  });
   314→
   315→  it('HMAC-SHA512 produces 128 hex characters', () => {
   316→    const body = 'test';
   317→    const secret = 'secret';
   318→    const sig = computePaystackSignature(body, secret);
   319→    expect(sig.length).toBe(128); // SHA-512 = 64 bytes = 128 hex chars
   320→  });
   321→});
