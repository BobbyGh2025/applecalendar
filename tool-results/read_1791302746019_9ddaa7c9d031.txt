   169→/**
   170→ * Create a full paid-booking lifecycle setup:
   171→ *   - PENDING booking
   172→ *   - Reserved inventory (reservedCount += quantity)
   173→ *   - PENDING payment
   174→ *   - PENDING tickets
   175→ *
   176→ * Returns all created entities for assertions.
   177→ */
   178→async function createPaidBookingWithReservation(
   179→  user: { id: string },
   180→  event: { id: string },
   181→  ticketType: { id: string },
   182→  quantity: number = 2,
   183→) {
   184→  const price = 5000; // GHS 50.00 in minor units
   185→  const totalAmount = price * quantity;
   186→  const bookingRef = `BK-${uid()}`;
   187→
   188→  // 1. Create PENDING booking
   189→  const booking = await db.booking.create({
   190→    data: {
   191→      userId: user.id,
   192→      eventId: event.id,
   193→      totalAmount,
   194→      currency: 'GHS',
   195→      status: 'PENDING',
   196→      bookingRef,
   197→    },
   198→  });
   199→  created.bookings.push(booking.id);
   200→
   201→  // 2. Reserve inventory
   202→  await reserveInventory({ ticketTypeId: ticketType.id, quantity });
   203→
   204→  // 3. Create PENDING payment with providerRef for webhook lookup
   205→  const providerRef = `ps_ref_${uid()}`;
   206→  const idempotencyKey = generateIdempotencyKey();
   207→  const payment = await db.payment.create({
   208→    data: {
   209→      bookingId: booking.id,
   210→      userId: user.id,
   211→      amount: totalAmount,
   212→      currency: 'GHS',
   213→      provider: 'PAYSTACK',
   214→      status: 'PENDING',
   215→      idempotencyKey,
   216→      providerRef,
   217→      expiresAt: new Date(Date.now() + 15 * 60 * 1000), // 15 minutes from now
   218→    },
   219→  });
   220→  created.payments.push(payment.id);
   221→
   222→  // 4. Create PENDING tickets
   223→  const ticketResult = await createPendingTickets({
   224→    bookingId: booking.id,
   225→    ticketTypeId: ticketType.id,
   226→    quantity,
   227→    bookingRef,
   228→  });
   229→  created.tickets.push(...ticketResult.ticketIds);
   230→
   231→  return { booking, payment, providerRef, idempotencyKey, ticketIds: ticketResult.ticketIds, bookingRef };
   232→}
   233→
   234→/**
   235→ * Create a free booking lifecycle setup:
   236→ *   - CONFIRMED booking
   237→ *   - COMPLETED FREE payment
   238→ *   - VALID tickets
   239→ *   - soldCount incremented
   240→ *
   241→ * Returns all created entities for assertions.
   242→ */
   243→async function createFreeBookingConfirmed(
   244→  user: { id: string },
   245→  event: { id: string },
   246→  ticketType: { id: string },
   247→  quantity: number = 2,
   248→) {
   249→  const bookingRef = `BK-FREE-${uid()}`;
   250→
   251→  // 1. Create CONFIRMED booking
   252→  const booking = await db.booking.create({
   253→    data: {
   254→      userId: user.id,
   255→      eventId: event.id,
   256→      totalAmount: 0,
   257→      currency: 'GHS',
   258→      status: 'CONFIRMED',
   259→      bookingRef,
   260→      confirmedAt: new Date(),
   261→    },
   262→  });
   263→  created.bookings.push(booking.id);
   264→
   265→  // 2. Direct sold increment (no reservation for free)
   266→  await directSoldIncrement(ticketType.id, quantity);
   267→
   268→  // 3. Create COMPLETED FREE payment
   269→  const payment = await db.payment.create({
   270→    data: {
   271→      bookingId: booking.id,
   272→      userId: user.id,
   273→      amount: 0,
   274→      currency: 'GHS',
   275→      provider: 'FREE',
   276→      status: 'COMPLETED',
   277→      idempotencyKey: generateIdempotencyKey(),
   278→      completedAt: new Date(),
   279→    },
   280→  });
   281→  created.payments.push(payment.id);
   282→
   283→  // 4. Create VALID tickets directly
   284→  const ticketIds: string[] = [];
   285→  for (let i = 0; i < quantity; i++) {
   286→    const qrCode = `QR-${bookingRef}-${crypto.randomBytes(16).toString('hex').toUpperCase()}`;
   287→    const ticket = await db.ticket.create({
   288→      data: {
   289→        ticketTypeId: ticketType.id,
   290→        bookingId: booking.id,
   291→        qrCode,
   292→        status: 'VALID',
   293→      },
   294→    });
   295→    ticketIds.push(ticket.id);
   296→  }
   297→  created.tickets.push(...ticketIds);
   298→
   299→  return { booking, payment, ticketIds, bookingRef };
   300→}
