   325→
   326→    const event = await prisma.event.create({
   327→      data: {
   328→        title: 'Inventory Test Event',
   329→        slug: `inv-test-${Date.now()}`,

   340→      data: {
   341→        eventId: event.id,

   419→
   420→    // Two concurrent reservations of 2 each — only one should succeed
   421→    const results = await Promise.allSettled([
   422→      reserveInventory({ ticketTypeId: tt.id, quantity: 2 }),
   423→      reserveInventory({ ticketTypeId: tt.id, quantity: 2 }),

   436→    await prisma.ticketType.delete({ where: { id: tt.id } }).catch(() => {});
