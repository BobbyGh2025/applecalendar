   170→describe('Audit 1: processRefundCompletion() atomicity', () => {
   171→  it('updates both Refund.status and Payment.refundedAmount atomically in the same transaction', async () => {
   172→    const fixture = await createCompletedPaymentFixture(5000);
   173→    try {
   174→      // Request a full refund
   175→      const refundResult = await requestRefund({
   176→        paymentId: fixture.payment.id,
   177→        amount: 5000,
   178→        reason: 'Full refund atomicity test',
   179→        requestedBy: fixture.user.id,
   180→      });
   181→
   182→      // Move to PROCESSING (simulating provider interaction)
   183→      await markRefundProcessing(refundResult.refundId);
   184→
   185→      // Before completion: verify initial state
   186→      const refundBefore = await db.refund.findUniqueOrThrow({
   187→        where: { id: refundResult.refundId },
   188→        select: { status: true },
   189→      });
   190→      const paymentBefore = await db.payment.findUniqueOrThrow({
   191→        where: { id: fixture.payment.id },
   192→        select: { refundedAmount: true, status: true },
   193→      });
   194→      expect(refundBefore.status).toBe('PROCESSING');
   195→      expect(paymentBefore.refundedAmount).toBe(0);
   196→      expect(paymentBefore.status).toBe('COMPLETED');
   197→
   198→      // Complete the refund
   199→      const completion = await processRefundCompletion({
   200→        refundId: refundResult.refundId,
   201→        providerRef: 'refund-ref-atomicity',
   202→      });
   203→
   204→      // After completion: verify both records updated atomically
   205→      const refundAfter = await db.refund.findUniqueOrThrow({
   206→        where: { id: refundResult.refundId },
   207→        select: { status: true, processedAt: true, providerRef: true },
   208→      });
   209→      const paymentAfter = await db.payment.findUniqueOrThrow({
   210→        where: { id: fixture.payment.id },
   211→        select: { refundedAmount: true, status: true },
   212→      });
   213→
   214→      // Refund record must be COMPLETED
   215→      expect(refundAfter.status).toBe('COMPLETED');
   216→      expect(refundAfter.processedAt).toBeTruthy();
   217→      expect(refundAfter.providerRef).toBe('refund-ref-atomicity');
   218→
   219→      // Payment record must have refundedAmount incremented
   220→      expect(paymentAfter.refundedAmount).toBe(5000);
   221→
   222→      // Full refund → Payment status must be REFUNDED
   223→      expect(paymentAfter.status).toBe('REFUNDED');
   224→
   225→      // Verify the function returned the correct result
   226→      expect(completion.refundId).toBe(refundResult.refundId);
   227→      expect(completion.status).toBe('COMPLETED');
   228→      expect(completion.amount).toBe(5000);
   229→    } finally {
   230→      await cleanup({ eventId: fixture.event.id, userIds: [fixture.user.id, fixture.organizer.id] });
   231→    }
   232→  });
