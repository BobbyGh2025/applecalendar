   716→/// PaymentWebhookEvent: deduplication and audit for incoming payment webhooks.

   728→  processed       Boolean   @default(false)
   729→  processedAt     DateTime?
   730→  processingError String?
   731→  createdAt       DateTime  @default(now())
   732→
   733→  @@index([processed])
