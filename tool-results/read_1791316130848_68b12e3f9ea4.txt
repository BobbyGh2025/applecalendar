   696→/// PaymentAttempt: tracks each attempt to process a payment through a provider.
   697→/// Enables retry audit trail and provider-specific debugging.
   698→model PaymentAttempt {
   699→  id              String   @id @default(cuid())
   700→  paymentId       String
   701→  provider        String   // PAYSTACK, MANUAL
   702→  providerRef     String?  // Reference from provider for this attempt
   703→  status          String   @default("PENDING") // PENDING, SUCCESS, FAILED
   704→  requestPayload  String?  // JSON — what was sent to provider (redacted secrets)
   705→  responsePayload String?  // JSON — what provider returned
   706→  failureReason   String?
   707→  createdAt       DateTime @default(now())
   708→
   709→  payment Payment @relation(fields: [paymentId], references: [id], onDelete: Cascade)
   710→
   711→  @@index([paymentId])
   712→  @@index([status])
   713→}

   715→/// PaymentWebhookEvent: deduplication and audit for incoming payment webhooks.
   716→/// Prevents double-processing of the same provider event.
   717→/// No FK to Payment — webhook events arrive with providerRef; the service layer
   718→/// resolves the Payment by providerRef. This avoids FK failures for out-of-order events.
   719→model PaymentWebhookEvent {
   720→  id              String    @id @default(cuid())
   721→  provider        String    @default("PAYSTACK")
   722→  eventId         String    @unique // Provider's event ID — dedup key
   723→  eventType       String    // charge.success, charge.failed, refund.processed, etc.
   724→  eventReference  String?   // Provider's transaction reference (links to Payment.providerRef)
   725→  payload         String    // JSON — raw webhook body (for audit/debugging)
   726→  signature       String?   // HMAC signature received (for verification audit)
   727→  processed       Boolean   @default(false)
   728→  processedAt     DateTime?
   729→  processingError String?
   730→  createdAt       DateTime  @default(now())
   731→
   732→  @@index([processed])
   733→  @@index([eventReference])
   734→  @@index([createdAt])
   735→}

   760→/// AuditLog: records all significant admin/system actions for accountability.
   761→/// Never stores passwords, JWTs, refresh tokens, or other secrets.
   762→model AuditLog {
   763→  id          String    @id @default(cuid())
   764→  actorId     String?   // FK → User (null for system actions)
   765→  action      String    // CREATE, UPDATE, DELETE, APPROVE, REJECT, SUSPEND, REINSTATE, CANCEL
   766→  entityType  String    // Event, OrganizerProfile, User, OrganizerSubscription, etc.
   767→  entityId    String    // ID of the affected entity
   768→  oldValue    String?   // JSON of previous state (never secrets)
   769→  newValue    String?   // JSON of new state (never secrets)
   770→  reason      String?   // Reason (for rejections, suspensions)
   771→  ipAddress   String?   // Client IP where safely available
   772→  userAgent   String?   // Client user agent where safely available
   773→  createdAt   DateTime  @default(now())
   774→
   775→  actor User? @relation(fields: [actorId], references: [id])
   776→
   777→  @@index([actorId])
   778→  @@index([entityType, entityId])
   779→  @@index([action])
   780→  @@index([createdAt])
   781→}
